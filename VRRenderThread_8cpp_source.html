117 // HMD-facing movement for any VTK movement events that still reach Dolly3D.
118 void Dolly3D(vtkEventData* ed) override
119 {
120 if (!this->CurrentRenderer)
121 return;
122 auto* rwi = static_cast<vtkRenderWindowInteractor3D*>(this->Interactor);
123 auto* edd = static_cast<vtkEventDataDevice3D*>(ed);
124 vtkCamera* cam = this->CurrentRenderer->GetActiveCamera();
125 if (!cam)
126 return;
127
128 double* trans = rwi->GetPhysicalTranslation(cam);
129 if (!trans)
130 return;
131 if (edd->GetType() == vtkCommand::ViewerMovement3DEvent)
132 edd->GetTrackPadPosition(this->LastTrackPadPosition);
133
134 double speedScale = -this->LastTrackPadPosition[1];
135 double turnAxis = this->LastTrackPadPosition[0];
136 const double turnDeadzone = 0.25;
137 const double moveDeadzone = 0.20;
138 if (std::abs(turnAxis) < turnDeadzone)
139 turnAxis = 0.0;
140 if (std::abs(speedScale) < moveDeadzone)
141 speedScale = 0.0;
142 if (turnAxis == 0.0 && speedScale == 0.0)
143 return;
144
145 double physicalScale = rwi->GetPhysicalScale();
146
147 this->LastDolly3DEventTime->StopTimer();
148 double dt = std::min(this->LastDolly3DEventTime->GetElapsedTime(), 0.1);
149 this->LastDolly3DEventTime->StartTimer();
150
151 double* vd = rwi->GetPhysicalViewDirection();
152 if (vd && turnAxis != 0.0) {
153 const double yaw = turnAxis * 75.0 * dt;
154 const double rad = vtkMath::RadiansFromDegrees(yaw);
155 const double c = std::cos(rad);
156 const double s = std::sin(rad);
157 double nx = c * vd[0] - s * vd[2];
158 double nz = s * vd[0] + c * vd[2];
159 const double len = std::sqrt(nx * nx + vd[1] * vd[1] + nz * nz);
160 if (len > 1e-9)
161 rwi->SetPhysicalViewDirection(nx / len, vd[1] / len, nz / len);
162 }
163
164 double fwd[3];
165 cam->GetDirectionOfProjection(fwd);
166 fwd[1] = 0.0;
167 if (vtkMath::Norm(fwd) <= 1e-9 && vd) {
168 fwd[0] = vd[0];
169 fwd[1] = 0.0;
170 fwd[2] = vd[2];
171 }
172 if (vtkMath::Norm(fwd) <= 1e-9) {
173 fwd[0] = 0.0;
174 fwd[1] = 0.0;
175 fwd[2] = -1.0;
176 }
177 vtkMath::Normalize(fwd);
178
179 double dist = speedScale * this->DollyPhysicalSpeed * physicalScale
180 * g_locomotionScaleCompensation * dt;
181 rwi->SetPhysicalTranslation(cam,
182 trans[0] + fwd[0] * dist,
183 trans[1],
184 trans[2] + fwd[2] * dist);
185 markDollyMovementHandled();
186
187 if (this->AutoAdjustCameraClippingRange)
188 this->CurrentRenderer->ResetCameraClippingRange();
189 }

238 // fires a pick ray from the controller, updates the ray line endpoint and outline
239 void update(const double* pos, const double* ori, vtkRenderer* ren, double maxRay)
240 {
241 if (!ren) return;
242
243 // work out which direction the controller is pointing
244 vtkNew<vtkTransform> xform;
245 xform->RotateWXYZ(ori[0], ori[1], ori[2], ori[3]);
246 double fwd[3] = { 0.0, 0.0, -1.0 };
247 xform->TransformVector(fwd, fwd);
248
249 double p[3] = { pos[0], pos[1], pos[2] };
250 double o[4] = { ori[0], ori[1], ori[2], ori[3] };
251
252 // try to pick an actor along the ray
253 vtkActor* hit = nullptr;
254 double hitPt[3] = { 0.0, 0.0, 0.0 };
255 try {
256 picker->InitializePickList();
257 for (vtkActor* actor : pickActors) {
258 if (actor)
259 picker->AddPickList(actor);
260 }
261 picker->PickFromListOn();
262 picker->Pick3DRay(p, o, ren);
263 hit = vtkActor::SafeDownCast(picker->GetProp3D());
264 double* pt = picker->GetPickPosition();
265 hitPt[0] = pt[0]; hitPt[1] = pt[1]; hitPt[2] = pt[2];
266 } catch (...) {
267 hit = nullptr;
268 }
269
270 // ray goes from controller to hit point (or extends to max length)
271 rayLine->SetPoint1(pos[0], pos[1], pos[2]);
272 if (hit && picker->GetCellId() >= 0) {
273 rayLine->SetPoint2(hitPt[0], hitPt[1], hitPt[2]);
274 } else {
275 rayLine->SetPoint2(
276 pos[0] + fwd[0] * maxRay,
277 pos[1] + fwd[1] * maxRay,
278 pos[2] + fwd[2] * maxRay);
279 }
280 rayLine->Modified();
281
282 // show an outline box around whatever the ray is pointing at
283 if (hit != hoveredActor) {
284 if (hit && hit->GetMapper() && hit->GetMapper()->GetInput()) {
285 outlineFilter->SetInputData(hit->GetMapper()->GetInput());
286 outlineFilter->Update();
287 outlineActor->SetPosition(hit->GetPosition());
288 outlineActor->SetScale(hit->GetScale());
289 outlineActor->SetOrientation(hit->GetOrientation());
290 outlineActor->SetVisibility(1);
291 } else {
292 outlineActor->SetVisibility(0);
293 }
294 hoveredActor = hit;
295 }
296
297 // keep the outline tracking the actor if it moves
298 if (hoveredActor && outlineActor->GetVisibility()) {
299 outlineActor->SetPosition(hoveredActor->GetPosition());
300 outlineActor->SetScale(hoveredActor->GetScale());
301 outlineActor->SetOrientation(hoveredActor->GetOrientation());
302 }
303 }

305 // same as update() but takes a precomputed forward vector from OpenVR pose.
306 void updateFromForward(const double* pos, const double* fwdIn, vtkRenderer* ren, double maxRay)
307 {
308 if (!ren) return;
309
310 double fwd[3] = { fwdIn[0], fwdIn[1], fwdIn[2] };
311 vtkMath::Normalize(fwd);
312
313 vtkActor* hit = nullptr;
314 double hitPt[3] = { 0.0, 0.0, 0.0 };
315 try {
316 picker->InitializePickList();
317 for (vtkActor* actor : pickActors) {
318 if (actor) picker->AddPickList(actor);
319 }
320 picker->PickFromListOn();
321 double p[3] = { pos[0], pos[1], pos[2] };
322 double o[4] = { 0.0, 0.0, 1.0, 0.0 };
323 picker->Pick3DRay(p, o, ren);
324 hit = vtkActor::SafeDownCast(picker->GetProp3D());
325 double* pt = picker->GetPickPosition();
326 hitPt[0] = pt[0]; hitPt[1] = pt[1]; hitPt[2] = pt[2];
327 } catch (...) {
328 hit = nullptr;
329 }
330
331 rayLine->SetPoint1(pos[0], pos[1], pos[2]);
332 if (hit && picker->GetCellId() >= 0) {
333 rayLine->SetPoint2(hitPt[0], hitPt[1], hitPt[2]);
334 } else {
335 rayLine->SetPoint2(
336 pos[0] + fwd[0] * maxRay,
337 pos[1] + fwd[1] * maxRay,
338 pos[2] + fwd[2] * maxRay);
339 }
340 rayLine->Modified();
341
342 if (hit != hoveredActor) {
343 if (hit && hit->GetMapper() && hit->GetMapper()->GetInput()) {
344 outlineFilter->SetInputData(hit->GetMapper()->GetInput());
345 outlineFilter->Update();
346 outlineActor->SetPosition(hit->GetPosition());
347 outlineActor->SetScale(hit->GetScale());
348 outlineActor->SetOrientation(hit->GetOrientation());
349 outlineActor->SetVisibility(1);
350 } else {
351 outlineActor->SetVisibility(0);
352 }
353 hoveredActor = hit;
354 }
355
356 if (hoveredActor && outlineActor->GetVisibility()) {
357 outlineActor->SetPosition(hoveredActor->GetPosition());
358 outlineActor->SetScale(hoveredActor->GetScale());
359 outlineActor->SetOrientation(hoveredActor->GetOrientation());
360 }
361 }

364// handles Move3D events and updates both controller rays each frame
365class VRRayCallback : public vtkCommand {
366public:
367 static VRRayCallback* New() { return new VRRayCallback; }
368
369 vtkRenderer* renderer = nullptr;
370 std::map<int, vtkSmartPointer<vtkActor>>* activeActors = nullptr;
371 std::vector<vtkActor*> menuActors;
372 ControllerRay right;
373 ControllerRay left;
374
375 void init(vtkRenderer* ren)
376 {
377 renderer = ren;
378 right.init(ren, 0.2, 0.85, 1.0); // cyan for right
379 left.init(ren, 1.0, 0.6, 0.2); // orange for left
380 }
381
382 void Execute(vtkObject*, unsigned long eventId, void* callData) override
383 {
384 if (eventId != vtkCommand::Move3DEvent || !renderer)
385 return;
386
387 auto* ed = static_cast<vtkEventData*>(callData);
388 auto* d3d = ed ? ed->GetAsEventDataDevice3D() : nullptr;
389 if (!d3d) return;
390
391 auto device = d3d->GetDevice();
392 const double* pos = d3d->GetWorldPosition();
393 const double* ori = d3d->GetWorldOrientation();
394 auto pickActors = currentPickActors();
395
396 if (device == vtkEventDataDevice::RightController ||
397 device == vtkEventDataDevice::Unknown ||
398 device == vtkEventDataDevice::Any)
399 {
400 right.setPickActors(pickActors);
401 right.update(pos, ori, renderer, MAX_RAY);
402 }
403
404 if (device == vtkEventDataDevice::LeftController ||
405 device == vtkEventDataDevice::Unknown ||
406 device == vtkEventDataDevice::Any)
407 {
408 left.setPickActors(pickActors);
409 left.update(pos, ori, renderer, MAX_RAY);
410 }
411 }

413 // deterministic controller-ray update from OpenVR tracking poses.
414 void tickFromOpenVR(vtkOpenVRRenderWindow* renWin)
415 {
416 if (!renderer || !renWin) return;
417 vr::IVRSystem* hmd = renWin->GetHMD();
418 if (!hmd) return;
419
420 auto pickActors = currentPickActors();
421 right.setPickActors(pickActors);
422 left.setPickActors(pickActors);
423
424 vr::TrackedDevicePose_t poses[vr::k_unMaxTrackedDeviceCount];
425 vr::VRSystem()->GetDeviceToAbsoluteTrackingPose(
426 vr::TrackingUniverseStanding, 0.0f, poses, vr::k_unMaxTrackedDeviceCount);
427
428 auto updateOne = [&](vr::ETrackedControllerRole role, ControllerRay& ray) {
429 vr::TrackedDeviceIndex_t idx = hmd->GetTrackedDeviceIndexForControllerRole(role);
430 if (idx == vr::k_unTrackedDeviceIndexInvalid) return;
431 if (!poses[idx].bPoseIsValid) return;
432 auto& m = poses[idx].mDeviceToAbsoluteTracking;
433 double pos[3] = { m.m[0][3], m.m[1][3], m.m[2][3] };
434 // Controller forward in OpenVR space is -Z axis of the pose matrix.
435 double fwd[3] = { -m.m[0][2], -m.m[1][2], -m.m[2][2] };
436 ray.updateFromForward(pos, fwd, renderer, MAX_RAY);
437 };
438
439 updateOne(vr::TrackedControllerRole_RightHand, right);
440 updateOne(vr::TrackedControllerRole_LeftHand, left);
441 }
442
443private:
444 static constexpr double MAX_RAY = 10000.0; // ray length in mm

446 std::vector<vtkActor*> currentPickActors() const

448 std::vector<vtkActor*> actors;
449 if (activeActors) {
450 for (const auto& [id, actor] : *activeActors) {
451 if (actor)
452 actors.push_back(actor);
453 }
454 }
455 actors.insert(actors.end(), menuActors.begin(), menuActors.end());
456 return actors;
457 }

518 // positions the menu above the given world position, scaled for VR

520 {
521 visible = true;
522 targetPartID = partID;
523
524 // scale button sizes based on physical scale so they're always comfortable
525 double scale = physScale * 0.05; // 5% of physical scale gives good button size
526 double bw = BTN_W * scale;
527 double bh = BTN_H * scale;
528 double gap = GAP * scale;
529
530 // resize all button planes
531 for (auto& b : buttons) {
532 b.actor->SetScale(scale, scale, 1.0);
533 }
534
535 // top row: clip + shrink
536 double startX = x - (2 * bw + gap) * 0.5;
537 double startY = y + 40.0 * scale;
538
539 for (int i = 0; i < 2; i++) {
540 buttons[i].actor->SetPosition(
541 startX + i * (bw + gap), startY, z);
542 buttons[i].actor->SetVisibility(1);
543 }
544
545 // bottom row: colour swatches
546 double row2Y = startY - bh - gap;
547 int nCol = static_cast<int>(buttons.size()) - 2;
548 double row2StartX = x - (nCol * bw + (nCol - 1) * gap) * 0.5;
549 for (int i = 2; i < static_cast<int>(buttons.size()); i++) {
550 buttons[i].actor->SetPosition(
551 row2StartX + (i - 2) * (bw + gap), row2Y, z);
552 buttons[i].actor->SetVisibility(1);
553 }
554 }

778 void show(vtkRenderer* ren, vtkOpenVRRenderWindow* win)
779 {
780 if (!panelPlane || !panelActor || !ren)
781 return;
782
783 rebuildTexture();
784
785 vtkCamera* cam = ren->GetActiveCamera();
786 if (!cam)
787 return;
788
789 double* camPos = cam->GetPosition();
790 double* camFwd = cam->GetDirectionOfProjection();
791 double* camUp = cam->GetViewUp();
792 double fwd[3] = { camFwd[0], camFwd[1], camFwd[2] };
793 double up[3] = { camUp[0], camUp[1], camUp[2] };
794 vtkMath::Normalize(fwd);
795 vtkMath::Normalize(up);
796
797 double right[3];
798 vtkMath::Cross(fwd, up, right);

800 right[0] = 1.0;
801 right[1] = 0.0;
802 right[2] = 0.0;

806 vtkMath::Normalize(up);
807
808 const double physicalScale = win ? win->GetPhysicalScale() : 1.0;
809 const double width = physicalScale * 0.55;
810 const double height = physicalScale * 0.32;
811 double center[3] = {
812 camPos[0] + fwd[0] * physicalScale * 0.75,
813 camPos[1] + fwd[1] * physicalScale * 0.75 - up[1] * physicalScale * 0.04,
814 camPos[2] + fwd[2] * physicalScale * 0.75
815 };

818 panelCenter[i] = center[i];
819 panelNormal[i] = -fwd[i];
820 panelRight[i] = right[i];
821 panelUp[i] = up[i];

824 panelHeight = height;

827 center[1] - right[1] * width * 0.5 - up[1] * height * 0.5,
828 center[2] - right[2] * width * 0.5 - up[2] * height * 0.5);
829 panelPlane->SetPoint1(center[0] + right[0] * width * 0.5 - up[0] * height * 0.5,
830 center[1] + right[1] * width * 0.5 - up[1] * height * 0.5,
831 center[2] + right[2] * width * 0.5 - up[2] * height * 0.5);
832 panelPlane->SetPoint2(center[0] - right[0] * width * 0.5 + up[0] * height * 0.5,
833 center[1] - right[1] * width * 0.5 + up[1] * height * 0.5,
834 center[2] - right[2] * width * 0.5 + up[2] * height * 0.5);
835 panelPlane->Modified();
836 panelActor->SetVisibility(1);
837 visible = true;

840 HitResult hitTest(const double rayOrigin[3], const double rayDir[3]) const
841 {
842 if (!visible)
843 return {};
844 const double denom = vtkMath::Dot(panelNormal, rayDir);
845 if (std::abs(denom) < 1e-9)
846 return {};
847 double toPanel[3] = {
848 panelCenter[0] - rayOrigin[0],
849 panelCenter[1] - rayOrigin[1],
850 panelCenter[2] - rayOrigin[2]
851 };
852 const double t = vtkMath::Dot(toPanel, panelNormal) / denom;
853 if (t < 0.01 || t > panelWidth * 3.0)
854 return {};
855 double hit[3] = {
856 rayOrigin[0] + rayDir[0] * t,
857 rayOrigin[1] + rayDir[1] * t,
858 rayOrigin[2] + rayDir[2] * t
859 };
860 double local[3] = {
861 hit[0] - panelCenter[0],
862 hit[1] - panelCenter[1],
863 hit[2] - panelCenter[2]
864 };
865 const double u = vtkMath::Dot(local, panelRight) + panelWidth * 0.5;
866 const double v = vtkMath::Dot(local, panelUp) + panelHeight * 0.5;
867 if (u < 0.0 || u > panelWidth || v < 0.0 || v > panelHeight)
868 return {};
869
870 const double px = (u / panelWidth) * TexW;
871 const double py = (v / panelHeight) * TexH;
872 for (int i = 0; i < static_cast<int>(partIds.size()); ++i) {
873 const int row = i / CardCols;
874 const int col = i % CardCols;
875 const int x0 = CardStartX + col * (CardSize + CardGap);
876 const int y0 = CardStartY - row * (CardSize + CardGap);
877 if (px >= x0 && px <= x0 + CardSize && py >= y0 && py <= y0 + CardSize)
878 return { HitKind::Part, i, QColor() };
879 }
880
881 if (selectedPartID() >= 0) {
882 for (int slot = 0; slot < 8; ++slot) {
883 const int x0 = ControlStartX + slot * (ControlSize + ControlGap);
884 const int y0 = ControlY;
885 if (px < x0 || px > x0 + ControlSize || py < y0 || py > y0 + ControlSize)
886 continue;
887 switch (slot) {
888 case 0:
889 return { HitKind::Visible, -1, QColor() };
890 case 1:
891 return { HitKind::Clip, -1, QColor() };
892 case 2:
893 return { HitKind::Shrink, -1, QColor() };
894 case 3:
895 return { HitKind::Colour, -1, QColor(255, 50, 50) };
896 case 4:
897 return { HitKind::Colour, -1, QColor(50, 200, 50) };
898 case 5:
899 return { HitKind::Colour, -1, QColor(50, 110, 255) };
900 case 6:
901 return { HitKind::Colour, -1, QColor(255, 230, 50) };
902 case 7:
903 return { HitKind::Colour, -1, QColor(235, 235, 235) };
904 }
905 }
906 }
907 return { HitKind::Panel, -1, QColor() };
908 }

910 HitResult hitFromRayLine(vtkLineSource* line) const
911 {
912 if (!line)
913 return {};
914
915 double origin[3];
916 double end[3];
917 line->GetPoint1(origin);
918 line->GetPoint2(end);
919 double dir[3] = {
920 end[0] - origin[0],
921 end[1] - origin[1],
922 end[2] - origin[2]
923 };
924 if (vtkMath::Norm(dir) <= 1e-9)
925 return {};
926 vtkMath::Normalize(dir);
927
928 return hitTest(origin, dir);
929 }
930
931 HitResult hitFromRays(VRRayCallback* rays) const
932 {
933 if (!rays)
934 return {};
935 HitResult hit = hitFromRayLine(rays->right.rayLine);
936 if (hit.kind != HitKind::None)
937 return hit;
938 return hitFromRayLine(rays->left.rayLine);
939 }

1174// hold trigger on a hovered part to drag it; release to drop

1176struct TriggerGrabber {
1177 bool active = false;
1178 vtkActor* targetActor = nullptr;
1179 vr::TrackedDeviceIndex_t activeController = vr::k_unTrackedDeviceIndexInvalid;
1180 double localOffset[3] = { 0.0, 0.0, 0.0 };
1181
1182 bool isPartActor(vtkActor* a, std::map<int, vtkSmartPointer<vtkActor>>* actors)
1183 {
1184 if (!a || !actors) return false;
1185 for (auto& [id, act] : *actors)
1186 if (act.Get() == a) return true;
1187 return false;

1190 static bool isTriggerPressed(const vr::VRControllerState_t& state)
1191 {
1192 const bool digital = (state.ulButtonPressed & vr::ButtonMaskFromId(vr::k_EButton_SteamVR_Trigger)) != 0;
1193 const bool analog = state.rAxis[1].x > 0.55f;
1194 return digital || analog;
1195 }

1197 static bool getControllerPos(vr::TrackedDeviceIndex_t idx, vr::TrackedDevicePose_t* poses, double out[3])
1198 {
1199 if (idx >= vr::k_unMaxTrackedDeviceCount || !poses[idx].bPoseIsValid)
1200 return false;
1201 auto& m = poses[idx].mDeviceToAbsoluteTracking;
1202 out[0] = m.m[0][3];
1203 out[1] = m.m[1][3];
1204 out[2] = m.m[2][3];
1205 return true;
1206 }

1208 void update(vtkOpenVRRenderWindow* renWin, VRRayCallback* rays, std::map<int, vtkSmartPointer<vtkActor>>* activeActors)
1209 {
1210 vr::IVRSystem* hmd = renWin ? renWin->GetHMD() : nullptr;
1211 if (!hmd || !rays || !activeActors) return;
1212
1213 vr::TrackedDeviceIndex_t leftIdx =
1214 hmd->GetTrackedDeviceIndexForControllerRole(vr::TrackedControllerRole_LeftHand);
1215 vr::TrackedDeviceIndex_t rightIdx =
1216 hmd->GetTrackedDeviceIndexForControllerRole(vr::TrackedControllerRole_RightHand);
1217
1218 vr::VRControllerState_t lState{}, rState{};
1219 bool hasLeft = (leftIdx != vr::k_unTrackedDeviceIndexInvalid) &&
1220 hmd->GetControllerState(leftIdx, &lState, sizeof(lState));
1221 bool hasRight = (rightIdx != vr::k_unTrackedDeviceIndexInvalid) &&
1222 hmd->GetControllerState(rightIdx, &rState, sizeof(rState));
1223
1224 const bool leftTrigger = hasLeft && isTriggerPressed(lState);
1225 const bool rightTrigger = hasRight && isTriggerPressed(rState);
1226
1227 vr::TrackedDevicePose_t poses[vr::k_unMaxTrackedDeviceCount];
1228 vr::VRSystem()->GetDeviceToAbsoluteTrackingPose(
1229 vr::TrackingUniverseStanding, 0.0f, poses, vr::k_unMaxTrackedDeviceCount);
1230
1231 if (active) {
1232 bool stillPressed = false;
1233 if (activeController == leftIdx) stillPressed = leftTrigger;
1234 if (activeController == rightIdx) stillPressed = rightTrigger;
1235 if (!stillPressed || !targetActor) {
1236 active = false;
1237 targetActor = nullptr;
1238 activeController = vr::k_unTrackedDeviceIndexInvalid;
1239 return;
1240 }
1241
1242 double ctrlPos[3];
1243 if (!getControllerPos(activeController, poses, ctrlPos)) {
1244 active = false;
1245 targetActor = nullptr;
1246 activeController = vr::k_unTrackedDeviceIndexInvalid;
1247 return;
1248 }
1249
1250 targetActor->SetPosition(
1251 ctrlPos[0] + localOffset[0],
1252 ctrlPos[1] + localOffset[1],
1253 ctrlPos[2] + localOffset[2]);
1254 return;
1255 }
1256
1257 if (rightTrigger && rightIdx != vr::k_unTrackedDeviceIndexInvalid &&
1258 isPartActor(rays->right.hoveredActor, activeActors))
1259 {
1260 double ctrlPos[3];
1261 if (!getControllerPos(rightIdx, poses, ctrlPos)) return;
1262 targetActor = rays->right.hoveredActor;
1263 double* p = targetActor->GetPosition();
1264 localOffset[0] = p[0] - ctrlPos[0];
1265 localOffset[1] = p[1] - ctrlPos[1];
1266 localOffset[2] = p[2] - ctrlPos[2];
1267 activeController = rightIdx;
1268 active = true;
1269 return;
1270 }
1271
1272 if (leftTrigger && leftIdx != vr::k_unTrackedDeviceIndexInvalid &&
1273 isPartActor(rays->left.hoveredActor, activeActors))
1274 {
1275 double ctrlPos[3];
1276 if (!getControllerPos(leftIdx, poses, ctrlPos)) return;
1277 targetActor = rays->left.hoveredActor;
1278 double* p = targetActor->GetPosition();
1279 localOffset[0] = p[0] - ctrlPos[0];
1280 localOffset[1] = p[1] - ctrlPos[1];
1281 localOffset[2] = p[2] - ctrlPos[2];
1282 activeController = leftIdx;
1283 active = true;
1284 }
1285 }

1296
1297 static bool movementAxis(vr::IVRSystem* hmd, vr::ETrackedControllerRole role, double axis[2])
1298 {

1300 axis[1] = 0.0;
1301 if (!hmd) return false;
1302
1303 vr::TrackedDeviceIndex_t idx = hmd->GetTrackedDeviceIndexForControllerRole(role);
1304 if (idx == vr::k_unTrackedDeviceIndexInvalid) return false;

1307 if (!hmd->GetControllerState(idx, &state, sizeof(state))) return false;
1308
1309 axis[0] = state.rAxis[0].x;
1310 axis[1] = state.rAxis[0].y;
1311 return true;
1312 }
1313
1314 void update(vtkOpenVRRenderWindow* renWin, vtkOpenVRRenderWindowInteractor* interactor,
1315 vtkRenderer* renderer)
1316 {
1317 if (!renWin || !interactor || !renderer) return;
1318 vtkCamera* cam = renderer->GetActiveCamera();
1319 if (!cam) return;
1320
1321 if (!timerStarted) {
1322 timer->StartTimer();
1323 timerStarted = true;
1324 return;
1325 }
1326
1327 timer->StopTimer();
1328 const double dt = std::min(timer->GetElapsedTime(), 0.1);
1329 timer->StartTimer();
1330
1331 // If VTK delivered a movement event this frame, Dolly3D already handled it.
1332 if (dollyHandledMovementRecently())
1333 return;
1334
1335 vr::IVRSystem* hmd = renWin->GetHMD();
1336 if (!hmd) return;
1337
1338 double left[2], right[2];
1339 const bool hasLeft = movementAxis(hmd, vr::TrackedControllerRole_LeftHand, left);
1340 const bool hasRight = movementAxis(hmd, vr::TrackedControllerRole_RightHand, right);
1341
1342 double axis[2] = { 0.0, 0.0 };
1343 if (hasLeft) {
1344 axis[0] += left[0];
1345 axis[1] += left[1];
1346 }
1347 if (hasRight) {
1348 axis[0] += right[0];
1349 axis[1] += right[1];
1350 }
1351
1352 axis[0] = std::clamp(axis[0], -1.0, 1.0);
1353 axis[1] = std::clamp(axis[1], -1.0, 1.0);
1354
1355 const double deadzone = 0.18;
1356 if (std::abs(axis[0]) < deadzone) axis[0] = 0.0;
1357 if (std::abs(axis[1]) < deadzone) axis[1] = 0.0;
1358 if (axis[0] == 0.0 && axis[1] == 0.0) return;
1359
1360 double* viewDir = interactor->GetPhysicalViewDirection();
1361 if (viewDir && axis[0] != 0.0) {
1362 const double yaw = axis[0] * 75.0 * dt;
1363 const double rad = vtkMath::RadiansFromDegrees(yaw);
1364 const double c = std::cos(rad);
1365 const double s = std::sin(rad);
1366 double nx = c * viewDir[0] - s * viewDir[2];
1367 double nz = s * viewDir[0] + c * viewDir[2];
1368 const double len = std::sqrt(nx * nx + viewDir[1] * viewDir[1] + nz * nz);
1369 if (len > 1e-9) {
1370 interactor->SetPhysicalViewDirection(nx / len, viewDir[1] / len, nz / len);
1371 }
1372 }
1373
1374 if (axis[1] != 0.0) {
1375 double fwd[3];
1376 cam->GetDirectionOfProjection(fwd);
1377 fwd[1] = 0.0;
1378 if (vtkMath::Norm(fwd) <= 1e-9 && viewDir) {
1379 fwd[0] = viewDir[0];
1380 fwd[1] = 0.0;
1381 fwd[2] = viewDir[2];
1382 }
1383 if (vtkMath::Norm(fwd) <= 1e-9) {
1384 fwd[0] = 0.0;
1385 fwd[1] = 0.0;
1386 fwd[2] = -1.0;
1387 }
1388 vtkMath::Normalize(fwd);
1389
1390 double* trans = interactor->GetPhysicalTranslation(cam);
1391 if (!trans) return;
1392
1393 const double dist = -axis[1] * 2.0 * interactor->GetPhysicalScale()
1394 * g_locomotionScaleCompensation * dt;
1395 interactor->SetPhysicalTranslation(cam,
1396 trans[0] + fwd[0] * dist,
1397 trans[1],
1398 trans[2] + fwd[2] * dist);
1399 renderer->ResetCameraClippingRange();
1400 }
1401 }
1402};

1527 preImportActors.insert(a);

1547 bool isPartActor = false;

1600 // fallback: 2D skybox if no 3D garage available

1657 garageBounds[0], garageBounds[1], garageBounds[2], garageBounds[3],
1658 garageBounds[4], garageBounds[5],

1676 m_garageFloorY = garageFloorY;

1684 m_spawnY = viewerHeightOffset;

1742 std::clamp(desiredPartSpan / partMaxSpan, 0.03, 5.0) * g_stlSizeBoost,

1866// executes a single command from the GUI
1867void VRRenderThread::applyCommand(const CommandPacket& cmd) {
1868 switch (cmd.type) {
1869 case Command::EndRender:
1870 m_endRender = true;
1871 break;
1872 case Command::SetColour: {
1873 auto it = m_activeActors.find(cmd.partID);
1874 if (it != m_activeActors.end()) {
1875 QColor c = cmd.data.value<QColor>();
1876 it->second->GetProperty()->SetColor(c.redF(), c.greenF(), c.blueF());
1877 }
1878 break;
1879 }
1880 case Command::SetVisible: {
1881 auto it = m_activeActors.find(cmd.partID);
1882 if (it != m_activeActors.end()) {
1883 it->second->SetVisibility(cmd.data.toBool() ? 1 : 0);
1884 }
1885 break;
1886 }
1887 case Command::RotateY: {
1888 auto it = m_activeActors.find(cmd.partID);
1889 if (it != m_activeActors.end()) {
1890 it->second->RotateY(cmd.data.toDouble());
1891 }
1892 break;
1893 }
1894 case Command::SetTransform: {
1895 auto it = m_activeActors.find(cmd.partID);
1896 if (it != m_activeActors.end()) {
1897 QVector3D v = cmd.data.value<QVector3D>();
1898 it->second->SetPosition(v.x(), v.y(), v.z());
1899 }
1900 break;
1901 }
1902 case Command::ToggleShrink:
1903 case Command::ToggleClip: {
1904 if (!m_partList) break;
1905 ModelPart* p = m_partList->findByID(cmd.partID);
1906 if (!p) break;
1907 p->rebuildVRPipeline();
1908 break;
1909 }
1910 case Command::AddActor: {
1911 if (cmd.actor && m_renderer) {
1912 m_renderer->AddActor(cmd.actor);
1913 m_activeActors[cmd.partID] = cmd.actor;
1914 cmd.actor->PickableOn();
1915 cmd.actor->DragableOn();
1916
1917 double actorBounds[6];
1918 cmd.actor->GetBounds(actorBounds);
1919 const double partCx = (actorBounds[0] + actorBounds[1]) * 0.5;
1920 const double partCy = (actorBounds[2] + actorBounds[3]) * 0.5;
1921 const double partCz = (actorBounds[4] + actorBounds[5]) * 0.5;
1922 const double partMaxSpan = std::max({
1923 actorBounds[1] - actorBounds[0],
1924 actorBounds[3] - actorBounds[2],
1925 actorBounds[5] - actorBounds[4],
1926 1e-6
1927 });
1928 double stlScale = 0.10;
1929 if (m_hasGarageAnchor) {
1930 // Keep STL size stable even when user/world scale changes.
1931 const double desiredPartSpan = 0.75 * g_stlReferencePhysicalScale;
1932 stlScale = std::clamp(
1933 std::clamp(desiredPartSpan / partMaxSpan, 0.03, 5.0) * g_stlSizeBoost,
1934 0.03, 5.0);
1935 }
1936
1937 const double partBaseY = m_hasGarageAnchor ? m_garageFloorY : 0.0;
1938 double targetX = m_spawnX;
1939 double targetZ = m_spawnZ;
1940
1941 if (m_renderWindow && m_renderWindow->GetHMD()) {
1942 vr::TrackedDevicePose_t poses[vr::k_unMaxTrackedDeviceCount];
1943 vr::VRSystem()->GetDeviceToAbsoluteTrackingPose(
1944 vr::TrackingUniverseStanding, 0.0f, poses, vr::k_unMaxTrackedDeviceCount);
1945 const auto hmdIdx = vr::k_unTrackedDeviceIndex_Hmd;
1946 if (poses[hmdIdx].bPoseIsValid) {
1947 const auto& mat = poses[hmdIdx].mDeviceToAbsoluteTracking;
1948 double fwd[3] = { -mat.m[0][2], -mat.m[1][2], -mat.m[2][2] };
1949 vtkMath::Normalize(fwd);
1950 targetX = mat.m[0][3] + fwd[0] * 1.0;
1951 targetZ = mat.m[2][3] + fwd[2] * 1.0;
1952 }
1953 }
1954
1955 qDebug() << "[VR AddActor] part" << cmd.partID
1956 << "bounds:" << actorBounds[0] << actorBounds[1]
1957 << actorBounds[2] << actorBounds[3]
1958 << actorBounds[4] << actorBounds[5]
1959 << "scale:" << stlScale
1960 << "target:" << targetX << partBaseY << targetZ;
1961 writeVrDebug("a",
1962 "[VR AddActor]\n"
1963 "id=%d target=(%f,%f,%f) partBaseY=%f scale=%f bounds=[%f %f] [%f %f] [%f %f]\n",
1964 cmd.partID, targetX, partBaseY, targetZ, partBaseY, stlScale,
1965 actorBounds[0], actorBounds[1], actorBounds[2], actorBounds[3],
1966 actorBounds[4], actorBounds[5]);
1967
1968 cmd.actor->SetUserMatrix(nullptr);
1969 cmd.actor->SetUserTransform(nullptr);
1970 cmd.actor->SetOrientation(0.0, 0.0, 0.0);
1971 cmd.actor->SetScale(stlScale, stlScale, stlScale);
1972 cmd.actor->SetPosition(
1973 targetX - partCx * stlScale,
1974 partBaseY - actorBounds[2] * stlScale,
1975 targetZ - partCz * stlScale);
1976 cmd.actor->SetVisibility(1);
1977
1978 double finalB[6];
1979 cmd.actor->GetBounds(finalB);
1980 qDebug() << "[VR AddActor] final bounds:" << finalB[0] << finalB[1]
1981 << finalB[2] << finalB[3] << finalB[4] << finalB[5];
1982
1983 m_renderer->ResetCameraClippingRange();
1984 }
1985 break;
1986 }
1987 case Command::RemoveActor: {
1988 auto it = m_activeActors.find(cmd.partID);
1989 if (it != m_activeActors.end() && m_renderer) {
1990 m_renderer->RemoveActor(it->second);
1991 m_activeActors.erase(it);
1992 m_renderer->ResetCameraClippingRange();
1993 }
1994 break;
1995 }
1996 case Command::ToggleSkybox: {
1997 if (m_skybox && m_renderer) {
1998 if (cmd.data.toBool())
1999 m_renderer->AddActor(m_skybox);
2000 else
2001 m_renderer->RemoveActor(m_skybox);
2002 m_renderer->ResetCameraClippingRange();
2003 }
2004 break;
2005 }
2006 }
