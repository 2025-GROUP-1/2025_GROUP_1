69 // overrides dolly to negate Y direction (trackpad-up = away, down = towards)
70 void Dolly3D(vtkEventData* ed) override
71 {
72 if (!this->CurrentRenderer)
73 return;
74 auto* rwi = static_cast<vtkRenderWindowInteractor3D*>(this->Interactor);
75 auto* edd = static_cast<vtkEventDataDevice3D*>(ed);
76 const double* wori = edd->GetWorldOrientation();
77
78 // convert controller orientation to a forward vector
79 vtkQuaternion<double> q1;
80 q1.SetRotationAngleAndAxis(vtkMath::RadiansFromDegrees(wori[0]), wori[1], wori[2], wori[3]);
81 double elem[3][3];
82 q1.ToMatrix3x3(elem);
83 double vdir[3] = { 0.0, 0.0, -1.0 };
84 vtkMatrix3x3::MultiplyPoint(elem[0], vdir, vdir);
85
86 double* trans = rwi->GetPhysicalTranslation(this->CurrentRenderer->GetActiveCamera());
87 if (edd->GetType() == vtkCommand::ViewerMovement3DEvent)
88 edd->GetTrackPadPosition(this->LastTrackPadPosition);
89
90 // negate so up = push away, down = pull closer
91 double speedScale = -this->LastTrackPadPosition[1];
92 double physicalScale = rwi->GetPhysicalScale();
93
94 this->LastDolly3DEventTime->StopTimer();
95 double dist = speedScale * this->DollyPhysicalSpeed * physicalScale
96 * this->LastDolly3DEventTime->GetElapsedTime();
97 this->LastDolly3DEventTime->StartTimer();
98
99 rwi->SetPhysicalTranslation(this->CurrentRenderer->GetActiveCamera(),
100 trans[0] - vdir[0] * dist,
101 trans[1] - vdir[1] * dist,
102 trans[2] - vdir[2] * dist);
103
104 if (this->AutoAdjustCameraClippingRange)
105 this->CurrentRenderer->ResetCameraClippingRange();
106 }

108 // -----------------------------------------------------------------------
109 // Dual-stick locomotion - polls OpenVR joysticks directly so it works
110 // even when the VTK action manifest fails to load (legacy input mode).
111 //
112 // Left stick: walk (forward/back + strafe left/right) on flat ground
113 // Right stick X: snap turn (45-degree increments)
114 // Right stick Y: dolly (push/pull scene)
115 //
116 // Movement is FLAT (horizontal only) and clamped to floor level.
117 // Works on Quest Touch controllers AND Vive Pro wands (trackpad).
118 // -----------------------------------------------------------------------
119 double m_floorY = 0.0;
120 bool m_snapTurnReady = true; // prevents repeated snap turns while held
121
122 void setFloorY(double y) { m_floorY = y; }
123
124 void pollJoystickDolly(vtkOpenVRRenderWindow* renWin)
125 {
126 if (!this->CurrentRenderer || !renWin)
127 return;
128
129 vr::IVRSystem* hmd = renWin->GetHMD();
130 if (!hmd) return;
131
132 auto* rwi = static_cast<vtkRenderWindowInteractor3D*>(this->Interactor);
133 if (!rwi) return;
134
135 vtkCamera* cam = this->CurrentRenderer->GetActiveCamera();
136 if (!cam) return;
137
138 double physicalScale = rwi->GetPhysicalScale();
139 static constexpr float DEADZONE = 0.2f;
140 double moveSpeed = physicalScale * 0.015;
141
142 // use the play-space forward direction (not head tilt) for horizontal movement
143 double* physDir = renWin->GetPhysicalViewDirection();
144 // project physical view direction onto horizontal plane (Y=0)
145 double fwd[3] = { physDir[0], 0.0, physDir[2] };
146 double fLen = sqrt(fwd[0]*fwd[0] + fwd[2]*fwd[2]);
147 if (fLen > 1e-6) { fwd[0] /= fLen; fwd[2] /= fLen; }
148 // right vector perpendicular to forward on the ground
149 double right[3] = { fwd[2], 0.0, -fwd[0] };
150
151 // --- LEFT STICK: walk (flat ground only) ---
152 vr::TrackedDeviceIndex_t leftIdx =
153 hmd->GetTrackedDeviceIndexForControllerRole(vr::TrackedControllerRole_LeftHand);
154
155 if (leftIdx != vr::k_unTrackedDeviceIndexInvalid) {
156 vr::VRControllerState_t lState;
157 if (hmd->GetControllerState(leftIdx, &lState, sizeof(lState))) {
158 float lx = lState.rAxis[0].x;
159 float ly = lState.rAxis[0].y;
160
161 if (std::abs(lx) <= DEADZONE) lx = 0.0f;
162 if (std::abs(ly) <= DEADZONE) ly = 0.0f;
163
164 if (lx != 0.0f || ly != 0.0f) {
165 double dx = (fwd[0] * ly + right[0] * lx) * moveSpeed;
166 double dz = (fwd[2] * ly + right[2] * lx) * moveSpeed;
167
168 double* trans = renWin->GetPhysicalTranslation();
169 double newX = trans[0] + dx;
170 double newZ = trans[2] + dz;
171
172 // clamp to floor - don't allow going below floor level
173 renWin->SetPhysicalTranslation(newX, m_floorY, newZ);
174 }
175 }
176 }
177
178 // --- RIGHT STICK: snap turn (X) + dolly (Y) ---
179 vr::TrackedDeviceIndex_t rightIdx =
180 hmd->GetTrackedDeviceIndexForControllerRole(vr::TrackedControllerRole_RightHand);
181
182 if (rightIdx != vr::k_unTrackedDeviceIndexInvalid) {
183 vr::VRControllerState_t rState;
184 if (hmd->GetControllerState(rightIdx, &rState, sizeof(rState))) {
185 float rx = rState.rAxis[0].x;
186 float ry = rState.rAxis[0].y;
187
188 // snap turn: 45 degrees per flick
189 // rotates the physical view direction AND the translation
190 // so the user stays in place but the world rotates around them
191 if (std::abs(rx) > 0.7f) {
192 if (m_snapTurnReady) {
193 double angle = (rx > 0) ? -45.0 : 45.0;
194 double rad = angle * vtkMath::Pi() / 180.0;
195 double cosA = cos(rad);
196 double sinA = sin(rad);
197
198 // rotate physical view direction around Y
199 double newDir[3] = {
200 physDir[0] * cosA - physDir[2] * sinA,
201 physDir[1],
202 physDir[0] * sinA + physDir[2] * cosA
203 };
204 renWin->SetPhysicalViewDirection(newDir[0], newDir[1], newDir[2]);
205
206 // also rotate the physical view up if it's not pure Y-up
207 double* viewUp = renWin->GetPhysicalViewUp();
208 double newUp[3] = {
209 viewUp[0] * cosA - viewUp[2] * sinA,
210 viewUp[1],
211 viewUp[0] * sinA + viewUp[2] * cosA
212 };
213 renWin->SetPhysicalViewUp(newUp[0], newUp[1], newUp[2]);
214
215 m_snapTurnReady = false;
216 }
217 } else {
218 m_snapTurnReady = true;
219 }
220
221 // right stick Y = dolly (push/pull scene along ground)
222 if (std::abs(ry) > DEADZONE) {
223 double speed = static_cast<double>(ry) * moveSpeed;
224 double* trans = renWin->GetPhysicalTranslation();
225 renWin->SetPhysicalTranslation(
226 trans[0] + fwd[0] * speed,
227 m_floorY,
228 trans[2] + fwd[2] * speed);
229 }
230 }
231 }
232 }

240struct ControllerRay {
241 vtkNew<vtkCellPicker> picker;
242 vtkNew<vtkLineSource> rayLine;
243 vtkNew<vtkActor> rayActor;
244 vtkNew<vtkOutlineFilter> outlineFilter;
245 vtkNew<vtkPolyDataMapper> outlineMapper;
246 vtkNew<vtkActor> outlineActor;
247 vtkActor* hoveredActor = nullptr;
248 std::vector<vtkActor*> pickActors;
249
250 // creates the ray line and outline actors, adds them to the renderer
251 void init(vtkRenderer* ren, double r, double g, double b)
252 {
253 picker->SetTolerance(0.005);
254
255 rayLine->SetPoint1(0, 0, 0);
256 rayLine->SetPoint2(0, 0, -1);
257 vtkNew<vtkPolyDataMapper> rayMapper;
258 rayMapper->SetInputConnection(rayLine->GetOutputPort());
259 rayActor->SetMapper(rayMapper);
260 rayActor->GetProperty()->SetColor(r, g, b);
261 rayActor->GetProperty()->SetLineWidth(2.0);
262 rayActor->GetProperty()->SetOpacity(0.7);
263 rayActor->PickableOff();
264 ren->AddActor(rayActor);
265
266 outlineMapper->SetInputConnection(outlineFilter->GetOutputPort());
267 outlineActor->SetMapper(outlineMapper);
268 outlineActor->GetProperty()->SetColor(0.1, 1.0, 0.4);
269 outlineActor->GetProperty()->SetLineWidth(3.0);
270 outlineActor->PickableOff();
271 outlineActor->SetVisibility(0);
272 ren->AddActor(outlineActor);
273 }
274
275 void setPickActors(const std::vector<vtkActor*>& actors)
276 {
277 pickActors = actors;
278 }

280 // fires a pick ray from the controller, updates the ray line endpoint and outline
281 void update(const double* pos, const double* ori, vtkRenderer* ren, double maxRay)
282 {
283 if (!ren) return;
284
285 // work out which direction the controller is pointing
286 vtkNew<vtkTransform> xform;
287 xform->RotateWXYZ(ori[0], ori[1], ori[2], ori[3]);
288 double fwd[3] = { 0.0, 0.0, -1.0 };
289 xform->TransformVector(fwd, fwd);
290
291 double p[3] = { pos[0], pos[1], pos[2] };
292 double o[4] = { ori[0], ori[1], ori[2], ori[3] };
293
294 // try to pick an actor along the ray
295 vtkActor* hit = nullptr;
296 double hitPt[3] = { 0.0, 0.0, 0.0 };
297 try {
298 picker->InitializePickList();
299 for (vtkActor* actor : pickActors) {
300 if (actor)
301 picker->AddPickList(actor);
302 }
303 picker->PickFromListOn();
304 picker->Pick3DRay(p, o, ren);
305 hit = vtkActor::SafeDownCast(picker->GetProp3D());
306 double* pt = picker->GetPickPosition();
307 hitPt[0] = pt[0]; hitPt[1] = pt[1]; hitPt[2] = pt[2];
308 } catch (...) {
309 hit = nullptr;
310 }
311
312 // ray goes from controller to hit point (or extends to max length)
313 rayLine->SetPoint1(pos[0], pos[1], pos[2]);
314 if (hit && picker->GetCellId() >= 0) {
315 rayLine->SetPoint2(hitPt[0], hitPt[1], hitPt[2]);
316 } else {
317 rayLine->SetPoint2(
318 pos[0] + fwd[0] * maxRay,
319 pos[1] + fwd[1] * maxRay,
320 pos[2] + fwd[2] * maxRay);
321 }
322 rayLine->Modified();
323
324 // show an outline box around whatever the ray is pointing at
325 if (hit != hoveredActor) {
326 if (hit && hit->GetMapper() && hit->GetMapper()->GetInput()) {
327 outlineFilter->SetInputData(hit->GetMapper()->GetInput());
328 outlineFilter->Update();
329 outlineActor->SetPosition(hit->GetPosition());
330 outlineActor->SetScale(hit->GetScale());
331 outlineActor->SetOrientation(hit->GetOrientation());
332 outlineActor->SetVisibility(1);
333 } else {
334 outlineActor->SetVisibility(0);
335 }
336 hoveredActor = hit;
337 }
338
339 // keep the outline tracking the actor if it moves
340 if (hoveredActor && outlineActor->GetVisibility()) {
341 outlineActor->SetPosition(hoveredActor->GetPosition());
342 outlineActor->SetScale(hoveredActor->GetScale());
343 outlineActor->SetOrientation(hoveredActor->GetOrientation());
344 }
345 }
346};

348// handles Move3D events and updates both controller rays each frame
349class VRRayCallback : public vtkCommand {
350public:
351 static VRRayCallback* New() { return new VRRayCallback; }
352
353 vtkRenderer* renderer = nullptr;
354 std::map<int, vtkSmartPointer<vtkActor>>* activeActors = nullptr;
355 std::vector<vtkActor*> menuActors;
356 ControllerRay right;
357 ControllerRay left;
358
359 void init(vtkRenderer* ren)
360 {
361 renderer = ren;
362 right.init(ren, 0.2, 0.85, 1.0); // cyan for right
363 left.init(ren, 1.0, 0.6, 0.2); // orange for left
364 }
365
366 void Execute(vtkObject*, unsigned long eventId, void* callData) override
367 {
368 if (eventId != vtkCommand::Move3DEvent || !renderer)
369 return;
370
371 auto* ed = static_cast<vtkEventData*>(callData);
372 auto* d3d = ed ? ed->GetAsEventDataDevice3D() : nullptr;
373 if (!d3d) return;
374
375 auto device = d3d->GetDevice();
376 const double* pos = d3d->GetWorldPosition();
377 const double* ori = d3d->GetWorldOrientation();
378 auto pickActors = currentPickActors();
379
380 if (device == vtkEventDataDevice::RightController ||
381 device == vtkEventDataDevice::Unknown ||
382 device == vtkEventDataDevice::Any)
383 {
384 right.setPickActors(pickActors);
385 right.update(pos, ori, renderer, MAX_RAY);
386 }
387
388 if (device == vtkEventDataDevice::LeftController ||
389 device == vtkEventDataDevice::Unknown ||
390 device == vtkEventDataDevice::Any)
391 {
392 left.setPickActors(pickActors);
393 left.update(pos, ori, renderer, MAX_RAY);
394 }
395 }
396
397private:
398 static constexpr double MAX_RAY = 10000.0; // ray length in mm
399
400 std::vector<vtkActor*> currentPickActors() const
401 {
402 std::vector<vtkActor*> actors;
403 if (activeActors) {
404 for (const auto& [id, actor] : *activeActors) {
405 if (actor)
406 actors.push_back(actor);
407 }
408 }
409 actors.insert(actors.end(), menuActors.begin(), menuActors.end());
410 return actors;
411 }

414// ---------------------------------------------------------------------------
415// in-VR editing menu
416// shows coloured buttons above a hovered part for clip/shrink/colour changes
417// ---------------------------------------------------------------------------
418struct VRMenu {
419 enum Action { Clip, Shrink, ColRed, ColGreen, ColBlue, ColYellow, ColWhite, NUM_ACTIONS };
420
421 struct Button {
422 vtkSmartPointer<vtkActor> actor;
423 Action action;
424 };
425
426 std::vector<Button> buttons;
427 bool visible = false;
428 int targetPartID = -1;
429 vtkRenderer* ren = nullptr;
430
431 // button sizes (scaled by physical scale at show time)
432 double BTN_W = 60.0;
433 double BTN_H = 30.0;
434 double GAP = 8.0;
435
436 // builds all the menu button actors
437 void create(vtkRenderer* r)
438 {
439 ren = r;
440 addButton(Clip, 1.0, 0.5, 0.0); // orange
441 addButton(Shrink, 0.6, 0.2, 0.8); // purple
442 addButton(ColRed, 1.0, 0.15, 0.15);
443 addButton(ColGreen, 0.15, 0.8, 0.15);
444 addButton(ColBlue, 0.15, 0.4, 1.0);
445 addButton(ColYellow, 1.0, 1.0, 0.2);
446 addButton(ColWhite, 0.9, 0.9, 0.9);
447 }
448
449 void addButton(Action a, double cr, double cg, double cb)
450 {
451 vtkNew<vtkPlaneSource> plane;
452 plane->SetOrigin(0, 0, 0);
453 plane->SetPoint1(BTN_W, 0, 0);
454 plane->SetPoint2(0, BTN_H, 0);
455 plane->Update();
456
457 vtkNew<vtkPolyDataMapper> m;
458 m->SetInputConnection(plane->GetOutputPort());
459
460 vtkSmartPointer<vtkActor> act = vtkSmartPointer<vtkActor>::New();
461 act->SetMapper(m);
462 act->GetProperty()->SetColor(cr, cg, cb);
463 act->GetProperty()->SetAmbient(1.0);
464 act->GetProperty()->SetDiffuse(0.0);
465 act->SetVisibility(0);
466 act->PickableOn();
467 ren->AddActor(act);
468
469 buttons.push_back({act, a});
470 }
471
472 // positions the menu above the given world position, scaled for VR
473 void show(double x, double y, double z, int partID, double physScale = 1.0)
474 {
475 visible = true;
476 targetPartID = partID;
477
478 // scale button sizes based on physical scale so they're always comfortable
479 double scale = physScale * 0.05; // 5% of physical scale gives good button size
480 double bw = BTN_W * scale;
481 double bh = BTN_H * scale;
482 double gap = GAP * scale;
483
484 // resize all button planes
485 for (auto& b : buttons) {
486 b.actor->SetScale(scale, scale, 1.0);
487 }
488
489 // top row: clip + shrink
490 double startX = x - (2 * bw + gap) * 0.5;
491 double startY = y + 40.0 * scale;
492
493 for (int i = 0; i < 2; i++) {
494 buttons[i].actor->SetPosition(
495 startX + i * (bw + gap), startY, z);
496 buttons[i].actor->SetVisibility(1);
497 }
498
499 // bottom row: colour swatches
500 double row2Y = startY - bh - gap;
501 int nCol = static_cast<int>(buttons.size()) - 2;
502 double row2StartX = x - (nCol * bw + (nCol - 1) * gap) * 0.5;
503 for (int i = 2; i < static_cast<int>(buttons.size()); i++) {
504 buttons[i].actor->SetPosition(
505 row2StartX + (i - 2) * (bw + gap), row2Y, z);
506 buttons[i].actor->SetVisibility(1);
507 }
508 }
509
510 void hide()
511 {
512 visible = false;
513 targetPartID = -1;
514 for (auto& b : buttons)
515 b.actor->SetVisibility(0);
516 }
517
518 int findButton(vtkActor* a) const
519 {
520 for (int i = 0; i < static_cast<int>(buttons.size()); i++)
521 if (buttons[i].actor.Get() == a)
522 return i;
523 return -1;
524 }
525
526 bool isMenuActor(vtkActor* a) const { return findButton(a) >= 0; }
527
528 std::vector<vtkActor*> actors() const
529 {
530 std::vector<vtkActor*> result;
531 for (const auto& button : buttons)
532 result.push_back(button.actor);
533 return result;
534 }
535};

537// handles menu button presses - fires when the user hits the menu button on the controller
538class VRMenuCallback : public vtkCommand {
539public:
540 static VRMenuCallback* New() { return new VRMenuCallback; }
541
542 VRRayCallback* rays = nullptr;
543 VRMenu menu;
544 std::map<int, vtkSmartPointer<vtkActor>>* activeActors = nullptr;
545 ModelPartList* partList = nullptr;
546 VRRenderThread* vrThread = nullptr;
547
548 vtkOpenVRRenderWindow* renWindow = nullptr;
549
550 void Execute(vtkObject*, unsigned long, void*) override
551 {
552 // check what the controller is pointing at
553 vtkActor* hovered = nullptr;
554 if (rays->right.hoveredActor)
555 hovered = rays->right.hoveredActor;
556 else if (rays->left.hoveredActor)
557 hovered = rays->left.hoveredActor;
558
559 // if pointing at a menu button, run its action
560 int btnIdx = menu.findButton(hovered);
561 if (btnIdx >= 0) {
562 executeAction(menu.buttons[btnIdx]);
563 return;
564 }
565
566 // if pointing at a part, toggle the menu for that part
567 if (hovered) {
568 int pid = findPartID(hovered);
569 if (pid >= 0) {
570 if (menu.visible && menu.targetPartID == pid) {
571 menu.hide();
572 } else {
573 double* center = hovered->GetCenter();
574 double physScale = 1.0;
575 if (renWindow)
576 physScale = renWindow->GetPhysicalScale();
577 menu.show(center[0], center[1], center[2], pid, physScale);
578 }
579 return;
580 }
581 }
582
583 // pointing at nothing - close menu
584 if (menu.visible)
585 menu.hide();
586 }
587
588private:
589 // looks up which part ID an actor belongs to
590 int findPartID(vtkActor* a) const
591 {
592 if (!activeActors) return -1;
593 for (auto& [id, act] : *activeActors)
594 if (act.Get() == a) return id;
595 return -1;
596 }
597
598 // runs the selected menu action on the target part (VR thread - signals GUI to sync)
599 void executeAction(const VRMenu::Button& btn)
600 {
601 if (!partList || menu.targetPartID < 0) return;
602 ModelPart* part = partList->findByID(menu.targetPartID);
603 if (!part) return;
604
605 switch (btn.action) {
606 case VRMenu::Clip: {
607 bool newState = !part->getClipEnabled();
608 part->m_clipEnabled = newState;
609 part->rebuildVRPipeline();
610 if (vrThread) emit vrThread->partClipChanged(menu.targetPartID, newState);
611 break;
612 }
613 case VRMenu::Shrink: {
614 bool newState = !part->getShrinkEnabled();
615 part->m_shrinkEnabled = newState;
616 part->rebuildVRPipeline();
617 if (vrThread) emit vrThread->partShrinkChanged(menu.targetPartID, newState);
618 break;
619 }
620 case VRMenu::ColRed:
621 applyColourVR(part, QColor(255, 50, 50));
622 break;
623 case VRMenu::ColGreen:
624 applyColourVR(part, QColor(50, 200, 50));
625 break;
626 case VRMenu::ColBlue:
627 applyColourVR(part, QColor(50, 100, 255));
628 break;
629 case VRMenu::ColYellow:
630 applyColourVR(part, QColor(255, 255, 50));
631 break;
632 case VRMenu::ColWhite:
633 applyColourVR(part, QColor(240, 240, 240));
634 break;
635 default:
636 break;
637 }
638 }
639
640 // updates VR actor and signals GUI to apply the same colour on its side
641 void applyColourVR(ModelPart* part, const QColor& c)
642 {
643 part->m_colour = c;
644 vtkActor* vr = part->getVRActor();
645 if (vr)
646 vr->GetProperty()->SetColor(c.redF(), c.greenF(), c.blueF());
647 if (vrThread) emit vrThread->partColourChanged(menu.targetPartID, c);
648 }

676// thread-safe command queue - GUI pushes, VR loop pops

688// the main VR loop - sets up OpenVR, adds actors, runs render loop
689void VRRenderThread::run() {
690 m_renderer = vtkSmartPointer<vtkOpenVRRenderer>::New();
691 m_renderWindow = vtkSmartPointer<vtkOpenVRRenderWindow>::New();
692 m_interactor = vtkSmartPointer<vtkOpenVRRenderWindowInteractor>::New();
693
694 // point the interactor at our SteamVR action bindings
695 const std::string manifestPath =
696 (QCoreApplication::applicationDirPath() + "/vrbindings/vtk_openvr_actions.json").toStdString();
697 m_interactor->SetActionManifestFileName(manifestPath.c_str());
698 m_interactor->SetActionSetName("/actions/vtk");
699
700 // custom style: blocks VTK's built-in menu, inverts dolly, hides default rays
701 vtkNew<VRCustomStyle> customStyle;
702 m_interactor->SetInteractorStyle(customStyle);
703
704 m_renderWindow->AddRenderer(m_renderer);
705 m_interactor->SetRenderWindow(m_renderWindow);
706
707 // add all the parts that were queued before the thread started
708 for (auto& [id, actor] : m_pendingActors) {
709 m_renderer->AddActor(actor);
710 m_activeActors[id] = actor;
711 }
712 m_pendingActors.clear();
713
714 m_renderer->ResetCamera();
715 m_renderer->ResetCameraClippingRange();
716
717 // set up dual-controller ray casting + outline highlight
718 vtkNew<VRRayCallback> rayCallback;
719 rayCallback->init(m_renderer);
720 rayCallback->activeActors = &m_activeActors;
721 m_interactor->AddObserver(vtkCommand::Move3DEvent, rayCallback, 1.0);
722
723 // in-VR editing menu (clip / shrink / colour buttons)
724 vtkNew<VRMenuCallback> menuCallback;
725 menuCallback->rays = rayCallback;
726 menuCallback->activeActors = &m_activeActors;
727 menuCallback->partList = m_partList;
728 menuCallback->vrThread = this;
729 menuCallback->renWindow = m_renderWindow;
730 menuCallback->menu.create(m_renderer);
731 rayCallback->menuActors = menuCallback->menu.actors();
732 m_interactor->AddObserver(vtkCommand::UserEvent, menuCallback, 2.0);
733
734 m_renderWindow->Initialize();
735
736 // load 3D garage environment (garage__warehouse.glb or garage.glb) as actual geometry
737 QString glbPath;
738 QStringList glbNames = { "garage__warehouse.glb", "garage.glb" };
739 QStringList searchDirs = {
740 QCoreApplication::applicationDirPath(),
741 QCoreApplication::applicationDirPath() + "/..",
742 QCoreApplication::applicationDirPath() + "/../..",
743 QCoreApplication::applicationDirPath() + "/../../.."
744 };
745 for (const auto& dir : searchDirs) {
746 for (const auto& name : glbNames) {
747 QString candidate = dir + "/" + name;
748 if (QFile::exists(candidate)) { glbPath = candidate; break; }
749 }
750 if (!glbPath.isEmpty()) break;
751 }
752
753 bool garageLoaded = false;
754 if (!glbPath.isEmpty()) {
755 vtkNew<vtkGLTFImporter> gltfImporter;
756 gltfImporter->SetFileName(glbPath.toStdString().c_str());
757 gltfImporter->SetRenderWindow(m_renderWindow);
758
759 // import the 3D environment - adds actors, textures, materials, lighting
760 gltfImporter->Update();
761
762 // mark all imported actors as non-pickable (environment only)
763 // and collect them so we can position the environment correctly
764 std::vector<vtkActor*> envActors;
765 vtkActorCollection* actors = m_renderer->GetActors();
766 actors->InitTraversal();
767 vtkActor* a;
768 while ((a = actors->GetNextActor()) != nullptr) {
769 bool isPartActor = false;
770 for (auto& [id, partAct] : m_activeActors) {
771 if (partAct.Get() == a) { isPartActor = true; break; }
772 }
773 if (!isPartActor) {
774 a->PickableOff();
775 envActors.push_back(a);
776 }
777 }
778
779 // position the garage so the user's STL parts sit inside it at a reasonable height
780 // GLTF uses metres, VTK OpenVR also uses metres, so typically no scale is needed.
781 // If the garage is too large or small relative to the STL, adjust it.
782 if (!envActors.empty()) {
783 // compute bounds of all environment geometry
784 double envBounds[6] = { 1e30, -1e30, 1e30, -1e30, 1e30, -1e30 };
785 for (auto* ea : envActors) {
786 double b[6];
787 ea->GetBounds(b);
788 if (b[0] < envBounds[0]) envBounds[0] = b[0];
789 if (b[1] > envBounds[1]) envBounds[1] = b[1];
790 if (b[2] < envBounds[2]) envBounds[2] = b[2];
791 if (b[3] > envBounds[3]) envBounds[3] = b[3];
792 if (b[4] < envBounds[4]) envBounds[4] = b[4];
793 if (b[5] > envBounds[5]) envBounds[5] = b[5];
794 }
795
796 // compute bounds of loaded STL parts
797 double partBounds[6] = { 1e30, -1e30, 1e30, -1e30, 1e30, -1e30 };
798 bool hasParts = false;
799 for (auto& [id, partAct] : m_activeActors) {
800 double b[6];
801 partAct->GetBounds(b);
802 if (b[0] < partBounds[0]) partBounds[0] = b[0];
803 if (b[1] > partBounds[1]) partBounds[1] = b[1];
804 if (b[2] < partBounds[2]) partBounds[2] = b[2];
805 if (b[3] > partBounds[3]) partBounds[3] = b[3];
806 if (b[4] < partBounds[4]) partBounds[4] = b[4];
807 if (b[5] > partBounds[5]) partBounds[5] = b[5];
808 hasParts = true;
809 }
810
811 // if the garage is significantly different scale from parts, rescale it
812 // so the garage wraps around the parts comfortably
813 if (hasParts) {
814 double envSize = std::max({ envBounds[1]-envBounds[0],
815 envBounds[3]-envBounds[2],
816 envBounds[5]-envBounds[4] });
817 double partSize = std::max({ partBounds[1]-partBounds[0],
818 partBounds[3]-partBounds[2],
819 partBounds[5]-partBounds[4] });
820
821 // target: garage should be ~5x bigger than the parts
822 if (envSize > 1e-6 && partSize > 1e-6) {
823 double targetEnvSize = partSize * 5.0;
824 double scaleFactor = targetEnvSize / envSize;
825
826 // only rescale if the difference is significant (>2x off)
827 if (scaleFactor < 0.5 || scaleFactor > 2.0) {
828 for (auto* ea : envActors)
829 ea->SetScale(scaleFactor, scaleFactor, scaleFactor);
830 }
831 }
832
833 // centre the garage around the parts
834 double partCx = (partBounds[0] + partBounds[1]) * 0.5;
835 double partCy = partBounds[2]; // floor = bottom of parts
836 double partCz = (partBounds[4] + partBounds[5]) * 0.5;
837
838 double envCx = (envBounds[0] + envBounds[1]) * 0.5;
839 double envCy = envBounds[2]; // floor of garage
840 double envCz = (envBounds[4] + envBounds[5]) * 0.5;
841
842 double offsetX = partCx - envCx;
843 double offsetY = partCy - envCy;
844 double offsetZ = partCz - envCz;
845
846 for (auto* ea : envActors) {
847 double* pos = ea->GetPosition();
848 ea->SetPosition(pos[0] + offsetX, pos[1] + offsetY, pos[2] + offsetZ);
849 }
850 }
851 }
852
853 garageLoaded = true;
854 }
855
856 // fallback: 2D skybox if no 3D garage available
857 if (!garageLoaded) {
858 QString garagePath2D = QCoreApplication::applicationDirPath() + "/room2.png";
859 QString roomPath = QCoreApplication::applicationDirPath() + "/room.jpg";
860
861 vtkSmartPointer<vtkImageReader2> bgReader;
862 vtkNew<vtkPNGReader> pngReader;
863 vtkNew<vtkJPEGReader> jpgReader;
864
865 if (pngReader->CanReadFile(garagePath2D.toStdString().c_str())) {
866 pngReader->SetFileName(garagePath2D.toStdString().c_str());
867 pngReader->Update();
868 bgReader = pngReader;
869 } else if (jpgReader->CanReadFile(roomPath.toStdString().c_str())) {
870 jpgReader->SetFileName(roomPath.toStdString().c_str());
871 jpgReader->Update();
872 bgReader = jpgReader;
873 }
874
875 if (bgReader) {
876 vtkNew<vtkTexture> bgTexture;
877 bgTexture->SetInputConnection(bgReader->GetOutputPort());
878 bgTexture->InterpolateOn();
879
880 vtkNew<vtkSkybox> skybox;
881 skybox->SetTexture(bgTexture);
882 skybox->SetProjectionToSphere();
883 skybox->PickableOff();
884 m_renderer->AddActor(skybox);
885
886 m_skybox = skybox;
887 }
888 }
889
890 try {
891 m_interactor->Initialize();
892 } catch (...) {
893 // VR input system may not be available
894 }
895
896 if (m_renderer->GetActiveCamera())
897 m_renderer->ResetCameraClippingRange();
898
899 // position user standing on the floor, not inside it
900 // shifts VR origin so real-world floor = bottom of scene geometry
901 {
902 double bounds[6];
903 m_renderer->ComputeVisiblePropBounds(bounds);
904 double* trans = m_renderWindow->GetPhysicalTranslation();
905 double floorY = bounds[2]; // Y-min of all geometry
906 m_renderWindow->SetPhysicalTranslation(trans[0], floorY, trans[2]);
907 customStyle->setFloorY(floorY);
908 }
909
910 // render loop - process one VR frame, handle commands, repeat
911 // NOTE: DoOneEvent already submits frames to both eyes.
912 // An extra Render() causes double-submit which results in single-eye flickering.
913 m_endRender = false;
914 while (!m_endRender) {
915 m_interactor->DoOneEvent(m_renderWindow, m_renderer);
916 processCommands();
917
918 // fallback joystick locomotion - polls OpenVR directly so it works
919 // even when the action manifest fails to load (legacy input mode on Quest)
920 customStyle->pollJoystickDolly(m_renderWindow);
921 }
922
923 m_renderWindow->Finalize();
924 m_renderer = nullptr;
925 m_renderWindow = nullptr;
926 m_interactor = nullptr;
927}

937// executes a single command from the GUI
938void VRRenderThread::applyCommand(const CommandPacket& cmd) {
939 switch (cmd.type) {
940 case Command::EndRender:
941 m_endRender = true;
942 break;
943 case Command::SetColour: {
944 auto it = m_activeActors.find(cmd.partID);
945 if (it != m_activeActors.end()) {
946 QColor c = cmd.data.value<QColor>();
947 it->second->GetProperty()->SetColor(c.redF(), c.greenF(), c.blueF());
948 }
949 break;
950 }
951 case Command::SetVisible: {
952 auto it = m_activeActors.find(cmd.partID);
953 if (it != m_activeActors.end()) {
954 it->second->SetVisibility(cmd.data.toBool() ? 1 : 0);
955 }
956 break;
957 }
958 case Command::RotateY: {
959 auto it = m_activeActors.find(cmd.partID);
960 if (it != m_activeActors.end()) {
961 it->second->RotateY(cmd.data.toDouble());
962 }
963 break;
964 }
965 case Command::SetTransform: {
966 auto it = m_activeActors.find(cmd.partID);
967 if (it != m_activeActors.end()) {
968 QVector3D v = cmd.data.value<QVector3D>();
969 it->second->SetPosition(v.x(), v.y(), v.z());
970 }
971 break;
972 }
973 case Command::ToggleShrink:
974 case Command::ToggleClip: {
975 if (!m_partList) break;
976 ModelPart* p = m_partList->findByID(cmd.partID);
977 if (!p) break;
978 p->rebuildVRPipeline();
979 break;
980 }
981 case Command::AddActor: {
982 if (cmd.actor && m_renderer) {
983 m_renderer->AddActor(cmd.actor);
984 m_activeActors[cmd.partID] = cmd.actor;
985 m_renderer->ResetCameraClippingRange();
986 }
987 break;
988 }
989 case Command::RemoveActor: {
990 auto it = m_activeActors.find(cmd.partID);
991 if (it != m_activeActors.end() && m_renderer) {
992 m_renderer->RemoveActor(it->second);
993 m_activeActors.erase(it);
994 m_renderer->ResetCameraClippingRange();
995 }
996 break;
997 }
998 case Command::ToggleSkybox: {
999 if (m_skybox && m_renderer) {
1000 if (cmd.data.toBool())
1001 m_renderer->AddActor(m_skybox);
1002 else
1003 m_renderer->RemoveActor(m_skybox);
1004 m_renderer->ResetCameraClippingRange();
1005 }
1006 break;
1007 }
1008 }
