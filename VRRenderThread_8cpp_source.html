46// ---------------------------------------------------------------------------
47// custom VR interactor style
48// - blocks VTK's built-in probe/clip/exit menu
49// - fires a UserEvent instead so our menu callback gets it
50// - kills VTK's default red rays (we draw our own)
51// - inverts dolly direction so trackpad-up = push away
52// ---------------------------------------------------------------------------
53class VRCustomStyle : public vtkOpenVRInteractorStyle {
54public:
55 static VRCustomStyle* New();
56 vtkTypeMacro(VRCustomStyle, vtkOpenVRInteractorStyle);
57
58 // replaces VTK's built-in menu with our custom event
59 void OnMenu3D(vtkEventData* edata) override
60 {
61 if (this->Interactor)
62 this->Interactor->InvokeEvent(vtkCommand::UserEvent, edata);
63 }
64
65 // no-op so VTK never draws its red rays
66 void UpdateRay(vtkEventDataDevice) override {}
67
68 // overrides dolly to negate Y direction (trackpad-up = away, down = towards)
69 void Dolly3D(vtkEventData* ed) override
70 {
71 if (!this->CurrentRenderer)
72 return;
73 auto* rwi = static_cast<vtkRenderWindowInteractor3D*>(this->Interactor);
74 auto* edd = static_cast<vtkEventDataDevice3D*>(ed);
75 const double* wori = edd->GetWorldOrientation();
76
77 // convert controller orientation to a forward vector
78 vtkQuaternion<double> q1;
79 q1.SetRotationAngleAndAxis(vtkMath::RadiansFromDegrees(wori[0]), wori[1], wori[2], wori[3]);
80 double elem[3][3];
81 q1.ToMatrix3x3(elem);
82 double vdir[3] = { 0.0, 0.0, -1.0 };
83 vtkMatrix3x3::MultiplyPoint(elem[0], vdir, vdir);
84
85 double* trans = rwi->GetPhysicalTranslation(this->CurrentRenderer->GetActiveCamera());
86 if (edd->GetType() == vtkCommand::ViewerMovement3DEvent)
87 edd->GetTrackPadPosition(this->LastTrackPadPosition);
88
89 // negate so up = push away, down = pull closer
90 double speedScale = -this->LastTrackPadPosition[1];
91 double physicalScale = rwi->GetPhysicalScale();
92
93 this->LastDolly3DEventTime->StopTimer();
94 double dist = speedScale * this->DollyPhysicalSpeed * physicalScale
95 * this->LastDolly3DEventTime->GetElapsedTime();
96 this->LastDolly3DEventTime->StartTimer();
97
98 rwi->SetPhysicalTranslation(this->CurrentRenderer->GetActiveCamera(),
99 trans[0] - vdir[0] * dist,
100 trans[1] - vdir[1] * dist,
101 trans[2] - vdir[2] * dist);
102
103 if (this->AutoAdjustCameraClippingRange)
104 this->CurrentRenderer->ResetCameraClippingRange();
105 }
106
107};

111// controller ray + outline highlight
112// each controller gets a coloured ray and an outline box around hovered actors

114struct ControllerRay {
115 vtkNew<vtkCellPicker> picker;
116 vtkNew<vtkLineSource> rayLine;
117 vtkNew<vtkActor> rayActor;
118 vtkNew<vtkOutlineFilter> outlineFilter;
119 vtkNew<vtkPolyDataMapper> outlineMapper;
120 vtkNew<vtkActor> outlineActor;
121 vtkActor* hoveredActor = nullptr;
122 std::vector<vtkActor*> pickActors;
123
124 // creates the ray line and outline actors, adds them to the renderer
125 void init(vtkRenderer* ren, double r, double g, double b)
126 {
127 picker->SetTolerance(0.005);
128
129 rayLine->SetPoint1(0, 0, 0);
130 rayLine->SetPoint2(0, 0, -1);
131 vtkNew<vtkPolyDataMapper> rayMapper;
132 rayMapper->SetInputConnection(rayLine->GetOutputPort());
133 rayActor->SetMapper(rayMapper);
134 rayActor->GetProperty()->SetColor(r, g, b);
135 rayActor->GetProperty()->SetLineWidth(2.0);
136 rayActor->GetProperty()->SetOpacity(0.7);
137 rayActor->PickableOff();
138 ren->AddActor(rayActor);
139
140 outlineMapper->SetInputConnection(outlineFilter->GetOutputPort());
141 outlineActor->SetMapper(outlineMapper);
142 outlineActor->GetProperty()->SetColor(0.1, 1.0, 0.4);
143 outlineActor->GetProperty()->SetLineWidth(3.0);
144 outlineActor->PickableOff();
145 outlineActor->SetVisibility(0);
146 ren->AddActor(outlineActor);
147 }
148
149 void setPickActors(const std::vector<vtkActor*>& actors)
150 {
151 pickActors = actors;
152 }
153
154 // fires a pick ray from the controller, updates the ray line endpoint and outline
155 void update(const double* pos, const double* ori, vtkRenderer* ren, double maxRay)
156 {
157 if (!ren) return;
158
159 // work out which direction the controller is pointing
160 vtkNew<vtkTransform> xform;
161 xform->RotateWXYZ(ori[0], ori[1], ori[2], ori[3]);
162 double fwd[3] = { 0.0, 0.0, -1.0 };
163 xform->TransformVector(fwd, fwd);
164
165 double p[3] = { pos[0], pos[1], pos[2] };
166 double o[4] = { ori[0], ori[1], ori[2], ori[3] };
167
168 // try to pick an actor along the ray
169 vtkActor* hit = nullptr;
170 double hitPt[3] = { 0.0, 0.0, 0.0 };
171 try {
172 picker->InitializePickList();
173 for (vtkActor* actor : pickActors) {
174 if (actor)
175 picker->AddPickList(actor);
176 }
177 picker->PickFromListOn();
178 picker->Pick3DRay(p, o, ren);
179 hit = vtkActor::SafeDownCast(picker->GetProp3D());
180 double* pt = picker->GetPickPosition();
181 hitPt[0] = pt[0]; hitPt[1] = pt[1]; hitPt[2] = pt[2];
182 } catch (...) {
183 hit = nullptr;
184 }
185
186 // ray goes from controller to hit point (or extends to max length)
187 rayLine->SetPoint1(pos[0], pos[1], pos[2]);
188 if (hit && picker->GetCellId() >= 0) {
189 rayLine->SetPoint2(hitPt[0], hitPt[1], hitPt[2]);
190 } else {
191 rayLine->SetPoint2(
192 pos[0] + fwd[0] * maxRay,
193 pos[1] + fwd[1] * maxRay,
194 pos[2] + fwd[2] * maxRay);
195 }
196 rayLine->Modified();
197
198 // show an outline box around whatever the ray is pointing at
199 if (hit != hoveredActor) {
200 if (hit && hit->GetMapper() && hit->GetMapper()->GetInput()) {
201 outlineFilter->SetInputData(hit->GetMapper()->GetInput());
202 outlineFilter->Update();
203 outlineActor->SetPosition(hit->GetPosition());
204 outlineActor->SetScale(hit->GetScale());
205 outlineActor->SetOrientation(hit->GetOrientation());
206 outlineActor->SetVisibility(1);
207 } else {
208 outlineActor->SetVisibility(0);
209 }
210 hoveredActor = hit;
211 }
212
213 // keep the outline tracking the actor if it moves
214 if (hoveredActor && outlineActor->GetVisibility()) {
215 outlineActor->SetPosition(hoveredActor->GetPosition());
216 outlineActor->SetScale(hoveredActor->GetScale());
217 outlineActor->SetOrientation(hoveredActor->GetOrientation());
218 }
219 }
220};

222// handles Move3D events and updates both controller rays each frame
223class VRRayCallback : public vtkCommand {
224public:
225 static VRRayCallback* New() { return new VRRayCallback; }
226
227 vtkRenderer* renderer = nullptr;
228 std::map<int, vtkSmartPointer<vtkActor>>* activeActors = nullptr;
229 std::vector<vtkActor*> menuActors;
230 ControllerRay right;
231 ControllerRay left;
232
233 void init(vtkRenderer* ren)
234 {
235 renderer = ren;
236 right.init(ren, 0.2, 0.85, 1.0); // cyan for right
237 left.init(ren, 1.0, 0.6, 0.2); // orange for left
238 }
239
240 void Execute(vtkObject*, unsigned long eventId, void* callData) override
241 {
242 if (eventId != vtkCommand::Move3DEvent || !renderer)
243 return;
244
245 auto* ed = static_cast<vtkEventData*>(callData);
246 auto* d3d = ed ? ed->GetAsEventDataDevice3D() : nullptr;
247 if (!d3d) return;
248
249 auto device = d3d->GetDevice();
250 const double* pos = d3d->GetWorldPosition();
251 const double* ori = d3d->GetWorldOrientation();
252 auto pickActors = currentPickActors();
253
254 if (device == vtkEventDataDevice::RightController ||
255 device == vtkEventDataDevice::Unknown ||
256 device == vtkEventDataDevice::Any)
257 {
258 right.setPickActors(pickActors);
259 right.update(pos, ori, renderer, MAX_RAY);
260 }
261
262 if (device == vtkEventDataDevice::LeftController ||
263 device == vtkEventDataDevice::Unknown ||
264 device == vtkEventDataDevice::Any)
265 {
266 left.setPickActors(pickActors);
267 left.update(pos, ori, renderer, MAX_RAY);
268 }
269 }
270
271private:
272 static constexpr double MAX_RAY = 10000.0; // ray length in mm
273
274 std::vector<vtkActor*> currentPickActors() const
275 {
276 std::vector<vtkActor*> actors;
277 if (activeActors) {
278 for (const auto& [id, actor] : *activeActors) {
279 if (actor)
280 actors.push_back(actor);
281 }
282 }
283 actors.insert(actors.end(), menuActors.begin(), menuActors.end());
284 return actors;
285 }

289// in-VR editing menu
290// shows coloured buttons above a hovered part for clip/shrink/colour changes
291// ---------------------------------------------------------------------------
292struct VRMenu {
293 enum Action { Clip, Shrink, ColRed, ColGreen, ColBlue, ColYellow, ColWhite, NUM_ACTIONS };
294
295 struct Button {
296 vtkSmartPointer<vtkActor> actor;
297 Action action;
298 };
299
300 std::vector<Button> buttons;
301 bool visible = false;
302 int targetPartID = -1;
303 vtkRenderer* ren = nullptr;
304
305 // button sizes (scaled by physical scale at show time)
306 double BTN_W = 60.0;
307 double BTN_H = 30.0;
308 double GAP = 8.0;
309
310 // builds all the menu button actors
311 void create(vtkRenderer* r)
312 {
313 ren = r;
314 addButton(Clip, 1.0, 0.5, 0.0); // orange
315 addButton(Shrink, 0.6, 0.2, 0.8); // purple
316 addButton(ColRed, 1.0, 0.15, 0.15);
317 addButton(ColGreen, 0.15, 0.8, 0.15);
318 addButton(ColBlue, 0.15, 0.4, 1.0);
319 addButton(ColYellow, 1.0, 1.0, 0.2);
320 addButton(ColWhite, 0.9, 0.9, 0.9);
321 }
322
323 void addButton(Action a, double cr, double cg, double cb)
324 {
325 vtkNew<vtkPlaneSource> plane;
326 plane->SetOrigin(0, 0, 0);
327 plane->SetPoint1(BTN_W, 0, 0);
328 plane->SetPoint2(0, BTN_H, 0);
329 plane->Update();
330
331 vtkNew<vtkPolyDataMapper> m;
332 m->SetInputConnection(plane->GetOutputPort());
333
334 vtkSmartPointer<vtkActor> act = vtkSmartPointer<vtkActor>::New();
335 act->SetMapper(m);
336 act->GetProperty()->SetColor(cr, cg, cb);
337 act->GetProperty()->SetAmbient(1.0);
338 act->GetProperty()->SetDiffuse(0.0);
339 act->SetVisibility(0);
340 act->PickableOn();
341 ren->AddActor(act);
342
343 buttons.push_back({act, a});
344 }
345
346 // positions the menu above the given world position, scaled for VR
347 void show(double x, double y, double z, int partID, double physScale = 1.0)
348 {
349 visible = true;
350 targetPartID = partID;
351
352 // scale button sizes based on physical scale so they're always comfortable
353 double scale = physScale * 0.05; // 5% of physical scale gives good button size
354 double bw = BTN_W * scale;
355 double bh = BTN_H * scale;
356 double gap = GAP * scale;
357
358 // resize all button planes
359 for (auto& b : buttons) {
360 b.actor->SetScale(scale, scale, 1.0);
361 }
362
363 // top row: clip + shrink
364 double startX = x - (2 * bw + gap) * 0.5;
365 double startY = y + 40.0 * scale;
366
367 for (int i = 0; i < 2; i++) {
368 buttons[i].actor->SetPosition(
369 startX + i * (bw + gap), startY, z);
370 buttons[i].actor->SetVisibility(1);
371 }
372
373 // bottom row: colour swatches
374 double row2Y = startY - bh - gap;
375 int nCol = static_cast<int>(buttons.size()) - 2;
376 double row2StartX = x - (nCol * bw + (nCol - 1) * gap) * 0.5;
377 for (int i = 2; i < static_cast<int>(buttons.size()); i++) {
378 buttons[i].actor->SetPosition(
379 row2StartX + (i - 2) * (bw + gap), row2Y, z);
380 buttons[i].actor->SetVisibility(1);
381 }
382 }
383
384 void hide()
385 {
386 visible = false;
387 targetPartID = -1;
388 for (auto& b : buttons)
389 b.actor->SetVisibility(0);
390 }
391
392 int findButton(vtkActor* a) const
393 {
394 for (int i = 0; i < static_cast<int>(buttons.size()); i++)
395 if (buttons[i].actor.Get() == a)
396 return i;
397 return -1;
398 }
399
400 bool isMenuActor(vtkActor* a) const { return findButton(a) >= 0; }
401
402 std::vector<vtkActor*> actors() const
403 {
404 std::vector<vtkActor*> result;
405 for (const auto& button : buttons)
406 result.push_back(button.actor);
407 return result;
408 }
409};

411// handles menu button presses - fires when the user hits the menu button on the controller
412class VRMenuCallback : public vtkCommand {
413public:
414 static VRMenuCallback* New() { return new VRMenuCallback; }
415
416 VRRayCallback* rays = nullptr;
417 VRMenu menu;
418 std::map<int, vtkSmartPointer<vtkActor>>* activeActors = nullptr;
419 ModelPartList* partList = nullptr;
420 VRRenderThread* vrThread = nullptr;
421
422 vtkOpenVRRenderWindow* renWindow = nullptr;
423
424 void Execute(vtkObject*, unsigned long, void*) override
425 {
426 // check what the controller is pointing at
427 vtkActor* hovered = nullptr;
428 if (rays->right.hoveredActor)
429 hovered = rays->right.hoveredActor;
430 else if (rays->left.hoveredActor)
431 hovered = rays->left.hoveredActor;
432
433 // if pointing at a menu button, run its action
434 int btnIdx = menu.findButton(hovered);
435 if (btnIdx >= 0) {
436 executeAction(menu.buttons[btnIdx]);
437 return;
438 }
439
440 // if pointing at a part, toggle the menu for that part
441 if (hovered) {
442 int pid = findPartID(hovered);
443 if (pid >= 0) {
444 if (menu.visible && menu.targetPartID == pid) {
445 menu.hide();
446 } else {
447 double* center = hovered->GetCenter();
448 double physScale = 1.0;
449 if (renWindow)
450 physScale = renWindow->GetPhysicalScale();
451 menu.show(center[0], center[1], center[2], pid, physScale);
452 }
453 return;
454 }
455 }
456
457 // pointing at nothing - close menu
458 if (menu.visible)
459 menu.hide();
460 }
461
462private:
463 // looks up which part ID an actor belongs to
464 int findPartID(vtkActor* a) const
465 {
466 if (!activeActors) return -1;
467 for (auto& [id, act] : *activeActors)
468 if (act.Get() == a) return id;
469 return -1;
470 }
471
472 // runs the selected menu action on the target part (VR thread - signals GUI to sync)
473 void executeAction(const VRMenu::Button& btn)
474 {
475 if (!partList || menu.targetPartID < 0) return;
476 ModelPart* part = partList->findByID(menu.targetPartID);
477 if (!part) return;
478
479 switch (btn.action) {
480 case VRMenu::Clip: {
481 bool newState = !part->getClipEnabled();
482 part->m_clipEnabled = newState;
483 part->rebuildVRPipeline();
484 if (vrThread) emit vrThread->partClipChanged(menu.targetPartID, newState);
485 break;
486 }
487 case VRMenu::Shrink: {
488 bool newState = !part->getShrinkEnabled();
489 part->m_shrinkEnabled = newState;
490 part->rebuildVRPipeline();
491 if (vrThread) emit vrThread->partShrinkChanged(menu.targetPartID, newState);
492 break;
493 }
494 case VRMenu::ColRed:
495 applyColourVR(part, QColor(255, 50, 50));
496 break;
497 case VRMenu::ColGreen:
498 applyColourVR(part, QColor(50, 200, 50));
499 break;
500 case VRMenu::ColBlue:
501 applyColourVR(part, QColor(50, 100, 255));
502 break;
503 case VRMenu::ColYellow:
504 applyColourVR(part, QColor(255, 255, 50));
505 break;
506 case VRMenu::ColWhite:
507 applyColourVR(part, QColor(240, 240, 240));
508 break;
509 default:
510 break;
511 }
512 }
513
514 // updates VR actor and signals GUI to apply the same colour on its side
515 void applyColourVR(ModelPart* part, const QColor& c)
516 {
517 part->m_colour = c;
518 vtkActor* vr = part->getVRActor();
519 if (vr)
520 vr->GetProperty()->SetColor(c.redF(), c.greenF(), c.blueF());
521 if (vrThread) emit vrThread->partColourChanged(menu.targetPartID, c);
522 }
523};

562// the main VR loop - sets up OpenVR, adds actors, runs render loop
563void VRRenderThread::run() {
564 m_renderer = vtkSmartPointer<vtkOpenVRRenderer>::New();
565 m_renderWindow = vtkSmartPointer<vtkOpenVRRenderWindow>::New();
566 m_interactor = vtkSmartPointer<vtkOpenVRRenderWindowInteractor>::New();
567
568 // point the interactor at our SteamVR action bindings
569 const std::string manifestPath =
570 (QCoreApplication::applicationDirPath() + "/vrbindings/vtk_openvr_actions.json").toStdString();
571 m_interactor->SetActionManifestFileName(manifestPath.c_str());
572 m_interactor->SetActionSetName("/actions/vtk");
573
574 // custom style: blocks VTK's built-in menu, inverts dolly, hides default rays
575 vtkNew<VRCustomStyle> customStyle;
576 m_interactor->SetInteractorStyle(customStyle);
577
578 m_renderWindow->AddRenderer(m_renderer);
579 m_interactor->SetRenderWindow(m_renderWindow);
580
581 // add all the parts that were queued before the thread started
582 for (auto& [id, actor] : m_pendingActors) {
583 m_renderer->AddActor(actor);
584 m_activeActors[id] = actor;
585 }
586 m_pendingActors.clear();
587
588 m_renderer->ResetCamera();
589 m_renderer->ResetCameraClippingRange();
590
591 // set up dual-controller ray casting + outline highlight
592 vtkNew<VRRayCallback> rayCallback;
593 rayCallback->init(m_renderer);
594 rayCallback->activeActors = &m_activeActors;
595 m_interactor->AddObserver(vtkCommand::Move3DEvent, rayCallback, 1.0);
596
597 // in-VR editing menu (clip / shrink / colour buttons)
598 vtkNew<VRMenuCallback> menuCallback;
599 menuCallback->rays = rayCallback;
600 menuCallback->activeActors = &m_activeActors;
601 menuCallback->partList = m_partList;
602 menuCallback->vrThread = this;
603 menuCallback->renWindow = m_renderWindow;
604 menuCallback->menu.create(m_renderer);
605 rayCallback->menuActors = menuCallback->menu.actors();
606 m_interactor->AddObserver(vtkCommand::UserEvent, menuCallback, 2.0);
607
608 m_renderWindow->Initialize();
609
610 // load 3D garage environment (garage__warehouse.glb or garage.glb) as actual geometry
611 QString glbPath;
612 QStringList glbNames = { "garage__warehouse.glb", "garage.glb" };
613 QStringList searchDirs = {
614 QCoreApplication::applicationDirPath(),
615 QCoreApplication::applicationDirPath() + "/..",
616 QCoreApplication::applicationDirPath() + "/../..",
617 QCoreApplication::applicationDirPath() + "/../../.."
618 };
619 for (const auto& dir : searchDirs) {
620 for (const auto& name : glbNames) {
621 QString candidate = dir + "/" + name;
622 if (QFile::exists(candidate)) { glbPath = candidate; break; }
623 }
624 if (!glbPath.isEmpty()) break;
625 }
626
627 bool garageLoaded = false;
628 bool haveStartupPose = false;
629 double startupPose[3] = { 0.0, 0.0, 0.0 };
630 if (!glbPath.isEmpty()) {
631 vtkNew<vtkGLTFImporter> gltfImporter;
632 gltfImporter->SetFileName(glbPath.toStdString().c_str());
633 gltfImporter->SetRenderWindow(m_renderWindow);
634
635 // import the 3D environment - adds actors, textures, materials, lighting
636 gltfImporter->Update();
637
638 // mark all imported actors as non-pickable (environment only)
639 // and collect them so we can position the environment correctly
640 std::vector<vtkActor*> envActors;
641 vtkActorCollection* actors = m_renderer->GetActors();
642 actors->InitTraversal();
643 vtkActor* a;
644 while ((a = actors->GetNextActor()) != nullptr) {
645 bool isPartActor = false;
646 for (auto& [id, partAct] : m_activeActors) {
647 if (partAct.Get() == a) { isPartActor = true; break; }
648 }
649 if (!isPartActor) {
650 a->PickableOff();
651 envActors.push_back(a);
652 }
653 }
654
655 // position the garage so the user's STL parts sit inside it at a reasonable height
656 // GLTF uses metres, VTK OpenVR also uses metres, so typically no scale is needed.
657 // If the garage is too large or small relative to the STL, adjust it.
658 if (!envActors.empty()) {
659 // compute bounds of all environment geometry
660 double envBounds[6] = { 1e30, -1e30, 1e30, -1e30, 1e30, -1e30 };
661 for (auto* ea : envActors) {
662 double b[6];
663 ea->GetBounds(b);
664 if (b[0] < envBounds[0]) envBounds[0] = b[0];
665 if (b[1] > envBounds[1]) envBounds[1] = b[1];
666 if (b[2] < envBounds[2]) envBounds[2] = b[2];
667 if (b[3] > envBounds[3]) envBounds[3] = b[3];
668 if (b[4] < envBounds[4]) envBounds[4] = b[4];
669 if (b[5] > envBounds[5]) envBounds[5] = b[5];
670 }
671
672 // compute bounds of loaded STL parts
673 double partBounds[6] = { 1e30, -1e30, 1e30, -1e30, 1e30, -1e30 };
674 bool hasParts = false;
675 for (auto& [id, partAct] : m_activeActors) {
676 double b[6];
677 partAct->GetBounds(b);
678 if (b[0] < partBounds[0]) partBounds[0] = b[0];
679 if (b[1] > partBounds[1]) partBounds[1] = b[1];
680 if (b[2] < partBounds[2]) partBounds[2] = b[2];
681 if (b[3] > partBounds[3]) partBounds[3] = b[3];
682 if (b[4] < partBounds[4]) partBounds[4] = b[4];
683 if (b[5] > partBounds[5]) partBounds[5] = b[5];
684 hasParts = true;
685 }
686
687 // if the garage is significantly different scale from parts, rescale it
688 // so the garage wraps around the parts comfortably
689 if (hasParts) {
690 double envSize = std::max({ envBounds[1]-envBounds[0],
691 envBounds[3]-envBounds[2],
692 envBounds[5]-envBounds[4] });
693 double partSize = std::max({ partBounds[1]-partBounds[0],
694 partBounds[3]-partBounds[2],
695 partBounds[5]-partBounds[4] });
696
697 // target: garage should be ~5x bigger than the parts
698 if (envSize > 1e-6 && partSize > 1e-6) {
699 double targetEnvSize = partSize * 5.0;
700 double scaleFactor = targetEnvSize / envSize;
701
702 // only rescale if the difference is significant (>2x off)
703 if (scaleFactor < 0.5 || scaleFactor > 2.0) {
704 for (auto* ea : envActors)
705 ea->SetScale(scaleFactor, scaleFactor, scaleFactor);
706 }
707 }
708
709 // centre the garage around the parts
710 double partCx = (partBounds[0] + partBounds[1]) * 0.5;
711 double partCy = partBounds[2]; // floor = bottom of parts
712 double partCz = (partBounds[4] + partBounds[5]) * 0.5;
713
714 double envCx = (envBounds[0] + envBounds[1]) * 0.5;
715 double envCy = envBounds[2]; // floor of garage
716 double envCz = (envBounds[4] + envBounds[5]) * 0.5;
717
718 double offsetX = partCx - envCx;
719 double offsetY = partCy - envCy;
720 double offsetZ = partCz - envCz;
721
722 for (auto* ea : envActors) {
723 double* pos = ea->GetPosition();
724 ea->SetPosition(pos[0] + offsetX, pos[1] + offsetY, pos[2] + offsetZ);
725 }
726
727 startupPose[0] = partCx;
728 startupPose[1] = partCy;
729 startupPose[2] = partCz;
730 haveStartupPose = true;
731 }
732 }
733
734 garageLoaded = true;
735 }
736
737 // fallback: 2D skybox if no 3D garage available
738 if (!garageLoaded) {
739 QString garagePath2D = QCoreApplication::applicationDirPath() + "/room2.png";
740 QString roomPath = QCoreApplication::applicationDirPath() + "/room.jpg";
741
742 vtkSmartPointer<vtkImageReader2> bgReader;
743 vtkNew<vtkPNGReader> pngReader;
744 vtkNew<vtkJPEGReader> jpgReader;
745
746 if (pngReader->CanReadFile(garagePath2D.toStdString().c_str())) {
747 pngReader->SetFileName(garagePath2D.toStdString().c_str());
748 pngReader->Update();
749 bgReader = pngReader;
750 } else if (jpgReader->CanReadFile(roomPath.toStdString().c_str())) {
751 jpgReader->SetFileName(roomPath.toStdString().c_str());
752 jpgReader->Update();
753 bgReader = jpgReader;
754 }
755
756 if (bgReader) {
757 vtkNew<vtkTexture> bgTexture;
758 bgTexture->SetInputConnection(bgReader->GetOutputPort());
759 bgTexture->InterpolateOn();
760
761 vtkNew<vtkSkybox> skybox;
762 skybox->SetTexture(bgTexture);
763 skybox->SetProjectionToSphere();
764 skybox->PickableOff();
765 m_renderer->AddActor(skybox);
766
767 m_skybox = skybox;
768 }
769 }
770
771 try {
772 m_interactor->Initialize();
773 } catch (...) {
774 // VR input system may not be available
775 }
776
777 if (m_renderer->GetActiveCamera())
778 m_renderer->ResetCameraClippingRange();
779
780 // Place the VR origin inside the loaded scene when the session starts.
781 {
782 double bounds[6];
783 m_renderer->ComputeVisiblePropBounds(bounds);
784 if (bounds[0] <= bounds[1] && bounds[2] <= bounds[3] && bounds[4] <= bounds[5]) {
785 if (!haveStartupPose) {
786 startupPose[0] = (bounds[0] + bounds[1]) * 0.5;
787 startupPose[1] = bounds[2];
788 startupPose[2] = (bounds[4] + bounds[5]) * 0.5;
789 }
790
791 m_renderWindow->SetPhysicalTranslation(
792 startupPose[0], startupPose[1], startupPose[2]);
793 }
794 }
795
796 // render loop - process one VR frame, handle commands, repeat
797 // NOTE: DoOneEvent already submits frames to both eyes.
798 // An extra Render() causes double-submit which results in single-eye flickering.
799 m_endRender = false;
800 while (!m_endRender) {
801 m_interactor->DoOneEvent(m_renderWindow, m_renderer);
802 processCommands();
803
804 }
805
806 m_renderWindow->Finalize();
807 m_renderer = nullptr;
808 m_renderWindow = nullptr;
809 m_interactor = nullptr;
810}

820// executes a single command from the GUI
821void VRRenderThread::applyCommand(const CommandPacket& cmd) {
822 switch (cmd.type) {
823 case Command::EndRender:
824 m_endRender = true;
825 break;
826 case Command::SetColour: {
827 auto it = m_activeActors.find(cmd.partID);
828 if (it != m_activeActors.end()) {
829 QColor c = cmd.data.value<QColor>();
830 it->second->GetProperty()->SetColor(c.redF(), c.greenF(), c.blueF());
831 }
832 break;
833 }
834 case Command::SetVisible: {
835 auto it = m_activeActors.find(cmd.partID);
836 if (it != m_activeActors.end()) {
837 it->second->SetVisibility(cmd.data.toBool() ? 1 : 0);
838 }
839 break;
840 }
841 case Command::RotateY: {
842 auto it = m_activeActors.find(cmd.partID);
843 if (it != m_activeActors.end()) {
844 it->second->RotateY(cmd.data.toDouble());
845 }
846 break;
847 }
848 case Command::SetTransform: {
849 auto it = m_activeActors.find(cmd.partID);
850 if (it != m_activeActors.end()) {
851 QVector3D v = cmd.data.value<QVector3D>();
852 it->second->SetPosition(v.x(), v.y(), v.z());
853 }
854 break;
855 }
856 case Command::ToggleShrink:
857 case Command::ToggleClip: {
858 if (!m_partList) break;
859 ModelPart* p = m_partList->findByID(cmd.partID);
860 if (!p) break;
861 p->rebuildVRPipeline();
862 break;
863 }
864 case Command::AddActor: {
865 if (cmd.actor && m_renderer) {
866 m_renderer->AddActor(cmd.actor);
867 m_activeActors[cmd.partID] = cmd.actor;
868 m_renderer->ResetCameraClippingRange();
869 }
870 break;
871 }
872 case Command::RemoveActor: {
873 auto it = m_activeActors.find(cmd.partID);
874 if (it != m_activeActors.end() && m_renderer) {
875 m_renderer->RemoveActor(it->second);
876 m_activeActors.erase(it);
877 m_renderer->ResetCameraClippingRange();
878 }
879 break;
880 }
881 case Command::ToggleSkybox: {
882 if (m_skybox && m_renderer) {
883 if (cmd.data.toBool())
884 m_renderer->AddActor(m_skybox);
885 else
886 m_renderer->RemoveActor(m_skybox);
887 m_renderer->ResetCameraClippingRange();
888 }
889 break;
890 }
891 }
