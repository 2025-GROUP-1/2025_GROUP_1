94 // HMD-facing movement for any VTK movement events that still reach Dolly3D.
95 void Dolly3D(vtkEventData* ed) override
96 {
97 if (!this->CurrentRenderer)
98 return;
99 auto* rwi = static_cast<vtkRenderWindowInteractor3D*>(this->Interactor);
100 auto* edd = static_cast<vtkEventDataDevice3D*>(ed);
101 vtkCamera* cam = this->CurrentRenderer->GetActiveCamera();
102 if (!cam)
103 return;
104
105 double* trans = rwi->GetPhysicalTranslation(cam);
106 if (!trans)
107 return;
108 if (edd->GetType() == vtkCommand::ViewerMovement3DEvent)
109 edd->GetTrackPadPosition(this->LastTrackPadPosition);
110
111 double speedScale = -this->LastTrackPadPosition[1];
112 double turnAxis = this->LastTrackPadPosition[0];
113 const double turnDeadzone = 0.25;
114 const double moveDeadzone = 0.20;
115 if (std::abs(turnAxis) < turnDeadzone)
116 turnAxis = 0.0;
117 if (std::abs(speedScale) < moveDeadzone)
118 speedScale = 0.0;
119 if (turnAxis == 0.0 && speedScale == 0.0)
120 return;
121
122 double physicalScale = rwi->GetPhysicalScale();
123
124 this->LastDolly3DEventTime->StopTimer();
125 double dt = std::min(this->LastDolly3DEventTime->GetElapsedTime(), 0.1);
126 this->LastDolly3DEventTime->StartTimer();
127
128 double* vd = rwi->GetPhysicalViewDirection();
129 if (vd && turnAxis != 0.0) {
130 const double yaw = turnAxis * 75.0 * dt;
131 const double rad = vtkMath::RadiansFromDegrees(yaw);
132 const double c = std::cos(rad);
133 const double s = std::sin(rad);
134 double nx = c * vd[0] - s * vd[2];
135 double nz = s * vd[0] + c * vd[2];
136 const double len = std::sqrt(nx * nx + vd[1] * vd[1] + nz * nz);
137 if (len > 1e-9)
138 rwi->SetPhysicalViewDirection(nx / len, vd[1] / len, nz / len);
139 }
140
141 double fwd[3];
142 cam->GetDirectionOfProjection(fwd);
143 fwd[1] = 0.0;
144 if (vtkMath::Norm(fwd) <= 1e-9 && vd) {
145 fwd[0] = vd[0];
146 fwd[1] = 0.0;
147 fwd[2] = vd[2];
148 }
149 if (vtkMath::Norm(fwd) <= 1e-9) {
150 fwd[0] = 0.0;
151 fwd[1] = 0.0;
152 fwd[2] = -1.0;
153 }
154 vtkMath::Normalize(fwd);
155
156 double dist = speedScale * this->DollyPhysicalSpeed * physicalScale * dt;
157 rwi->SetPhysicalTranslation(cam,
158 trans[0] + fwd[0] * dist,
159 trans[1],
160 trans[2] + fwd[2] * dist);
161 markDollyMovementHandled();
162
163 if (this->AutoAdjustCameraClippingRange)
164 this->CurrentRenderer->ResetCameraClippingRange();
165 }

214 // fires a pick ray from the controller, updates the ray line endpoint and outline
215 void update(const double* pos, const double* ori, vtkRenderer* ren, double maxRay)
216 {
217 if (!ren) return;
218
219 // work out which direction the controller is pointing
220 vtkNew<vtkTransform> xform;
221 xform->RotateWXYZ(ori[0], ori[1], ori[2], ori[3]);
222 double fwd[3] = { 0.0, 0.0, -1.0 };
223 xform->TransformVector(fwd, fwd);
224
225 double p[3] = { pos[0], pos[1], pos[2] };
226 double o[4] = { ori[0], ori[1], ori[2], ori[3] };
227
228 // try to pick an actor along the ray
229 vtkActor* hit = nullptr;
230 double hitPt[3] = { 0.0, 0.0, 0.0 };
231 try {
232 picker->InitializePickList();
233 for (vtkActor* actor : pickActors) {
234 if (actor)
235 picker->AddPickList(actor);
236 }
237 picker->PickFromListOn();
238 picker->Pick3DRay(p, o, ren);
239 hit = vtkActor::SafeDownCast(picker->GetProp3D());
240 double* pt = picker->GetPickPosition();
241 hitPt[0] = pt[0]; hitPt[1] = pt[1]; hitPt[2] = pt[2];
242 } catch (...) {
243 hit = nullptr;
244 }
245
246 // ray goes from controller to hit point (or extends to max length)
247 rayLine->SetPoint1(pos[0], pos[1], pos[2]);
248 if (hit && picker->GetCellId() >= 0) {
249 rayLine->SetPoint2(hitPt[0], hitPt[1], hitPt[2]);
250 } else {
251 rayLine->SetPoint2(
252 pos[0] + fwd[0] * maxRay,
253 pos[1] + fwd[1] * maxRay,
254 pos[2] + fwd[2] * maxRay);
255 }
256 rayLine->Modified();
257
258 // show an outline box around whatever the ray is pointing at
259 if (hit != hoveredActor) {
260 if (hit && hit->GetMapper() && hit->GetMapper()->GetInput()) {
261 outlineFilter->SetInputData(hit->GetMapper()->GetInput());
262 outlineFilter->Update();
263 outlineActor->SetPosition(hit->GetPosition());
264 outlineActor->SetScale(hit->GetScale());
265 outlineActor->SetOrientation(hit->GetOrientation());
266 outlineActor->SetVisibility(1);
267 } else {
268 outlineActor->SetVisibility(0);
269 }
270 hoveredActor = hit;
271 }
272
273 // keep the outline tracking the actor if it moves
274 if (hoveredActor && outlineActor->GetVisibility()) {
275 outlineActor->SetPosition(hoveredActor->GetPosition());
276 outlineActor->SetScale(hoveredActor->GetScale());
277 outlineActor->SetOrientation(hoveredActor->GetOrientation());
278 }
279 }

281 // same as update() but takes a precomputed forward vector from OpenVR pose.
282 void updateFromForward(const double* pos, const double* fwdIn, vtkRenderer* ren, double maxRay)
283 {
284 if (!ren) return;
285
286 double fwd[3] = { fwdIn[0], fwdIn[1], fwdIn[2] };
287 vtkMath::Normalize(fwd);
288
289 vtkActor* hit = nullptr;
290 double hitPt[3] = { 0.0, 0.0, 0.0 };
291 try {
292 picker->InitializePickList();
293 for (vtkActor* actor : pickActors) {
294 if (actor) picker->AddPickList(actor);
295 }
296 picker->PickFromListOn();
297 double p[3] = { pos[0], pos[1], pos[2] };
298 double o[4] = { 0.0, 0.0, 1.0, 0.0 };
299 picker->Pick3DRay(p, o, ren);
300 hit = vtkActor::SafeDownCast(picker->GetProp3D());
301 double* pt = picker->GetPickPosition();
302 hitPt[0] = pt[0]; hitPt[1] = pt[1]; hitPt[2] = pt[2];
303 } catch (...) {
304 hit = nullptr;
305 }
306
307 rayLine->SetPoint1(pos[0], pos[1], pos[2]);
308 if (hit && picker->GetCellId() >= 0) {
309 rayLine->SetPoint2(hitPt[0], hitPt[1], hitPt[2]);
310 } else {
311 rayLine->SetPoint2(
312 pos[0] + fwd[0] * maxRay,
313 pos[1] + fwd[1] * maxRay,
314 pos[2] + fwd[2] * maxRay);
315 }
316 rayLine->Modified();
317
318 if (hit != hoveredActor) {
319 if (hit && hit->GetMapper() && hit->GetMapper()->GetInput()) {
320 outlineFilter->SetInputData(hit->GetMapper()->GetInput());
321 outlineFilter->Update();
322 outlineActor->SetPosition(hit->GetPosition());
323 outlineActor->SetScale(hit->GetScale());
324 outlineActor->SetOrientation(hit->GetOrientation());
325 outlineActor->SetVisibility(1);
326 } else {
327 outlineActor->SetVisibility(0);
328 }
329 hoveredActor = hit;
330 }
331
332 if (hoveredActor && outlineActor->GetVisibility()) {
333 outlineActor->SetPosition(hoveredActor->GetPosition());
334 outlineActor->SetScale(hoveredActor->GetScale());
335 outlineActor->SetOrientation(hoveredActor->GetOrientation());
336 }
337 }

340// handles Move3D events and updates both controller rays each frame
341class VRRayCallback : public vtkCommand {
342public:
343 static VRRayCallback* New() { return new VRRayCallback; }
344
345 vtkRenderer* renderer = nullptr;
346 std::map<int, vtkSmartPointer<vtkActor>>* activeActors = nullptr;
347 std::vector<vtkActor*> menuActors;
348 ControllerRay right;
349 ControllerRay left;
350
351 void init(vtkRenderer* ren)
352 {
353 renderer = ren;
354 right.init(ren, 0.2, 0.85, 1.0); // cyan for right
355 left.init(ren, 1.0, 0.6, 0.2); // orange for left
356 }
357
358 void Execute(vtkObject*, unsigned long eventId, void* callData) override
359 {
360 if (eventId != vtkCommand::Move3DEvent || !renderer)
361 return;
362
363 auto* ed = static_cast<vtkEventData*>(callData);
364 auto* d3d = ed ? ed->GetAsEventDataDevice3D() : nullptr;
365 if (!d3d) return;
366
367 auto device = d3d->GetDevice();
368 const double* pos = d3d->GetWorldPosition();
369 const double* ori = d3d->GetWorldOrientation();
370 auto pickActors = currentPickActors();
371
372 if (device == vtkEventDataDevice::RightController ||
373 device == vtkEventDataDevice::Unknown ||
374 device == vtkEventDataDevice::Any)
375 {
376 right.setPickActors(pickActors);
377 right.update(pos, ori, renderer, MAX_RAY);
378 }
379
380 if (device == vtkEventDataDevice::LeftController ||
381 device == vtkEventDataDevice::Unknown ||
382 device == vtkEventDataDevice::Any)
383 {
384 left.setPickActors(pickActors);
385 left.update(pos, ori, renderer, MAX_RAY);
386 }
387 }
388
389 // deterministic controller-ray update from OpenVR tracking poses.
390 void tickFromOpenVR(vtkOpenVRRenderWindow* renWin)
391 {
392 if (!renderer || !renWin) return;
393 vr::IVRSystem* hmd = renWin->GetHMD();
394 if (!hmd) return;
395
396 auto pickActors = currentPickActors();
397 right.setPickActors(pickActors);
398 left.setPickActors(pickActors);
399
400 vr::TrackedDevicePose_t poses[vr::k_unMaxTrackedDeviceCount];
401 vr::VRSystem()->GetDeviceToAbsoluteTrackingPose(
402 vr::TrackingUniverseStanding, 0.0f, poses, vr::k_unMaxTrackedDeviceCount);
403
404 auto updateOne = [&](vr::ETrackedControllerRole role, ControllerRay& ray) {
405 vr::TrackedDeviceIndex_t idx = hmd->GetTrackedDeviceIndexForControllerRole(role);
406 if (idx == vr::k_unTrackedDeviceIndexInvalid) return;
407 if (!poses[idx].bPoseIsValid) return;
408 auto& m = poses[idx].mDeviceToAbsoluteTracking;
409 double pos[3] = { m.m[0][3], m.m[1][3], m.m[2][3] };
410 // Controller forward in OpenVR space is -Z axis of the pose matrix.
411 double fwd[3] = { -m.m[0][2], -m.m[1][2], -m.m[2][2] };
412 ray.updateFromForward(pos, fwd, renderer, MAX_RAY);
413 };
414
415 updateOne(vr::TrackedControllerRole_RightHand, right);
416 updateOne(vr::TrackedControllerRole_LeftHand, left);
417 }
418
419private:
420 static constexpr double MAX_RAY = 10000.0; // ray length in mm
421
422 std::vector<vtkActor*> currentPickActors() const
423 {
424 std::vector<vtkActor*> actors;
425 if (activeActors) {
426 for (const auto& [id, actor] : *activeActors) {
427 if (actor)
428 actors.push_back(actor);
429 }
430 }
431 actors.insert(actors.end(), menuActors.begin(), menuActors.end());
432 return actors;
433 }

440struct VRMenu {
441 enum Action { Clip, Shrink, ColRed, ColGreen, ColBlue, ColYellow, ColWhite, NUM_ACTIONS };
442
443 struct Button {
444 vtkSmartPointer<vtkActor> actor;
445 Action action;
446 };
447
448 std::vector<Button> buttons;
449 bool visible = false;
450 int targetPartID = -1;
451 vtkRenderer* ren = nullptr;
452
453 // button sizes (scaled by physical scale at show time)
454 double BTN_W = 60.0;
455 double BTN_H = 30.0;
456 double GAP = 8.0;
457
458 // builds all the menu button actors
459 void create(vtkRenderer* r)
460 {
461 ren = r;
462 addButton(Clip, 1.0, 0.5, 0.0); // orange
463 addButton(Shrink, 0.6, 0.2, 0.8); // purple
464 addButton(ColRed, 1.0, 0.15, 0.15);
465 addButton(ColGreen, 0.15, 0.8, 0.15);
466 addButton(ColBlue, 0.15, 0.4, 1.0);
467 addButton(ColYellow, 1.0, 1.0, 0.2);
468 addButton(ColWhite, 0.9, 0.9, 0.9);
469 }
470
471 void addButton(Action a, double cr, double cg, double cb)
472 {
473 vtkNew<vtkPlaneSource> plane;
474 plane->SetOrigin(0, 0, 0);
475 plane->SetPoint1(BTN_W, 0, 0);
476 plane->SetPoint2(0, BTN_H, 0);
477 plane->Update();
478
479 vtkNew<vtkPolyDataMapper> m;
480 m->SetInputConnection(plane->GetOutputPort());
481
482 vtkSmartPointer<vtkActor> act = vtkSmartPointer<vtkActor>::New();
483 act->SetMapper(m);
484 act->GetProperty()->SetColor(cr, cg, cb);
485 act->GetProperty()->SetAmbient(1.0);
486 act->GetProperty()->SetDiffuse(0.0);
487 act->SetVisibility(0);
488 act->PickableOn();
489 ren->AddActor(act);
490
491 buttons.push_back({act, a});
492 }
493
494 // positions the menu above the given world position, scaled for VR
495 void show(double x, double y, double z, int partID, double physScale = 1.0)
496 {
497 visible = true;
498 targetPartID = partID;
499
500 // scale button sizes based on physical scale so they're always comfortable
501 double scale = physScale * 0.05; // 5% of physical scale gives good button size
502 double bw = BTN_W * scale;
503 double bh = BTN_H * scale;
504 double gap = GAP * scale;
505
506 // resize all button planes
507 for (auto& b : buttons) {
508 b.actor->SetScale(scale, scale, 1.0);
509 }
510
511 // top row: clip + shrink
512 double startX = x - (2 * bw + gap) * 0.5;
513 double startY = y + 40.0 * scale;
514
515 for (int i = 0; i < 2; i++) {
516 buttons[i].actor->SetPosition(
517 startX + i * (bw + gap), startY, z);
518 buttons[i].actor->SetVisibility(1);
519 }
520
521 // bottom row: colour swatches
522 double row2Y = startY - bh - gap;
523 int nCol = static_cast<int>(buttons.size()) - 2;
524 double row2StartX = x - (nCol * bw + (nCol - 1) * gap) * 0.5;
525 for (int i = 2; i < static_cast<int>(buttons.size()); i++) {
526 buttons[i].actor->SetPosition(
527 row2StartX + (i - 2) * (bw + gap), row2Y, z);
528 buttons[i].actor->SetVisibility(1);
529 }
530 }
531
532 void hide()
533 {
534 visible = false;
535 targetPartID = -1;
536 for (auto& b : buttons)
537 b.actor->SetVisibility(0);
538 }
539
540 int findButton(vtkActor* a) const
541 {
542 for (int i = 0; i < static_cast<int>(buttons.size()); i++)
543 if (buttons[i].actor.Get() == a)
544 return i;
545 return -1;
546 }
547
548 bool isMenuActor(vtkActor* a) const { return findButton(a) >= 0; }
549
550 std::vector<vtkActor*> actors() const
551 {
552 std::vector<vtkActor*> result;
553 for (const auto& button : buttons)
554 result.push_back(button.actor);
555 return result;
556 }
557};

559// handles menu button presses - fires when the user hits the menu button on the controller
560class VRMenuCallback : public vtkCommand {
561public:
562 static VRMenuCallback* New() { return new VRMenuCallback; }
563
564 VRRayCallback* rays = nullptr;
565 VRMenu menu;
566 std::map<int, vtkSmartPointer<vtkActor>>* activeActors = nullptr;
567 ModelPartList* partList = nullptr;
568 VRRenderThread* vrThread = nullptr;
569
570 vtkOpenVRRenderWindow* renWindow = nullptr;
571
572 void Execute(vtkObject*, unsigned long, void*) override
573 {
574 // check what the controller is pointing at
575 vtkActor* hovered = nullptr;
576 if (rays->right.hoveredActor)
577 hovered = rays->right.hoveredActor;
578 else if (rays->left.hoveredActor)
579 hovered = rays->left.hoveredActor;
580
581 // if pointing at a menu button, run its action
582 int btnIdx = menu.findButton(hovered);
583 if (btnIdx >= 0) {
584 executeAction(menu.buttons[btnIdx]);
585 return;
586 }
587
588 // if pointing at a part, toggle the menu for that part
589 if (hovered) {
590 int pid = findPartID(hovered);
591 if (pid >= 0) {
592 if (menu.visible && menu.targetPartID == pid) {
593 menu.hide();
594 } else {
595 double* center = hovered->GetCenter();
596 double physScale = 1.0;
597 if (renWindow)
598 physScale = renWindow->GetPhysicalScale();
599 menu.show(center[0], center[1], center[2], pid, physScale);
600 }
601 return;
602 }
603 }
604
605 // pointing at nothing - close menu
606 if (menu.visible)
607 menu.hide();
608 }
609
610private:
611 // looks up which part ID an actor belongs to
612 int findPartID(vtkActor* a) const
613 {
614 if (!activeActors) return -1;
615 for (auto& [id, act] : *activeActors)
616 if (act.Get() == a) return id;
617 return -1;
618 }

620 // runs the selected menu action on the target part (VR thread - signals GUI to sync)
621 void executeAction(const VRMenu::Button& btn)
622 {
623 if (!partList || menu.targetPartID < 0) return;
624 ModelPart* part = partList->findByID(menu.targetPartID);
625 if (!part) return;
626
627 switch (btn.action) {
628 case VRMenu::Clip: {
629 bool newState = !part->getClipEnabled();
630 part->m_clipEnabled = newState;
631 part->rebuildVRPipeline();
632 if (vrThread) emit vrThread->partClipChanged(menu.targetPartID, newState);
633 break;
634 }
635 case VRMenu::Shrink: {
636 bool newState = !part->getShrinkEnabled();
637 part->m_shrinkEnabled = newState;
638 part->rebuildVRPipeline();
639 if (vrThread) emit vrThread->partShrinkChanged(menu.targetPartID, newState);
640 break;
641 }
642 case VRMenu::ColRed:
643 applyColourVR(part, QColor(255, 50, 50));
644 break;
645 case VRMenu::ColGreen:
646 applyColourVR(part, QColor(50, 200, 50));
647 break;
648 case VRMenu::ColBlue:
649 applyColourVR(part, QColor(50, 100, 255));
650 break;
651 case VRMenu::ColYellow:
652 applyColourVR(part, QColor(255, 255, 50));
653 break;
654 case VRMenu::ColWhite:
655 applyColourVR(part, QColor(240, 240, 240));
656 break;
657 default:
658 break;
659 }
660 }
661
662 // updates VR actor and signals GUI to apply the same colour on its side
663 void applyColourVR(ModelPart* part, const QColor& c)
664 {
665 part->m_colour = c;
666 vtkActor* vr = part->getVRActor();
667 if (vr)
668 vr->GetProperty()->SetColor(c.redF(), c.greenF(), c.blueF());
669 if (vrThread) emit vrThread->partColourChanged(menu.targetPartID, c);
670 }

673// ---------------------------------------------------------------------------
674// trigger-based grab (single controller)
675// hold trigger on a hovered part to drag it; release to drop
676// ---------------------------------------------------------------------------
677struct TriggerGrabber {
678 bool active = false;
679 vtkActor* targetActor = nullptr;
680 vr::TrackedDeviceIndex_t activeController = vr::k_unTrackedDeviceIndexInvalid;
681 double localOffset[3] = { 0.0, 0.0, 0.0 };
682
683 bool isPartActor(vtkActor* a, std::map<int, vtkSmartPointer<vtkActor>>* actors)
684 {
685 if (!a || !actors) return false;
686 for (auto& [id, act] : *actors)
687 if (act.Get() == a) return true;
688 return false;
689 }
690
691 static bool isTriggerPressed(const vr::VRControllerState_t& state)
692 {
693 const bool digital = (state.ulButtonPressed & vr::ButtonMaskFromId(vr::k_EButton_SteamVR_Trigger)) != 0;
694 const bool analog = state.rAxis[1].x > 0.55f;
695 return digital || analog;
696 }
697
698 static bool getControllerPos(vr::TrackedDeviceIndex_t idx, vr::TrackedDevicePose_t* poses, double out[3])
699 {
700 if (idx >= vr::k_unMaxTrackedDeviceCount || !poses[idx].bPoseIsValid)
701 return false;
702 auto& m = poses[idx].mDeviceToAbsoluteTracking;
703 out[0] = m.m[0][3];
704 out[1] = m.m[1][3];
705 out[2] = m.m[2][3];
706 return true;
707 }
708
709 void update(vtkOpenVRRenderWindow* renWin, VRRayCallback* rays, std::map<int, vtkSmartPointer<vtkActor>>* activeActors)
710 {
711 vr::IVRSystem* hmd = renWin ? renWin->GetHMD() : nullptr;
712 if (!hmd || !rays || !activeActors) return;
713
714 vr::TrackedDeviceIndex_t leftIdx =
715 hmd->GetTrackedDeviceIndexForControllerRole(vr::TrackedControllerRole_LeftHand);
716 vr::TrackedDeviceIndex_t rightIdx =
717 hmd->GetTrackedDeviceIndexForControllerRole(vr::TrackedControllerRole_RightHand);
718
719 vr::VRControllerState_t lState{}, rState{};
720 bool hasLeft = (leftIdx != vr::k_unTrackedDeviceIndexInvalid) &&
721 hmd->GetControllerState(leftIdx, &lState, sizeof(lState));
722 bool hasRight = (rightIdx != vr::k_unTrackedDeviceIndexInvalid) &&
723 hmd->GetControllerState(rightIdx, &rState, sizeof(rState));
724
725 const bool leftTrigger = hasLeft && isTriggerPressed(lState);
726 const bool rightTrigger = hasRight && isTriggerPressed(rState);
727
728 vr::TrackedDevicePose_t poses[vr::k_unMaxTrackedDeviceCount];
729 vr::VRSystem()->GetDeviceToAbsoluteTrackingPose(
730 vr::TrackingUniverseStanding, 0.0f, poses, vr::k_unMaxTrackedDeviceCount);
731
732 if (active) {
733 bool stillPressed = false;
734 if (activeController == leftIdx) stillPressed = leftTrigger;
735 if (activeController == rightIdx) stillPressed = rightTrigger;
736 if (!stillPressed || !targetActor) {
737 active = false;
738 targetActor = nullptr;
739 activeController = vr::k_unTrackedDeviceIndexInvalid;
740 return;
741 }
742
743 double ctrlPos[3];
744 if (!getControllerPos(activeController, poses, ctrlPos)) {
745 active = false;
746 targetActor = nullptr;
747 activeController = vr::k_unTrackedDeviceIndexInvalid;
748 return;
749 }
750
751 targetActor->SetPosition(
752 ctrlPos[0] + localOffset[0],
753 ctrlPos[1] + localOffset[1],
754 ctrlPos[2] + localOffset[2]);
755 return;
756 }
757
758 if (rightTrigger && rightIdx != vr::k_unTrackedDeviceIndexInvalid &&
759 isPartActor(rays->right.hoveredActor, activeActors))
760 {
761 double ctrlPos[3];
762 if (!getControllerPos(rightIdx, poses, ctrlPos)) return;
763 targetActor = rays->right.hoveredActor;
764 double* p = targetActor->GetPosition();
765 localOffset[0] = p[0] - ctrlPos[0];
766 localOffset[1] = p[1] - ctrlPos[1];
767 localOffset[2] = p[2] - ctrlPos[2];
768 activeController = rightIdx;
769 active = true;
770 return;
771 }
772
773 if (leftTrigger && leftIdx != vr::k_unTrackedDeviceIndexInvalid &&
774 isPartActor(rays->left.hoveredActor, activeActors))
775 {
776 double ctrlPos[3];
777 if (!getControllerPos(leftIdx, poses, ctrlPos)) return;
778 targetActor = rays->left.hoveredActor;
779 double* p = targetActor->GetPosition();
780 localOffset[0] = p[0] - ctrlPos[0];
781 localOffset[1] = p[1] - ctrlPos[1];
782 localOffset[2] = p[2] - ctrlPos[2];
783 activeController = leftIdx;
784 active = true;
785 }
786 }
787};

789// ---------------------------------------------------------------------------
790// click-free locomotion
791// Polls controller axes directly; this avoids VTK's StartMovement action
792// getting stuck until some other button/trigger event wakes it up.
793// ---------------------------------------------------------------------------
794struct PolledLocomotion {
795 vtkNew<vtkTimerLog> timer;
796 bool timerStarted = false;
797
798 static bool movementAxis(vr::IVRSystem* hmd, vr::ETrackedControllerRole role, double axis[2])
799 {
800 axis[0] = 0.0;
801 axis[1] = 0.0;
802 if (!hmd) return false;
803
804 vr::TrackedDeviceIndex_t idx = hmd->GetTrackedDeviceIndexForControllerRole(role);
805 if (idx == vr::k_unTrackedDeviceIndexInvalid) return false;
806
807 vr::VRControllerState_t state{};
808 if (!hmd->GetControllerState(idx, &state, sizeof(state))) return false;
809
810 axis[0] = state.rAxis[0].x;
811 axis[1] = state.rAxis[0].y;
812 return true;
813 }
814
815 void update(vtkOpenVRRenderWindow* renWin, vtkOpenVRRenderWindowInteractor* interactor,
816 vtkRenderer* renderer)
817 {
818 if (!renWin || !interactor || !renderer) return;
819 vtkCamera* cam = renderer->GetActiveCamera();
820 if (!cam) return;
821
822 if (!timerStarted) {
823 timer->StartTimer();
824 timerStarted = true;
825 return;
826 }
827
828 timer->StopTimer();
829 const double dt = std::min(timer->GetElapsedTime(), 0.1);
830 timer->StartTimer();
831
832 // If VTK delivered a movement event this frame, Dolly3D already handled it.
833 if (dollyHandledMovementRecently())
834 return;
835
836 vr::IVRSystem* hmd = renWin->GetHMD();
837 if (!hmd) return;
838
839 double left[2], right[2];
840 const bool hasLeft = movementAxis(hmd, vr::TrackedControllerRole_LeftHand, left);
841 const bool hasRight = movementAxis(hmd, vr::TrackedControllerRole_RightHand, right);
842
843 double axis[2] = { 0.0, 0.0 };
844 if (hasLeft) {
845 axis[0] += left[0];
846 axis[1] += left[1];
847 }
848 if (hasRight) {
849 axis[0] += right[0];
850 axis[1] += right[1];
851 }
852
853 axis[0] = std::clamp(axis[0], -1.0, 1.0);
854 axis[1] = std::clamp(axis[1], -1.0, 1.0);
855
856 const double deadzone = 0.18;
857 if (std::abs(axis[0]) < deadzone) axis[0] = 0.0;
858 if (std::abs(axis[1]) < deadzone) axis[1] = 0.0;
859 if (axis[0] == 0.0 && axis[1] == 0.0) return;
860
861 double* viewDir = interactor->GetPhysicalViewDirection();
862 if (viewDir && axis[0] != 0.0) {
863 const double yaw = axis[0] * 75.0 * dt;
864 const double rad = vtkMath::RadiansFromDegrees(yaw);
865 const double c = std::cos(rad);
866 const double s = std::sin(rad);
867 double nx = c * viewDir[0] - s * viewDir[2];
868 double nz = s * viewDir[0] + c * viewDir[2];
869 const double len = std::sqrt(nx * nx + viewDir[1] * viewDir[1] + nz * nz);
870 if (len > 1e-9) {
871 interactor->SetPhysicalViewDirection(nx / len, viewDir[1] / len, nz / len);
872 }
873 }
874
875 if (axis[1] != 0.0) {
876 double fwd[3];
877 cam->GetDirectionOfProjection(fwd);
878 fwd[1] = 0.0;
879 if (vtkMath::Norm(fwd) <= 1e-9 && viewDir) {
880 fwd[0] = viewDir[0];
881 fwd[1] = 0.0;
882 fwd[2] = viewDir[2];
883 }
884 if (vtkMath::Norm(fwd) <= 1e-9) {
885 fwd[0] = 0.0;
886 fwd[1] = 0.0;
887 fwd[2] = -1.0;
888 }
889 vtkMath::Normalize(fwd);
890
891 double* trans = interactor->GetPhysicalTranslation(cam);
892 if (!trans) return;
893
894 const double dist = -axis[1] * 2.0 * interactor->GetPhysicalScale() * dt;
895 interactor->SetPhysicalTranslation(cam,
896 trans[0] + fwd[0] * dist,
897 trans[1],
898 trans[2] + fwd[2] * dist);
899 renderer->ResetCameraClippingRange();
900 }
901 }
902};

941// the main VR loop - sets up OpenVR, adds actors, runs render loop
942void VRRenderThread::run() {
943 m_renderer = vtkSmartPointer<vtkOpenVRRenderer>::New();
944 m_renderWindow = vtkSmartPointer<vtkOpenVRRenderWindow>::New();
945 m_interactor = vtkSmartPointer<vtkOpenVRRenderWindowInteractor>::New();
946 m_hasGarageAnchor = false;
947 m_garageCenterX = 0.0;
948 m_garageFloorY = 0.0;
949 m_garageCenterZ = 0.0;
950 m_spawnX = 0.0;
951 m_spawnY = 0.0;
952 m_spawnZ = 0.0;
953
954 // point the interactor at our SteamVR action bindings
955 const std::string manifestPath =
956 (QCoreApplication::applicationDirPath() + "/vrbindings/vtk_openvr_actions.json").toStdString();
957 m_interactor->SetActionManifestFileName(manifestPath.c_str());
958 m_interactor->SetActionSetName("/actions/vtk");
959
960 // custom style: blocks VTK's built-in menu, inverts dolly, hides default rays
961 vtkNew<VRCustomStyle> customStyle;
962 m_interactor->SetInteractorStyle(customStyle);
963
964 m_renderWindow->AddRenderer(m_renderer);
965 m_interactor->SetRenderWindow(m_renderWindow);
966
967 // add all the parts that were queued before the thread started
968 for (auto& [id, actor] : m_pendingActors) {
969 m_renderer->AddActor(actor);
970 m_activeActors[id] = actor;
971 }
972 m_pendingActors.clear();
973
974 // set up dual-controller ray casting + outline highlight
975 vtkNew<VRRayCallback> rayCallback;
976 rayCallback->init(m_renderer);
977 rayCallback->activeActors = &m_activeActors;
978 m_interactor->AddObserver(vtkCommand::Move3DEvent, rayCallback, 1.0);
979
980 // in-VR editing menu (clip / shrink / colour buttons)
981 vtkNew<VRMenuCallback> menuCallback;
982 menuCallback->rays = rayCallback;
983 menuCallback->activeActors = &m_activeActors;
984 menuCallback->partList = m_partList;
985 menuCallback->vrThread = this;
986 menuCallback->renWindow = m_renderWindow;
987 menuCallback->menu.create(m_renderer);
988 rayCallback->menuActors = menuCallback->menu.actors();
989 m_interactor->AddObserver(vtkCommand::UserEvent, menuCallback, 2.0);
990
991 m_renderWindow->Initialize();
992
993 // load 3D garage environment (garage__warehouse.glb or garage.glb) as actual geometry
994 QString glbPath;
995 QStringList glbNames = { "garage__warehouse.glb", "garage.glb" };
996 QStringList searchDirs = {
997 QCoreApplication::applicationDirPath(),
998 QCoreApplication::applicationDirPath() + "/..",
999 QCoreApplication::applicationDirPath() + "/../..",
1000 QCoreApplication::applicationDirPath() + "/../../.."
1001 };
1002 for (const auto& dir : searchDirs) {
1003 for (const auto& name : glbNames) {
1004 QString candidate = dir + "/" + name;
1005 if (QFile::exists(candidate)) { glbPath = candidate; break; }
1006 }
1007 if (!glbPath.isEmpty()) break;
1008 }
1009
1010 bool garageLoaded = false;
1011 bool garageHasBounds = false;
1012 double garageBounds[6] = { 0, 0, 0, 0, 0, 0 };
1013 double worldUnitsPerMeter = 1.0;
1014 if (!glbPath.isEmpty()) {
1015 vtkNew<vtkGLTFImporter> gltfImporter;
1016 gltfImporter->SetFileName(glbPath.toStdString().c_str());
1017 gltfImporter->SetRenderWindow(m_renderWindow);
1018
1019 vtkActorCollection* actors = m_renderer->GetActors();
1020 std::unordered_set<vtkActor*> preImportActors;
1021 actors->InitTraversal();
1022 vtkActor* a;
1023 while ((a = actors->GetNextActor()) != nullptr)
1024 preImportActors.insert(a);
1025
1026 // import the 3D environment - adds actors, textures, materials, lighting
1027 gltfImporter->Update();
1028
1029 // identify only newly imported garage actors (exclude helper actors that already existed)
1030 std::vector<vtkActor*> envActors;
1031 actors->InitTraversal();
1032 while ((a = actors->GetNextActor()) != nullptr) {
1033 if (preImportActors.find(a) != preImportActors.end())
1034 continue;
1035 envActors.push_back(a);
1036 a->PickableOff();
1037 a->DragableOff();
1038 }
1039
1040 if (envActors.empty()) {
1041 // Fallback: if actor diffing fails for this GLB, use all non-part actors.
1042 actors->InitTraversal();
1043 while ((a = actors->GetNextActor()) != nullptr) {
1044 bool isPartActor = false;
1045 for (auto& [id, partAct] : m_activeActors) {
1046 if (partAct.Get() == a) { isPartActor = true; break; }
1047 }
1048 if (!isPartActor) {
1049 envActors.push_back(a);
1050 a->PickableOff();
1051 a->DragableOff();
1052 }
1053 }
1054 }
1055
1056 if (!envActors.empty()) {
1057 bool first = true;
1058 for (vtkActor* env : envActors) {
1059 double b[6];
1060 env->GetBounds(b);
1061 if (first) {
1062 for (int i = 0; i < 6; ++i) garageBounds[i] = b[i];
1063 first = false;
1064 } else {
1065 if (b[0] < garageBounds[0]) garageBounds[0] = b[0];
1066 if (b[1] > garageBounds[1]) garageBounds[1] = b[1];
1067 if (b[2] < garageBounds[2]) garageBounds[2] = b[2];
1068 if (b[3] > garageBounds[3]) garageBounds[3] = b[3];
1069 if (b[4] < garageBounds[4]) garageBounds[4] = b[4];
1070 if (b[5] > garageBounds[5]) garageBounds[5] = b[5];
1071 }
1072 }
1073 garageHasBounds = true;
1074 const double garageHeight = garageBounds[3] - garageBounds[2];
1075 // Estimate world-units-per-meter from the garage shell height.
1076 // Typical interior garage height is around 2.6m.
1077 if (garageHeight > 1e-6) {
1078 worldUnitsPerMeter = std::clamp(garageHeight / 2.6, 0.5, 2500.0);
1079 }
1080 }
1081
1082 garageLoaded = true;
1083 }
1084
1085 // fallback: 2D skybox if no 3D garage available
1086 if (!garageLoaded) {
1087 QString garagePath2D = QCoreApplication::applicationDirPath() + "/room2.png";
1088 QString roomPath = QCoreApplication::applicationDirPath() + "/room.jpg";
1089
1090 vtkSmartPointer<vtkImageReader2> bgReader;
1091 vtkNew<vtkPNGReader> pngReader;
1092 vtkNew<vtkJPEGReader> jpgReader;
1093
1094 if (pngReader->CanReadFile(garagePath2D.toStdString().c_str())) {
1095 pngReader->SetFileName(garagePath2D.toStdString().c_str());
1096 pngReader->Update();
1097 bgReader = pngReader;
1098 } else if (jpgReader->CanReadFile(roomPath.toStdString().c_str())) {
1099 jpgReader->SetFileName(roomPath.toStdString().c_str());
1100 jpgReader->Update();
1101 bgReader = jpgReader;
1102 }
1103
1104 if (bgReader) {
1105 vtkNew<vtkTexture> bgTexture;
1106 bgTexture->SetInputConnection(bgReader->GetOutputPort());
1107 bgTexture->InterpolateOn();
1108
1109 vtkNew<vtkSkybox> skybox;
1110 skybox->SetTexture(bgTexture);
1111 skybox->SetProjectionToSphere();
1112 skybox->PickableOff();
1113 m_renderer->AddActor(skybox);
1114
1115 m_skybox = skybox;
1116 }
1117 }
1118
1119 try {
1120 m_interactor->Initialize();
1121 } catch (...) {
1122 // VR input system may not be available
1123 }
1124
1125 // Calibrate user scale so real-world height feels correct relative to scene geometry.
1126 m_interactor->SetPhysicalScale(worldUnitsPerMeter);
1127 constexpr double viewerHeightBoostMeters = 0.25;
1128 const double viewerHeightBoost = viewerHeightBoostMeters * m_interactor->GetPhysicalScale();
1129
1130 if (m_renderer->GetActiveCamera())
1131 m_renderer->ResetCameraClippingRange();
1132
1133 // Set user spawn position from garage bounds (works with or without STL)
1134 if (garageLoaded && garageHasBounds) {
1135 const double garageCx = (garageBounds[0] + garageBounds[1]) * 0.5;
1136 const double garageRawFloorY = garageBounds[2];
1137 const double garageCz = (garageBounds[4] + garageBounds[5]) * 0.5;
1138 const double garageHeight = garageBounds[3] - garageBounds[2];
1139 // Some GLB environments include hidden geometry below the visible floor.
1140 // Lift anchor from raw min-Y to a more usable interior floor estimate.
1141 const double garageFloorY = garageRawFloorY + std::max(0.0, garageHeight * 0.12);
1142 m_hasGarageAnchor = true;
1143 m_garageCenterX = garageCx;
1144 m_garageFloorY = garageFloorY;
1145 m_garageCenterZ = garageCz;
1146 m_spawnX = garageCx;
1147 // In this OpenVR transform, lowering physical origin raises the perceived viewpoint.
1148 m_spawnY = garageFloorY + 0.05 - viewerHeightBoost;
1149 m_spawnZ = garageCz;
1150 } else {
1151 m_spawnX = 0.0;
1152 m_spawnY = -viewerHeightBoost;
1153 m_spawnZ = 0.0;
1154 }
1155
1156 {
1157 auto* rwi3d = static_cast<vtkRenderWindowInteractor3D*>(m_interactor.Get());
1158 rwi3d->SetPhysicalTranslation(m_renderer->GetActiveCamera(),
1159 m_spawnX, m_spawnY, m_spawnZ);
1160 }
1161 m_renderer->ResetCameraClippingRange();
1162
1163 // Place STL actors in front of user if any are loaded
1164 if (garageLoaded && garageHasBounds && !m_activeActors.empty()) {
1165 const double garageWidth = std::max(garageBounds[1] - garageBounds[0], 1e-6);
1166 const double garageDepth = std::max(garageBounds[5] - garageBounds[4], 1e-6);
1167 const double garageMinSpan = std::min(garageWidth, garageDepth);
1168 const double desiredPartSpanMeters = std::clamp((garageMinSpan / m_interactor->GetPhysicalScale()) * 0.16, 0.30, 1.20);
1169 const double desiredPartSpan = desiredPartSpanMeters * m_interactor->GetPhysicalScale();
1170
1171 double targetX = m_spawnX;
1172 double targetY = m_spawnY + 0.55;
1173 double targetZ = m_spawnZ;
1174 if (m_renderWindow && m_renderWindow->GetHMD()) {
1175 vr::TrackedDevicePose_t poses[vr::k_unMaxTrackedDeviceCount];
1176 vr::VRSystem()->GetDeviceToAbsoluteTrackingPose(
1177 vr::TrackingUniverseStanding, 0.0f, poses, vr::k_unMaxTrackedDeviceCount);
1178 const auto hmdIdx = vr::k_unTrackedDeviceIndex_Hmd;
1179 if (poses[hmdIdx].bPoseIsValid) {
1180 const auto& m = poses[hmdIdx].mDeviceToAbsoluteTracking;
1181 double fwd[3] = { -m.m[0][2], -m.m[1][2], -m.m[2][2] };
1182 vtkMath::Normalize(fwd);
1183 targetX = m.m[0][3] + fwd[0] * 1.0;
1184 targetY = std::clamp(static_cast<double>(m.m[1][3]), m_spawnY + 0.35, m_spawnY + 1.10);
1185 targetZ = m.m[2][3] + fwd[2] * 1.0;
1186 }
1187 }
1188 for (auto& [id, partAct] : m_activeActors) {
1189 double b[6];
1190 partAct->GetBounds(b);
1191 const double partCx = (b[0] + b[1]) * 0.5;
1192 const double partCy = (b[2] + b[3]) * 0.5;
1193 const double partCz = (b[4] + b[5]) * 0.5;
1194 const double partMaxSpan = std::max({ b[1] - b[0], b[3] - b[2], b[5] - b[4], 1e-6 });
1195 const double stlScale = std::clamp(desiredPartSpan / partMaxSpan, 0.03, 5.0);
1196 partAct->SetScale(stlScale, stlScale, stlScale);
1197 partAct->SetPosition(
1198 targetX - partCx * stlScale,
1199 targetY - partCy * stlScale,
1200 targetZ - partCz * stlScale);
1201 partAct->SetVisibility(1);
1202 partAct->PickableOn();
1203 }
1204 m_renderer->ResetCameraClippingRange();
1205 }
1206
1207 m_renderer->ResetCameraClippingRange();
1208
1209 TriggerGrabber grabber;
1210 PolledLocomotion locomotion;
1211
1212 // render loop - process one VR frame, handle commands, repeat
1213 // NOTE: DoOneEvent already submits frames to both eyes.
1214 // An extra Render() causes double-submit which results in single-eye flickering.
1215 m_endRender = false;
1216 while (!m_endRender) {
1217 m_interactor->DoOneEvent(m_renderWindow, m_renderer);
1218 locomotion.update(m_renderWindow, m_interactor, m_renderer);
1219 processCommands();
1220 rayCallback->tickFromOpenVR(m_renderWindow);
1221 grabber.update(m_renderWindow, rayCallback, &m_activeActors);
1222 // Hard-lock environment interactivity every frame so only STL parts can move.
1223 vtkActorCollection* allActors = m_renderer->GetActors();
1224 allActors->InitTraversal();
1225 vtkActor* a = nullptr;
1226 while ((a = allActors->GetNextActor()) != nullptr) {
1227 bool isPartActor = false;
1228 for (auto& [id, partAct] : m_activeActors) {
1229 if (partAct.Get() == a) { isPartActor = true; break; }
1230 }
1231 if (isPartActor) {
1232 a->PickableOn();
1233 a->DragableOn();
1234 } else {
1235 a->PickableOff();
1236 a->DragableOff();
1237 }
1238 }
1239
1240 }
1241
1242 m_renderWindow->Finalize();
1243 m_renderer = nullptr;
1244 m_renderWindow = nullptr;
1245 m_interactor = nullptr;
1246}

1256// executes a single command from the GUI
1257void VRRenderThread::applyCommand(const CommandPacket& cmd) {
1258 switch (cmd.type) {
1259 case Command::EndRender:
1260 m_endRender = true;
1261 break;
1262 case Command::SetColour: {
1263 auto it = m_activeActors.find(cmd.partID);
1264 if (it != m_activeActors.end()) {
1265 QColor c = cmd.data.value<QColor>();
1266 it->second->GetProperty()->SetColor(c.redF(), c.greenF(), c.blueF());
1267 }
1268 break;
1269 }
