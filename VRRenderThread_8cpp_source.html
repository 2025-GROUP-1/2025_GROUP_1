38// ---------------------------------------------------------------------------
39// custom VR interactor style
40// - blocks VTK's built-in probe/clip/exit menu
41// - fires a UserEvent instead so our menu callback gets it
42// - kills VTK's default red rays (we draw our own)
43// - inverts dolly direction so trackpad-up = push away
44// ---------------------------------------------------------------------------
45class VRCustomStyle : public vtkOpenVRInteractorStyle {
46public:
47 static VRCustomStyle* New();
48 vtkTypeMacro(VRCustomStyle, vtkOpenVRInteractorStyle);
49
50 // replaces VTK's built-in menu with our custom event
51 void OnMenu3D(vtkEventData* edata) override
52 {
53 if (this->Interactor)
54 this->Interactor->InvokeEvent(vtkCommand::UserEvent, edata);
55 }
56
57 // no-op so VTK never draws its red rays
58 void UpdateRay(vtkEventDataDevice) override {}
59
60 // overrides dolly to negate Y direction (trackpad-up = away, down = towards)
61 void Dolly3D(vtkEventData* ed) override
62 {
63 if (!this->CurrentRenderer)
64 return;
65 auto* rwi = static_cast<vtkRenderWindowInteractor3D*>(this->Interactor);
66 auto* edd = static_cast<vtkEventDataDevice3D*>(ed);
67 const double* wori = edd->GetWorldOrientation();
68
69 // convert controller orientation to a forward vector
70 vtkQuaternion<double> q1;
71 q1.SetRotationAngleAndAxis(vtkMath::RadiansFromDegrees(wori[0]), wori[1], wori[2], wori[3]);
72 double elem[3][3];
73 q1.ToMatrix3x3(elem);
74 double vdir[3] = { 0.0, 0.0, -1.0 };
75 vtkMatrix3x3::MultiplyPoint(elem[0], vdir, vdir);
76
77 double* trans = rwi->GetPhysicalTranslation(this->CurrentRenderer->GetActiveCamera());
78 if (edd->GetType() == vtkCommand::ViewerMovement3DEvent)
79 edd->GetTrackPadPosition(this->LastTrackPadPosition);
80
81 // negate so up = push away, down = pull closer
82 double speedScale = -this->LastTrackPadPosition[1];
83 double physicalScale = rwi->GetPhysicalScale();
84
85 this->LastDolly3DEventTime->StopTimer();
86 double dist = speedScale * this->DollyPhysicalSpeed * physicalScale
87 * this->LastDolly3DEventTime->GetElapsedTime();
88 this->LastDolly3DEventTime->StartTimer();
89
90 rwi->SetPhysicalTranslation(this->CurrentRenderer->GetActiveCamera(),
91 trans[0] - vdir[0] * dist,
92 trans[1] - vdir[1] * dist,
93 trans[2] - vdir[2] * dist);
94
95 if (this->AutoAdjustCameraClippingRange)
96 this->CurrentRenderer->ResetCameraClippingRange();
97 }
98};

101// ---------------------------------------------------------------------------
102// controller ray + outline highlight
103// each controller gets a coloured ray and an outline box around hovered actors
104// ---------------------------------------------------------------------------
105struct ControllerRay {
106 vtkNew<vtkCellPicker> picker;
107 vtkNew<vtkLineSource> rayLine;
108 vtkNew<vtkActor> rayActor;
109 vtkNew<vtkOutlineFilter> outlineFilter;
110 vtkNew<vtkPolyDataMapper> outlineMapper;
111 vtkNew<vtkActor> outlineActor;
112 vtkActor* hoveredActor = nullptr;
113
114 // creates the ray line and outline actors, adds them to the renderer
115 void init(vtkRenderer* ren, double r, double g, double b)
116 {
117 picker->SetTolerance(5.0);
118
119 rayLine->SetPoint1(0, 0, 0);
120 rayLine->SetPoint2(0, 0, -1);
121 vtkNew<vtkPolyDataMapper> rayMapper;
122 rayMapper->SetInputConnection(rayLine->GetOutputPort());
123 rayActor->SetMapper(rayMapper);
124 rayActor->GetProperty()->SetColor(r, g, b);
125 rayActor->GetProperty()->SetLineWidth(2.0);
126 rayActor->GetProperty()->SetOpacity(0.7);
127 rayActor->PickableOff();
128 ren->AddActor(rayActor);
129
130 outlineMapper->SetInputConnection(outlineFilter->GetOutputPort());
131 outlineActor->SetMapper(outlineMapper);
132 outlineActor->GetProperty()->SetColor(0.1, 1.0, 0.4);
133 outlineActor->GetProperty()->SetLineWidth(3.0);
134 outlineActor->PickableOff();
135 outlineActor->SetVisibility(0);
136 ren->AddActor(outlineActor);
137 }
138
139 // fires a pick ray from the controller, updates the ray line endpoint and outline
140 void update(const double* pos, const double* ori, vtkRenderer* ren, double maxRay)
141 {
142 if (!ren) return;
143
144 // work out which direction the controller is pointing
145 vtkNew<vtkTransform> xform;
146 xform->RotateWXYZ(ori[0], ori[1], ori[2], ori[3]);
147 double fwd[3] = { 0.0, 0.0, -1.0 };
148 xform->TransformVector(fwd, fwd);
149
150 double p[3] = { pos[0], pos[1], pos[2] };
151 double o[4] = { ori[0], ori[1], ori[2], ori[3] };
152
153 // try to pick an actor along the ray
154 vtkActor* hit = nullptr;
155 double hitPt[3] = { 0.0, 0.0, 0.0 };
156 try {
157 picker->Pick3DRay(p, o, ren);
158 hit = vtkActor::SafeDownCast(picker->GetProp3D());
159 double* pt = picker->GetPickPosition();
160 hitPt[0] = pt[0]; hitPt[1] = pt[1]; hitPt[2] = pt[2];
161 } catch (...) {
162 hit = nullptr;
163 }
164
165 // ray goes from controller to hit point (or extends to max length)
166 rayLine->SetPoint1(pos[0], pos[1], pos[2]);
167 if (hit && picker->GetCellId() >= 0) {
168 rayLine->SetPoint2(hitPt[0], hitPt[1], hitPt[2]);
169 } else {
170 rayLine->SetPoint2(
171 pos[0] + fwd[0] * maxRay,
172 pos[1] + fwd[1] * maxRay,
173 pos[2] + fwd[2] * maxRay);
174 }
175 rayLine->Modified();
176
177 // show an outline box around whatever the ray is pointing at
178 if (hit != hoveredActor) {
179 if (hit && hit->GetMapper() && hit->GetMapper()->GetInput()) {
180 outlineFilter->SetInputData(hit->GetMapper()->GetInput());
181 outlineFilter->Update();
182 outlineActor->SetPosition(hit->GetPosition());
183 outlineActor->SetScale(hit->GetScale());
184 outlineActor->SetOrientation(hit->GetOrientation());
185 outlineActor->SetVisibility(1);
186 } else {
187 outlineActor->SetVisibility(0);
188 }
189 hoveredActor = hit;
190 }
191
192 // keep the outline tracking the actor if it moves
193 if (hoveredActor && outlineActor->GetVisibility()) {
194 outlineActor->SetPosition(hoveredActor->GetPosition());
195 outlineActor->SetScale(hoveredActor->GetScale());
196 outlineActor->SetOrientation(hoveredActor->GetOrientation());
197 }
198 }
199};

201// handles Move3D events and updates both controller rays each frame
202class VRRayCallback : public vtkCommand {
203public:
204 static VRRayCallback* New() { return new VRRayCallback; }
205
206 vtkRenderer* renderer = nullptr;
207 ControllerRay right;
208 ControllerRay left;
209
210 void init(vtkRenderer* ren)
211 {
212 renderer = ren;
213 right.init(ren, 0.2, 0.85, 1.0); // cyan for right
214 left.init(ren, 1.0, 0.6, 0.2); // orange for left
215 }
216
217 void Execute(vtkObject*, unsigned long eventId, void* callData) override
218 {
219 if (eventId != vtkCommand::Move3DEvent || !renderer)
220 return;
221
222 auto* ed = static_cast<vtkEventData*>(callData);
223 auto* d3d = ed ? ed->GetAsEventDataDevice3D() : nullptr;
224 if (!d3d) return;
225
226 auto device = d3d->GetDevice();
227 const double* pos = d3d->GetWorldPosition();
228 const double* ori = d3d->GetWorldOrientation();
229
230 if (device == vtkEventDataDevice::RightController ||
231 device == vtkEventDataDevice::Unknown ||
232 device == vtkEventDataDevice::Any)
233 {
234 right.update(pos, ori, renderer, MAX_RAY);
235 }
236
237 if (device == vtkEventDataDevice::LeftController ||
238 device == vtkEventDataDevice::Unknown ||
239 device == vtkEventDataDevice::Any)
240 {
241 left.update(pos, ori, renderer, MAX_RAY);
242 }
243 }
244
245private:
246 static constexpr double MAX_RAY = 10000.0; // ray length in mm

250// in-VR editing menu
251// shows coloured buttons above a hovered part for clip/shrink/colour changes
252// ---------------------------------------------------------------------------
253struct VRMenu {
254 enum Action { Clip, Shrink, ColRed, ColGreen, ColBlue, ColYellow, ColWhite, NUM_ACTIONS };
255
256 struct Button {
257 vtkSmartPointer<vtkActor> actor;
258 Action action;
259 };
260
261 std::vector<Button> buttons;
262 bool visible = false;
263 int targetPartID = -1;
264 vtkRenderer* ren = nullptr;
265
266 // button sizes in mm
267 static constexpr double BTN_W = 60.0;
268 static constexpr double BTN_H = 30.0;
269 static constexpr double GAP = 8.0;
270
271 // builds all the menu button actors
272 void create(vtkRenderer* r)
273 {
274 ren = r;
275 addButton(Clip, 1.0, 0.5, 0.0); // orange
276 addButton(Shrink, 0.6, 0.2, 0.8); // purple
277 addButton(ColRed, 1.0, 0.15, 0.15);
278 addButton(ColGreen, 0.15, 0.8, 0.15);
279 addButton(ColBlue, 0.15, 0.4, 1.0);
280 addButton(ColYellow, 1.0, 1.0, 0.2);
281 addButton(ColWhite, 0.9, 0.9, 0.9);
282 }
283
284 void addButton(Action a, double cr, double cg, double cb)
285 {
286 vtkNew<vtkPlaneSource> plane;
287 plane->SetOrigin(0, 0, 0);
288 plane->SetPoint1(BTN_W, 0, 0);
289 plane->SetPoint2(0, BTN_H, 0);
290 plane->Update();
291
292 vtkNew<vtkPolyDataMapper> m;
293 m->SetInputConnection(plane->GetOutputPort());
294
295 vtkSmartPointer<vtkActor> act = vtkSmartPointer<vtkActor>::New();
296 act->SetMapper(m);
297 act->GetProperty()->SetColor(cr, cg, cb);
298 act->GetProperty()->SetAmbient(1.0);
299 act->GetProperty()->SetDiffuse(0.0);
300 act->SetVisibility(0);
301 act->PickableOn();
302 ren->AddActor(act);
303
304 buttons.push_back({act, a});
305 }
306
307 // positions the menu above the given world position
308 void show(double x, double y, double z, int partID)
309 {
310 visible = true;
311 targetPartID = partID;
312
313 // top row: clip + shrink
314 double startX = x - (2 * BTN_W + GAP) * 0.5;
315 double startY = y + 40.0;
316
317 for (int i = 0; i < 2; i++) {
318 buttons[i].actor->SetPosition(
319 startX + i * (BTN_W + GAP), startY, z);
320 buttons[i].actor->SetVisibility(1);
321 }
322
323 // bottom row: colour swatches
324 double row2Y = startY - BTN_H - GAP;
325 int nCol = static_cast<int>(buttons.size()) - 2;
326 double row2StartX = x - (nCol * BTN_W + (nCol - 1) * GAP) * 0.5;
327 for (int i = 2; i < static_cast<int>(buttons.size()); i++) {
328 buttons[i].actor->SetPosition(
329 row2StartX + (i - 2) * (BTN_W + GAP), row2Y, z);
330 buttons[i].actor->SetVisibility(1);
331 }
332 }
333
334 void hide()
335 {
336 visible = false;
337 targetPartID = -1;
338 for (auto& b : buttons)
339 b.actor->SetVisibility(0);
340 }
341
342 int findButton(vtkActor* a) const
343 {
344 for (int i = 0; i < static_cast<int>(buttons.size()); i++)
345 if (buttons[i].actor.Get() == a)
346 return i;
347 return -1;
348 }
349
350 bool isMenuActor(vtkActor* a) const { return findButton(a) >= 0; }
351};

353// handles menu button presses - fires when the user hits the menu button on the controller
354class VRMenuCallback : public vtkCommand {
355public:
356 static VRMenuCallback* New() { return new VRMenuCallback; }
357
358 VRRayCallback* rays = nullptr;
359 VRMenu menu;
360 std::map<int, vtkSmartPointer<vtkActor>>* activeActors = nullptr;
361 ModelPartList* partList = nullptr;
362
363 void Execute(vtkObject*, unsigned long, void*) override
364 {
365 // check what the controller is pointing at
366 vtkActor* hovered = nullptr;
367 if (rays->right.hoveredActor)
368 hovered = rays->right.hoveredActor;
369 else if (rays->left.hoveredActor)
370 hovered = rays->left.hoveredActor;
371
372 // if pointing at a menu button, run its action
373 int btnIdx = menu.findButton(hovered);
374 if (btnIdx >= 0) {
375 executeAction(menu.buttons[btnIdx]);
376 return;
377 }
378
379 // if pointing at a part, toggle the menu for that part
380 if (hovered) {
381 int pid = findPartID(hovered);
382 if (pid >= 0) {
383 if (menu.visible && menu.targetPartID == pid) {
384 menu.hide();
385 } else {
386 double* center = hovered->GetCenter();
387 menu.show(center[0], center[1], center[2], pid);
388 }
389 return;
390 }
391 }
392
393 // pointing at nothing - close menu
394 if (menu.visible)
395 menu.hide();
396 }
397
398private:
399 // looks up which part ID an actor belongs to
400 int findPartID(vtkActor* a) const
401 {
402 if (!activeActors) return -1;
403 for (auto& [id, act] : *activeActors)
404 if (act.Get() == a) return id;
405 return -1;
406 }
407
408 // runs the selected menu action on the target part
409 void executeAction(const VRMenu::Button& btn)
410 {
411 if (!partList || menu.targetPartID < 0) return;
412 ModelPart* part = partList->findByID(menu.targetPartID);
413 if (!part) return;
414
415 switch (btn.action) {
416 case VRMenu::Clip:
417 part->setClipFilter(!part->getClipEnabled());
418 part->rebuildVRPipeline();
419 break;
420 case VRMenu::Shrink:
421 part->setShrinkFilter(!part->getShrinkEnabled());
422 part->rebuildVRPipeline();
423 break;
424 case VRMenu::ColRed:
425 applyColour(part, QColor(255, 50, 50));
426 break;
427 case VRMenu::ColGreen:
428 applyColour(part, QColor(50, 200, 50));
429 break;
430 case VRMenu::ColBlue:
431 applyColour(part, QColor(50, 100, 255));
432 break;
433 case VRMenu::ColYellow:
434 applyColour(part, QColor(255, 255, 50));
435 break;
436 case VRMenu::ColWhite:
437 applyColour(part, QColor(240, 240, 240));
438 break;
439 default:
440 break;
441 }
442 }
443
444 void applyColour(ModelPart* part, const QColor& c)
445 {
446 part->setColour(c);
447 vtkActor* vr = part->getVRActor();
448 if (vr)
449 vr->GetProperty()->SetColor(c.redF(), c.greenF(), c.blueF());
450 }
451};

484// the main VR loop - sets up OpenVR, adds actors, runs render loop
485void VRRenderThread::run() {
486 m_renderer = vtkSmartPointer<vtkOpenVRRenderer>::New();
487 m_renderWindow = vtkSmartPointer<vtkOpenVRRenderWindow>::New();
488 m_interactor = vtkSmartPointer<vtkOpenVRRenderWindowInteractor>::New();
489
490 // point the interactor at our SteamVR action bindings
491 const std::string manifestPath =
492 (QCoreApplication::applicationDirPath() + "/vrbindings/vtk_openvr_actions.json").toStdString();
493 m_interactor->SetActionManifestFileName(manifestPath.c_str());
494 m_interactor->SetActionSetName("/actions/vtk");
495
496 // custom style: blocks VTK's built-in menu, inverts dolly, hides default rays
497 vtkNew<VRCustomStyle> customStyle;
498 m_interactor->SetInteractorStyle(customStyle);
499
500 m_renderWindow->AddRenderer(m_renderer);
501 m_interactor->SetRenderWindow(m_renderWindow);
502
503 // add all the parts that were queued before the thread started
504 for (auto& [id, actor] : m_pendingActors) {
505 m_renderer->AddActor(actor);
506 m_activeActors[id] = actor;
507 }
508 m_pendingActors.clear();
509
510 m_renderer->ResetCamera();
511 m_renderer->ResetCameraClippingRange();
512
513 // set up dual-controller ray casting + outline highlight
514 vtkNew<VRRayCallback> rayCallback;
515 rayCallback->init(m_renderer);
516 m_interactor->AddObserver(vtkCommand::Move3DEvent, rayCallback, 1.0);
517
518 // in-VR editing menu (clip / shrink / colour buttons)
519 vtkNew<VRMenuCallback> menuCallback;
520 menuCallback->rays = rayCallback;
521 menuCallback->activeActors = &m_activeActors;
522 menuCallback->partList = m_partList;
523 menuCallback->menu.create(m_renderer);
524 m_interactor->AddObserver(vtkCommand::UserEvent, menuCallback, 2.0);
525
526 // load the skybox image
527 QString roomPath = QCoreApplication::applicationDirPath() + "/room.jpg";
528 vtkNew<vtkJPEGReader> bgReader;
529
530 if (bgReader->CanReadFile(roomPath.toStdString().c_str())) {
531 bgReader->SetFileName(roomPath.toStdString().c_str());
532 bgReader->Update();
533
534 vtkNew<vtkTexture> bgTexture;
535 bgTexture->SetInputConnection(bgReader->GetOutputPort());
536 bgTexture->InterpolateOn();
537
538 vtkNew<vtkSkybox> skybox;
539 skybox->SetTexture(bgTexture);
540 skybox->SetProjectionToSphere();
541 skybox->PickableOff();
542 m_renderer->AddActor(skybox);
543
544 m_skybox = skybox;
545 }
546
547 m_renderWindow->Initialize();
548
549 try {
550 m_interactor->Initialize();
551 } catch (...) {
552 // VR input system may not be available
553 }
554
555 if (m_renderer->GetActiveCamera())
556 m_renderer->ResetCameraClippingRange();
557
558 // position user standing on the floor, not inside it
559 // shifts VR origin so real-world floor = bottom of scene geometry
560 {
561 double bounds[6];
562 m_renderer->ComputeVisiblePropBounds(bounds);
563 double* trans = m_renderWindow->GetPhysicalTranslation();
564 double floorY = bounds[2]; // Y-min of all geometry
565 m_renderWindow->SetPhysicalTranslation(trans[0], floorY, trans[2]);
566 }
567
568 // render loop - process one VR frame, handle commands, repeat
569 m_endRender = false;
570 while (!m_endRender) {
571 m_interactor->DoOneEvent(m_renderWindow, m_renderer);
572 processCommands();
573 m_renderWindow->Render();
574 }
575
576 m_renderWindow->Finalize();
577 m_renderer = nullptr;
578 m_renderWindow = nullptr;
579 m_interactor = nullptr;
580}

590// executes a single command from the GUI
591void VRRenderThread::applyCommand(const CommandPacket& cmd) {
592 switch (cmd.type) {
593 case Command::EndRender:
594 m_endRender = true;
595 break;
596 case Command::SetColour: {
597 auto it = m_activeActors.find(cmd.partID);
598 if (it != m_activeActors.end()) {
599 QColor c = cmd.data.value<QColor>();
600 it->second->GetProperty()->SetColor(c.redF(), c.greenF(), c.blueF());
601 }
602 break;
603 }
604 case Command::SetVisible: {
605 auto it = m_activeActors.find(cmd.partID);
606 if (it != m_activeActors.end()) {
607 it->second->SetVisibility(cmd.data.toBool() ? 1 : 0);
608 }
609 break;
610 }
611 case Command::RotateY: {
612 auto it = m_activeActors.find(cmd.partID);
613 if (it != m_activeActors.end()) {
614 it->second->RotateY(cmd.data.toDouble());
615 }
616 break;
617 }
618 case Command::SetTransform: {
619 auto it = m_activeActors.find(cmd.partID);
620 if (it != m_activeActors.end()) {
621 QVector3D v = cmd.data.value<QVector3D>();
622 it->second->SetPosition(v.x(), v.y(), v.z());
623 }
624 break;
625 }
626 case Command::ToggleShrink:
627 case Command::ToggleClip: {
628 if (!m_partList) break;
629 ModelPart* p = m_partList->findByID(cmd.partID);
630 if (!p) break;
631 p->rebuildVRPipeline();
632 break;
633 }
634 case Command::AddActor:
635 case Command::RemoveActor:
636 break;
637 case Command::ToggleSkybox: {
638 if (m_skybox && m_renderer) {
639 if (cmd.data.toBool())
640 m_renderer->AddActor(m_skybox);
641 else
642 m_renderer->RemoveActor(m_skybox);
643 m_renderer->ResetCameraClippingRange();
644 }
645 break;
646 }
647 }
