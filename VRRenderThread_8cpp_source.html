129 // HMD-facing movement for any VTK movement events that still reach Dolly3D.
130 void Dolly3D(vtkEventData* ed) override
131 {
132 if (!this->CurrentRenderer)
133 return;
134 auto* rwi = static_cast<vtkRenderWindowInteractor3D*>(this->Interactor);
135 auto* edd = static_cast<vtkEventDataDevice3D*>(ed);
136 vtkCamera* cam = this->CurrentRenderer->GetActiveCamera();
137 if (!cam)
138 return;
139
140 double* trans = rwi->GetPhysicalTranslation(cam);
141 if (!trans)
142 return;
143 if (edd->GetType() == vtkCommand::ViewerMovement3DEvent)
144 edd->GetTrackPadPosition(this->LastTrackPadPosition);
145
146 double speedScale = -this->LastTrackPadPosition[1];
147 double turnAxis = this->LastTrackPadPosition[0];
148 const double turnDeadzone = 0.25;
149 const double moveDeadzone = 0.20;
150 if (std::abs(turnAxis) < turnDeadzone)
151 turnAxis = 0.0;
152 if (std::abs(speedScale) < moveDeadzone)
153 speedScale = 0.0;
154 if (turnAxis == 0.0 && speedScale == 0.0)
155 return;
156
157 double physicalScale = rwi->GetPhysicalScale();
158
159 this->LastDolly3DEventTime->StopTimer();
160 double dt = std::min(this->LastDolly3DEventTime->GetElapsedTime(), 0.1);
161 this->LastDolly3DEventTime->StartTimer();
162
163 double* vd = rwi->GetPhysicalViewDirection();
164 if (vd && turnAxis != 0.0) {
165 const double yaw = turnAxis * 75.0 * dt;
166 const double rad = vtkMath::RadiansFromDegrees(yaw);
167 const double c = std::cos(rad);
168 const double s = std::sin(rad);
169 double nx = c * vd[0] - s * vd[2];
170 double nz = s * vd[0] + c * vd[2];
171 const double len = std::sqrt(nx * nx + vd[1] * vd[1] + nz * nz);
172 if (len > 1e-9)
173 rwi->SetPhysicalViewDirection(nx / len, vd[1] / len, nz / len);
174 }
175
176 double fwd[3];
177 cam->GetDirectionOfProjection(fwd);
178 fwd[1] = 0.0;
179 if (vtkMath::Norm(fwd) <= 1e-9 && vd) {
180 fwd[0] = vd[0];
181 fwd[1] = 0.0;
182 fwd[2] = vd[2];
183 }
184 if (vtkMath::Norm(fwd) <= 1e-9) {
185 fwd[0] = 0.0;
186 fwd[1] = 0.0;
187 fwd[2] = -1.0;
188 }
189 vtkMath::Normalize(fwd);
190
191 double dist = speedScale * this->DollyPhysicalSpeed * physicalScale
192 * g_locomotionScaleCompensation * dt;
193 rwi->SetPhysicalTranslation(cam,
194 trans[0] + fwd[0] * dist,
195 trans[1],
196 trans[2] + fwd[2] * dist);
197 markDollyMovementHandled();
198
199 if (this->AutoAdjustCameraClippingRange)
200 this->CurrentRenderer->ResetCameraClippingRange();
201 }

254 // fires a pick ray from the controller, updates the ray line endpoint and outline
255 void update(const double* pos, const double* ori, vtkRenderer* ren, double maxRay)
256 {
257 if (!ren) return;
258
259 // work out which direction the controller is pointing
260 vtkNew<vtkTransform> xform;
261 xform->RotateWXYZ(ori[0], ori[1], ori[2], ori[3]);
262 double fwd[3] = { 0.0, 0.0, -1.0 };
263 xform->TransformVector(fwd, fwd);
264
265 double p[3] = { pos[0], pos[1], pos[2] };
266 double o[4] = { ori[0], ori[1], ori[2], ori[3] };
267
268 // try to pick an actor along the ray
269 vtkActor* hit = nullptr;
270 double hitPt[3] = { 0.0, 0.0, 0.0 };
271 try {
272 picker->InitializePickList();
273 for (vtkActor* actor : pickActors) {
274 if (actor)
275 picker->AddPickList(actor);
276 }
277 picker->PickFromListOn();
278 picker->Pick3DRay(p, o, ren);
279 hit = vtkActor::SafeDownCast(picker->GetProp3D());
280 double* pt = picker->GetPickPosition();
281 hitPt[0] = pt[0]; hitPt[1] = pt[1]; hitPt[2] = pt[2];
282 } catch (...) {
283 hit = nullptr;
284 }
285
286 // ray goes from controller to hit point (or extends to max length)
287 rayLine->SetPoint1(pos[0], pos[1], pos[2]);
288 if (hit && picker->GetCellId() >= 0) {
289 rayLine->SetPoint2(hitPt[0], hitPt[1], hitPt[2]);
290 } else {
291 rayLine->SetPoint2(
292 pos[0] + fwd[0] * maxRay,
293 pos[1] + fwd[1] * maxRay,
294 pos[2] + fwd[2] * maxRay);
295 }
296 rayLine->Modified();
297
298 // show an outline box around whatever the ray is pointing at
299 if (hit != hoveredActor) {
300 if (hit && hit->GetMapper() && hit->GetMapper()->GetInput()) {
301 outlineFilter->SetInputData(hit->GetMapper()->GetInput());
302 outlineFilter->Update();
303 outlineActor->SetPosition(hit->GetPosition());
304 outlineActor->SetScale(hit->GetScale());
305 outlineActor->SetOrientation(hit->GetOrientation());
306 outlineActor->SetVisibility(1);
307 } else {
308 outlineActor->SetVisibility(0);
309 }
310 hoveredActor = hit;
311 }
312
313 // keep the outline tracking the actor if it moves
314 if (hoveredActor && outlineActor->GetVisibility()) {
315 outlineActor->SetPosition(hoveredActor->GetPosition());
316 outlineActor->SetScale(hoveredActor->GetScale());
317 outlineActor->SetOrientation(hoveredActor->GetOrientation());
318 }

425 device == vtkEventDataDevice::Unknown ||

433 // deterministic controller-ray update from OpenVR tracking poses.
434 void tickFromOpenVR(vtkOpenVRRenderWindow* renWin)
435 {
436 if (!renderer || !renWin) return;
437 vr::IVRSystem* hmd = renWin->GetHMD();
438 if (!hmd) return;
439
440 auto pickActors = currentPickActors();
441 right.setPickActors(pickActors);
442 left.setPickActors(pickActors);
443
444 vr::TrackedDevicePose_t poses[vr::k_unMaxTrackedDeviceCount];
445 vr::VRSystem()->GetDeviceToAbsoluteTrackingPose(
446 vr::TrackingUniverseStanding, 0.0f, poses, vr::k_unMaxTrackedDeviceCount);
447
448 auto updateOne = [&](vr::ETrackedControllerRole role, ControllerRay& ray) {
449 vr::TrackedDeviceIndex_t idx = hmd->GetTrackedDeviceIndexForControllerRole(role);
450 if (idx == vr::k_unTrackedDeviceIndexInvalid) return;
451 if (!poses[idx].bPoseIsValid) return;
452 auto& m = poses[idx].mDeviceToAbsoluteTracking;
453 double pos[3] = { m.m[0][3], m.m[1][3], m.m[2][3] };
454 // Controller forward in OpenVR space is -Z axis of the pose matrix.
455 double fwd[3] = { -m.m[0][2], -m.m[1][2], -m.m[2][2] };
456 ray.updateFromForward(pos, fwd, renderer, MAX_RAY);
457 };
458
459 updateOne(vr::TrackedControllerRole_RightHand, right);
460 updateOne(vr::TrackedControllerRole_LeftHand, left);
461 }

987 static double edgeFunction(double ax, double ay, double bx, double by, double px, double py)
988 {
989 return (px - ax) * (by - ay) - (py - ay) * (bx - ax);
990 }

992 void fillTriangle(vtkImageCanvasSource2D* canvas,
993 int ax, int ay, int bx, int by, int cx, int cy,
994 int minX, int minY, int maxX, int maxY)
995 {
996 const int loX = std::clamp(std::min({ ax, bx, cx }) - 1, minX, maxX);
997 const int hiX = std::clamp(std::max({ ax, bx, cx }) + 1, minX, maxX);
998 const int loY = std::clamp(std::min({ ay, by, cy }) - 1, minY, maxY);
999 const int hiY = std::clamp(std::max({ ay, by, cy }) + 1, minY, maxY);
1000 const double area = edgeFunction(ax, ay, bx, by, cx, cy);
1001 if (std::abs(area) < 1e-6)
1002 return;
1003
1004 for (int y = loY; y <= hiY; ++y) {
1005 int runStart = -1;
1006 for (int x = loX; x <= hiX; ++x) {
1007 const double px = x + 0.5;
1008 const double py = y + 0.5;
1009 const double w0 = edgeFunction(bx, by, cx, cy, px, py);
1010 const double w1 = edgeFunction(cx, cy, ax, ay, px, py);
1011 const double w2 = edgeFunction(ax, ay, bx, by, px, py);
1012 const bool inside = (area > 0.0)
1013 ? (w0 >= 0.0 && w1 >= 0.0 && w2 >= 0.0)
1014 : (w0 <= 0.0 && w1 <= 0.0 && w2 <= 0.0);
1015 if (inside && runStart < 0)
1016 runStart = x;
1017 if ((!inside || x == hiX) && runStart >= 0) {
1018 const int runEnd = inside && x == hiX ? x : x - 1;
1019 canvas->FillBox(runStart, runEnd, y, y);
1020 runStart = -1;
1021 }
1022 }
1023 }

1026 void drawMeshThumbnail(vtkImageCanvasSource2D* canvas, vtkActor* actor,
1027 int x0, int y0, int size, const QColor& baseColour)
1028 {
1029 if (!canvas || !actor || !actor->GetMapper() || !actor->GetMapper()->GetInput())
1030 return;
1031
1032 vtkDataSet* data = actor->GetMapper()->GetInput();
1033 const vtkIdType nPts = data ? data->GetNumberOfPoints() : 0;
1034 if (nPts <= 0)
1035 return;
1036
1037 double minU = 0.0, maxU = 0.0, minV = 0.0, maxV = 0.0;
1038 bool haveBounds = false;
1039 const vtkIdType stride = std::max<vtkIdType>(1, nPts / 1800);
1040 for (vtkIdType i = 0; i < nPts; i += stride) {
1041 double p[3];
1042 data->GetPoint(i, p);
1043 const double u = p[0] - p[2] * 0.42;
1044 const double v = p[1] + p[2] * 0.30;
1045 if (!haveBounds) {
1046 minU = maxU = u;
1047 minV = maxV = v;
1048 haveBounds = true;
1049 } else {
1050 minU = std::min(minU, u);
1051 maxU = std::max(maxU, u);
1052 minV = std::min(minV, v);
1053 maxV = std::max(maxV, v);
1054 }
1055 }
1056 const double spanU = std::max(maxU - minU, 1e-9);
1057 const double spanV = std::max(maxV - minV, 1e-9);
1058 const double scale = (size - 10) / std::max(spanU, spanV);
1059 const double drawW = spanU * scale;
1060 const double drawH = spanV * scale;
1061 const double ox = x0 + (size - drawW) * 0.5;
1062 const double oy = y0 + (size - drawH) * 0.5;
1063
1064 canvas->SetDrawColor(30, 39, 52);
1065 canvas->FillBox(x0 + 2, x0 + size - 2, y0 + 2, y0 + size - 2);
1066 canvas->SetDrawColor(
1067 std::min(baseColour.red() + 54, 255),
1068 std::min(baseColour.green() + 54, 255),
1069 std::min(baseColour.blue() + 54, 255));
1070
1071 auto projectPoint = [&](vtkIdType pointId, int& px, int& py) {
1072 double p[3];
1073 data->GetPoint(pointId, p);
1074 const double u = p[0] - p[2] * 0.42;
1075 const double v = p[1] + p[2] * 0.30;
1076 px = static_cast<int>(std::round(ox + (u - minU) * scale));
1077 py = static_cast<int>(std::round(oy + (v - minV) * scale));
1078 };
1079
1080 const vtkIdType nCells = data->GetNumberOfCells();
1081 const vtkIdType cellStride = std::max<vtkIdType>(1, nCells / 1600);
1082 for (vtkIdType cellId = 0; cellId < nCells; cellId += cellStride) {
1083 vtkCell* cell = data->GetCell(cellId);
1084 if (!cell)
1085 continue;
1086 const vtkIdType nCellPts = cell->GetNumberOfPoints();
1087 if (nCellPts < 3)
1088 continue;
1089 int ax, ay;
1090 projectPoint(cell->GetPointId(0), ax, ay);
1091 for (vtkIdType j = 1; j + 1 < nCellPts; ++j) {
1092 int bx, by, cx, cy;
1093 projectPoint(cell->GetPointId(j), bx, by);
1094 projectPoint(cell->GetPointId(j + 1), cx, cy);
1095 fillTriangle(canvas, ax, ay, bx, by, cx, cy, x0, y0, x0 + size, y0 + size);
1096 }
1097 }
1098
1099 canvas->SetDrawColor(
1100 std::min(baseColour.red() + 88, 255),
1101 std::min(baseColour.green() + 88, 255),
1102 std::min(baseColour.blue() + 88, 255));
1103 for (vtkIdType cellId = 0; cellId < nCells; cellId += std::max<vtkIdType>(1, nCells / 220)) {
1104 vtkCell* cell = data->GetCell(cellId);
1105 if (!cell)
1106 continue;
1107 const vtkIdType nCellPts = cell->GetNumberOfPoints();
1108 if (nCellPts < 2)
1109 continue;
1110 for (vtkIdType j = 0; j < nCellPts; ++j) {
1111 int ax, ay, bx, by;
1112 projectPoint(cell->GetPointId(j), ax, ay);
1113 projectPoint(cell->GetPointId((j + 1) % nCellPts), bx, by);
1114 drawLine(canvas, ax, ay, bx, by, x0, y0, x0 + size, y0 + size);
1115 }
1116 }

1197 HitResult hitTest(const double rayOrigin[3], const double rayDir[3]) const
1198 {
1199 if (!visible)
1200 return {};
1201 const double denom = vtkMath::Dot(panelNormal, rayDir);
1202 if (std::abs(denom) < 1e-9)
1203 return {};
1204 double toPanel[3] = {
1205 panelCenter[0] - rayOrigin[0],
1206 panelCenter[1] - rayOrigin[1],
1207 panelCenter[2] - rayOrigin[2]
1208 };
1209 const double t = vtkMath::Dot(toPanel, panelNormal) / denom;
1210 if (t < 0.01 || t > panelWidth * 3.0)
1211 return {};
1212 double hit[3] = {
1213 rayOrigin[0] + rayDir[0] * t,
1214 rayOrigin[1] + rayDir[1] * t,
1215 rayOrigin[2] + rayDir[2] * t
1216 };
1217 double local[3] = {
1218 hit[0] - panelCenter[0],
1219 hit[1] - panelCenter[1],
1220 hit[2] - panelCenter[2]
1221 };
1222 const double u = vtkMath::Dot(local, panelRight) + panelWidth * 0.5;
1223 const double v = vtkMath::Dot(local, panelUp) + panelHeight * 0.5;
1224 if (u < 0.0 || u > panelWidth || v < 0.0 || v > panelHeight)
1225 return {};
1226
1227 const double px = (u / panelWidth) * TexW;
1228 const double py = (v / panelHeight) * TexH;
1229 for (int i = 0; i < static_cast<int>(partIds.size()); ++i) {
1230 const int row = i / CardCols;
1231 const int col = i % CardCols;
1232 const int x0 = CardStartX + col * (CardSize + CardGap);
1233 const int y0 = CardStartY - row * (CardSize + CardGap);
1234 if (px >= x0 && px <= x0 + CardSize && py >= y0 && py <= y0 + CardSize)
1235 return { HitKind::Part, i, QColor() };
1236 }
1237
1238 if (selectedPartID() >= 0) {

1241 if (px >= 107 && px <= 180 && py >= ControlY - 5 && py <= ControlY + ButtonH + 5)
1242 return { HitKind::Clip, -1, QColor() };
1243 if (px >= 177 && px <= 281 && py >= ControlY - 5 && py <= ControlY + ButtonH + 5)
1244 return { HitKind::Shrink, -1, QColor() };
1245
1246 for (int slot = 0; slot < 5; ++slot) {
1247 const int x0 = SwatchStartX + slot * (SwatchSize + SwatchGap);
1248 if (px < x0 || px > x0 + SwatchSize || py < ControlY || py > ControlY + SwatchSize)
1249 continue;
1250 switch (slot) {
1251 case 0:
1252 return { HitKind::Colour, -1, QColor(255, 50, 50) };
1253 case 1:
1254 return { HitKind::Colour, -1, QColor(50, 200, 50) };
1255 case 2:
1256 return { HitKind::Colour, -1, QColor(50, 110, 255) };
1257 case 3:
1258 return { HitKind::Colour, -1, QColor(255, 230, 50) };
1259 case 4:
1260 return { HitKind::Colour, -1, QColor(235, 235, 235) };
1261 }
1262 }

1270 return {};

1273 double end[3];

1275 line->GetPoint2(end);

1277 end[0] - origin[0],
1278 end[1] - origin[1],
1279 end[2] - origin[2]
1280 };
1281 if (vtkMath::Norm(dir) <= 1e-9)
1282 return {};
1283 vtkMath::Normalize(dir);
1284
1285 return hitTest(origin, dir);

1607struct TriggerGrabber {
1608 bool active = false;
1609 vtkActor* targetActor = nullptr;
1610 vr::TrackedDeviceIndex_t activeController = vr::k_unTrackedDeviceIndexInvalid;
1611 double localOffset[3] = { 0.0, 0.0, 0.0 };

1613 bool isPartActor(vtkActor* a, std::map<int, vtkSmartPointer<vtkActor>>* actors)
1614 {
1615 if (!a || !actors) return false;
1616 for (auto& [id, act] : *actors)
1617 if (act.Get() == a) return true;
1618 return false;
1619 }
1620
1621 static bool isTriggerPressed(const vr::VRControllerState_t& state)
1622 {
1623 const bool digital = (state.ulButtonPressed & vr::ButtonMaskFromId(vr::k_EButton_SteamVR_Trigger)) != 0;
1624 const bool analog = state.rAxis[1].x > 0.55f;
1625 return digital || analog;
1626 }
1627
1628 static bool getControllerPos(vr::TrackedDeviceIndex_t idx, vr::TrackedDevicePose_t* poses, double out[3])
1629 {
1630 if (idx >= vr::k_unMaxTrackedDeviceCount || !poses[idx].bPoseIsValid)
1631 return false;
1632 auto& m = poses[idx].mDeviceToAbsoluteTracking;
1633 out[0] = m.m[0][3];
1634 out[1] = m.m[1][3];
1635 out[2] = m.m[2][3];
1636 return true;
1637 }

1639 void update(vtkOpenVRRenderWindow* renWin, VRRayCallback* rays, std::map<int, vtkSmartPointer<vtkActor>>* activeActors)
1640 {
1641 vr::IVRSystem* hmd = renWin ? renWin->GetHMD() : nullptr;
1642 if (!hmd || !rays || !activeActors) return;
1643
1644 vr::TrackedDeviceIndex_t leftIdx =
1645 hmd->GetTrackedDeviceIndexForControllerRole(vr::TrackedControllerRole_LeftHand);
1646 vr::TrackedDeviceIndex_t rightIdx =
1647 hmd->GetTrackedDeviceIndexForControllerRole(vr::TrackedControllerRole_RightHand);
1648
1649 vr::VRControllerState_t lState{}, rState{};
1650 bool hasLeft = (leftIdx != vr::k_unTrackedDeviceIndexInvalid) &&
1651 hmd->GetControllerState(leftIdx, &lState, sizeof(lState));
1652 bool hasRight = (rightIdx != vr::k_unTrackedDeviceIndexInvalid) &&
1653 hmd->GetControllerState(rightIdx, &rState, sizeof(rState));
1654
1655 const bool leftTrigger = hasLeft && isTriggerPressed(lState);
1656 const bool rightTrigger = hasRight && isTriggerPressed(rState);
1657
1658 vr::TrackedDevicePose_t poses[vr::k_unMaxTrackedDeviceCount];
1659 vr::VRSystem()->GetDeviceToAbsoluteTrackingPose(
1660 vr::TrackingUniverseStanding, 0.0f, poses, vr::k_unMaxTrackedDeviceCount);
1661
1662 if (active) {
1663 bool stillPressed = false;
1664 if (activeController == leftIdx) stillPressed = leftTrigger;
1665 if (activeController == rightIdx) stillPressed = rightTrigger;
1666 if (!stillPressed || !targetActor) {
1667 active = false;
1668 targetActor = nullptr;
1669 activeController = vr::k_unTrackedDeviceIndexInvalid;
1670 return;
1671 }
1672
1673 double ctrlPos[3];
1674 if (!getControllerPos(activeController, poses, ctrlPos)) {
1675 active = false;
1676 targetActor = nullptr;
1677 activeController = vr::k_unTrackedDeviceIndexInvalid;
1678 return;
1679 }
1680
1681 targetActor->SetPosition(
1682 ctrlPos[0] + localOffset[0],
1683 ctrlPos[1] + localOffset[1],
1684 ctrlPos[2] + localOffset[2]);
1685 return;
1686 }
1687
1688 if (rightTrigger && rightIdx != vr::k_unTrackedDeviceIndexInvalid &&
1689 isPartActor(rays->right.hoveredActor, activeActors))
1690 {
1691 double ctrlPos[3];
1692 if (!getControllerPos(rightIdx, poses, ctrlPos)) return;
1693 targetActor = rays->right.hoveredActor;
1694 double* p = targetActor->GetPosition();
1695 localOffset[0] = p[0] - ctrlPos[0];
1696 localOffset[1] = p[1] - ctrlPos[1];
1697 localOffset[2] = p[2] - ctrlPos[2];
1698 activeController = rightIdx;
1699 active = true;
1700 return;
1701 }
1702
1703 if (leftTrigger && leftIdx != vr::k_unTrackedDeviceIndexInvalid &&
1704 isPartActor(rays->left.hoveredActor, activeActors))
1705 {
1706 double ctrlPos[3];
1707 if (!getControllerPos(leftIdx, poses, ctrlPos)) return;
1708 targetActor = rays->left.hoveredActor;
1709 double* p = targetActor->GetPosition();
1710 localOffset[0] = p[0] - ctrlPos[0];
1711 localOffset[1] = p[1] - ctrlPos[1];
1712 localOffset[2] = p[2] - ctrlPos[2];
1713 activeController = leftIdx;
1714 active = true;
1715 }
1716 }

1719// ---------------------------------------------------------------------------
1720// click-free locomotion
1721// Polls controller axes directly; this avoids VTK's StartMovement action
1722// getting stuck until some other button/trigger event wakes it up.
1723// ---------------------------------------------------------------------------
1728struct PolledLocomotion {
1729 vtkNew<vtkTimerLog> timer;
1730 bool timerStarted = false;
1731
1732 static bool movementAxis(vr::IVRSystem* hmd, vr::ETrackedControllerRole role, double axis[2])
1733 {
1734 axis[0] = 0.0;
1735 axis[1] = 0.0;
1736 if (!hmd) return false;
1737
1738 vr::TrackedDeviceIndex_t idx = hmd->GetTrackedDeviceIndexForControllerRole(role);
1739 if (idx == vr::k_unTrackedDeviceIndexInvalid) return false;
1740
1741 vr::VRControllerState_t state{};
1742 if (!hmd->GetControllerState(idx, &state, sizeof(state))) return false;
1743
1744 axis[0] = state.rAxis[0].x;
1745 axis[1] = state.rAxis[0].y;
1746 return true;
1747 }
1748
1749 void update(vtkOpenVRRenderWindow* renWin, vtkOpenVRRenderWindowInteractor* interactor,
1750 vtkRenderer* renderer)
1751 {
1752 if (!renWin || !interactor || !renderer) return;
1753 vtkCamera* cam = renderer->GetActiveCamera();
1754 if (!cam) return;
1755
1756 if (!timerStarted) {
1757 timer->StartTimer();
1758 timerStarted = true;
1759 return;
1760 }
1761
1762 timer->StopTimer();
1763 const double dt = std::min(timer->GetElapsedTime(), 0.1);
1764 timer->StartTimer();
1765
1766 // If VTK delivered a movement event this frame, Dolly3D already handled it.
1767 if (dollyHandledMovementRecently())
1768 return;
1769
1770 vr::IVRSystem* hmd = renWin->GetHMD();
1771 if (!hmd) return;
1772
1773 double left[2], right[2];
1774 const bool hasLeft = movementAxis(hmd, vr::TrackedControllerRole_LeftHand, left);
1775 const bool hasRight = movementAxis(hmd, vr::TrackedControllerRole_RightHand, right);
1776
1777 double axis[2] = { 0.0, 0.0 };
1778 if (hasLeft) {
1779 axis[0] += left[0];
1780 axis[1] += left[1];
1781 }
1782 if (hasRight) {
1783 axis[0] += right[0];
1784 axis[1] += right[1];
1785 }
1786
1787 axis[0] = std::clamp(axis[0], -1.0, 1.0);
1788 axis[1] = std::clamp(axis[1], -1.0, 1.0);
1789
1790 const double deadzone = 0.18;
1791 if (std::abs(axis[0]) < deadzone) axis[0] = 0.0;
1792 if (std::abs(axis[1]) < deadzone) axis[1] = 0.0;
1793 if (axis[0] == 0.0 && axis[1] == 0.0) return;
1794
1795 double* viewDir = interactor->GetPhysicalViewDirection();
1796 if (viewDir && axis[0] != 0.0) {
1797 const double yaw = axis[0] * 75.0 * dt;
1798 const double rad = vtkMath::RadiansFromDegrees(yaw);
1799 const double c = std::cos(rad);
1800 const double s = std::sin(rad);
1801 double nx = c * viewDir[0] - s * viewDir[2];
1802 double nz = s * viewDir[0] + c * viewDir[2];
1803 const double len = std::sqrt(nx * nx + viewDir[1] * viewDir[1] + nz * nz);
1804 if (len > 1e-9) {
1805 interactor->SetPhysicalViewDirection(nx / len, viewDir[1] / len, nz / len);
1806 }
1807 }
1808
1809 if (axis[1] != 0.0) {
1810 double fwd[3];
1811 cam->GetDirectionOfProjection(fwd);
1812 fwd[1] = 0.0;
1813 if (vtkMath::Norm(fwd) <= 1e-9 && viewDir) {
1814 fwd[0] = viewDir[0];
1815 fwd[1] = 0.0;
1816 fwd[2] = viewDir[2];
1817 }
1818 if (vtkMath::Norm(fwd) <= 1e-9) {
1819 fwd[0] = 0.0;
1820 fwd[1] = 0.0;
1821 fwd[2] = -1.0;
1822 }
1823 vtkMath::Normalize(fwd);
1824
1825 double* trans = interactor->GetPhysicalTranslation(cam);
1826 if (!trans) return;
1827
1828 const double dist = -axis[1] * 2.0 * interactor->GetPhysicalScale()
1829 * g_locomotionScaleCompensation * dt;
1830 interactor->SetPhysicalTranslation(cam,
1831 trans[0] + fwd[0] * dist,
1832 trans[1],
1833 trans[2] + fwd[2] * dist);
1834 renderer->ResetCameraClippingRange();
1835 }
1836 }
1837};
