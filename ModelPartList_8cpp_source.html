105// BFS search for a part by its unique ID
106ModelPart* ModelPartList::findByID(int partID) const {
107 QList<ModelPart*> stack;
108 stack.append(rootItem);
109 while (!stack.isEmpty()) {
110 ModelPart* item = stack.takeFirst();
111 if (item != rootItem && item->getID() == partID)
112 return item;
113 for (int i = 0; i < item->childCount(); ++i)
114 stack.append(item->child(i));
115 }
116 return nullptr;
117}
