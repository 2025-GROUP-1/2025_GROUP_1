93void ModelPart::rebuildPipeline() {
94 if (!file)
95 return;
96
97 mapper = vtkSmartPointer<vtkDataSetMapper>::New();
98
99 vtkAlgorithmOutput* source = file->GetOutputPort();
100
101 // Apply clip first (it operates on PolyData from the STL reader directly).
102 if (m_clipEnabled) {
103 // Compute clip plane position relative to the STL bounds so the
104 // slider range maps to the actual model.
105 double bounds[6];
106 file->GetOutput()->GetBounds(bounds);
107 double cx = (bounds[0] + bounds[1]) * 0.5;
108 double xMin = bounds[0];
109 double xMax = bounds[1];
110 // m_clipPlaneX is normalised (-1..1); map it to model space.
111 double planeX = cx + m_clipPlaneX * (xMax - xMin) * 0.5;
112
113 vtkSmartPointer<vtkPlane> plane = vtkSmartPointer<vtkPlane>::New();
114 plane->SetOrigin(planeX, 0.0, 0.0);
115 plane->SetNormal(-1.0, 0.0, 0.0);
116
117 clipFilter = vtkSmartPointer<vtkClipDataSet>::New();
118 clipFilter->SetInputConnection(source);
119 clipFilter->SetClipFunction(plane.Get());
120 clipFilter->Update();
121 source = clipFilter->GetOutputPort();
122 }
123
124 // Shrink filter operates on whatever's upstream.
125 if (m_shrinkEnabled) {
126 shrinkFilter = vtkSmartPointer<vtkShrinkFilter>::New();
127 shrinkFilter->SetInputConnection(source);
128 shrinkFilter->SetShrinkFactor(m_shrinkFactor);
129 shrinkFilter->Update();
130 source = shrinkFilter->GetOutputPort();
131 }
132
133 mapper->SetInputConnection(source);
134
135 if (!actor)
136 actor = vtkSmartPointer<vtkActor>::New();
137
138 actor->SetMapper(mapper);
139 actor->GetProperty()->SetColor(
140 m_colour.redF(), m_colour.greenF(), m_colour.blueF());
141 actor->SetVisibility(m_isVisible ? 1 : 0);
142}

204void ModelPart::applyExplode(double amount) {
205 if (!actor)
206 return;
207
208 double bounds[6];
209 actor->GetBounds(bounds);
210 double dx = bounds[1] - bounds[0];
211 double dy = bounds[3] - bounds[2];
212 double dz = bounds[5] - bounds[4];
213 double diag = std::sqrt(dx * dx + dy * dy + dz * dz);
214
215 double mag = amount * diag * 1.5;
216 actor->SetPosition(
217 m_explodeDir.x() * mag,
218 m_explodeDir.y() * mag,
219 m_explodeDir.z() * mag);
220}
