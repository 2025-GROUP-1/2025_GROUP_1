34QVariant ModelPartList::data( const QModelIndex& index, int role ) const {
35 /* If the item index isnt valid, return a new, empty QVariant (QVariant is generic datatype
36 * that could be any valid QT class) */
37 if( !index.isValid() )
38 return QVariant();
39
40 /* Role represents what this data will be used for, we only need deal with the case
41 * when QT is asking for data to create and display the treeview. Return a new,
42 * empty QVariant if any other request comes through. */
43 if (role != Qt::DisplayRole)
44 return QVariant();
45
46 /* Get a a pointer to the item referred to by the QModelIndex */
47 ModelPart* item = static_cast<ModelPart*>( index.internalPointer() );
48
49 /* Each item in the tree has a number of columns ("Part" and "Visible" in this
50 * initial example) return the column requested by the QModelIndex */
51 return item->data( index.column() );
52}

122QModelIndex ModelPartList::appendChild(QModelIndex& parent, const QList<QVariant>& data) {
123 ModelPart* parentPart;
124

126 parentPart = static_cast<ModelPart*>(parent.internalPointer());
127 else {
128 parentPart = rootItem;

134 ModelPart* childPart = new ModelPart( data, parentPart );

136 parentPart->appendChild(childPart);

140 endInsertRows();
