116void ModelPart::rebuildVRPipeline() {
117 if (!file)
118 return;
119
120 vtkSmartPointer<vtkDataSetMapper> newMapper = vtkSmartPointer<vtkDataSetMapper>::New();
121 vtkAlgorithmOutput* source = file->GetOutputPort();
122
123 // clip filter - cuts geometry along an X-axis plane
124 if (m_clipEnabled) {
125 double bounds[6];
126 file->GetOutput()->GetBounds(bounds);
127 double cx = (bounds[0] + bounds[1]) * 0.5;
128 double xMin = bounds[0], xMax = bounds[1];
129 double planeX = cx + m_clipPlaneX * (xMax - xMin) * 0.5;
130
131 vtkSmartPointer<vtkPlane> plane = vtkSmartPointer<vtkPlane>::New();
132 plane->SetOrigin(planeX, 0.0, 0.0);
133 plane->SetNormal(-1.0, 0.0, 0.0);
134
135 vrClipFilter = vtkSmartPointer<vtkClipDataSet>::New();
136 vrClipFilter->SetInputConnection(source);
137 vrClipFilter->SetClipFunction(plane.Get());
138 vrClipFilter->Update();
139 source = vrClipFilter->GetOutputPort();
140 } else {
141 vrClipFilter = nullptr;
142 }
143
144 // shrink filter - pulls faces inward
145 if (m_shrinkEnabled) {
146 vrShrinkFilter = vtkSmartPointer<vtkShrinkFilter>::New();
147 vrShrinkFilter->SetInputConnection(source);
148 vrShrinkFilter->SetShrinkFactor(m_shrinkFactor);
149 vrShrinkFilter->Update();
150 source = vrShrinkFilter->GetOutputPort();
151 } else {
152 vrShrinkFilter = nullptr;
153 }
154
155 newMapper->SetInputConnection(source);
156 vrMapper = newMapper;
157
158 if (!vrActor)
159 vrActor = vtkSmartPointer<vtkActor>::New();
160
161 vrActor->SetMapper(vrMapper);
162 vrActor->GetProperty()->SetColor(
163 m_colour.redF(), m_colour.greenF(), m_colour.blueF());
164 vrActor->SetVisibility(m_isVisible ? 1 : 0);
165}

178// builds: STL reader -> (clip) -> (shrink) -> mapper -> actor
179void ModelPart::rebuildPipeline() {
180 if (!file)
181 return;
182
183 mapper = vtkSmartPointer<vtkDataSetMapper>::New();
184
185 vtkAlgorithmOutput* source = file->GetOutputPort();
186
187 // clip first since it works on the raw polydata
188 if (m_clipEnabled) {
189 double bounds[6];
190 file->GetOutput()->GetBounds(bounds);
191 double cx = (bounds[0] + bounds[1]) * 0.5;
192 double xMin = bounds[0];
193 double xMax = bounds[1];
194 double planeX = cx + m_clipPlaneX * (xMax - xMin) * 0.5;
195
196 vtkSmartPointer<vtkPlane> plane = vtkSmartPointer<vtkPlane>::New();
197 plane->SetOrigin(planeX, 0.0, 0.0);
198 plane->SetNormal(-1.0, 0.0, 0.0);
199
200 clipFilter = vtkSmartPointer<vtkClipDataSet>::New();
201 clipFilter->SetInputConnection(source);
202 clipFilter->SetClipFunction(plane.Get());
203 clipFilter->Update();
204 source = clipFilter->GetOutputPort();
205 }
206
207 // shrink runs on whatever's upstream
208 if (m_shrinkEnabled) {
209 shrinkFilter = vtkSmartPointer<vtkShrinkFilter>::New();
210 shrinkFilter->SetInputConnection(source);
211 shrinkFilter->SetShrinkFactor(m_shrinkFactor);
212 shrinkFilter->Update();
213 source = shrinkFilter->GetOutputPort();
214 }
215
216 mapper->SetInputConnection(source);
217
218 if (!actor)
219 actor = vtkSmartPointer<vtkActor>::New();
220
221 actor->SetMapper(mapper);
222 actor->GetProperty()->SetColor(
223 m_colour.redF(), m_colour.greenF(), m_colour.blueF());
224 actor->SetVisibility(m_isVisible ? 1 : 0);
225}

297void ModelPart::computeExplodeDirection(double cx, double cy, double cz) {
298 QVector3D scene(static_cast<float>(cx), static_cast<float>(cy), static_cast<float>(cz));
299 m_explodeDir = m_originalCentre - scene;
300 if (m_explodeDir.length() > 0.0001f)
301 m_explodeDir.normalize();

304// shifts the actor position based on the explode amount (0 = normal, 1 = full)
305void ModelPart::applyExplode(double amount) {
306 if (!actor)
307 return;
308
309 double bounds[6];
310 actor->GetBounds(bounds);
311 double dx = bounds[1] - bounds[0];
312 double dy = bounds[3] - bounds[2];
313 double dz = bounds[5] - bounds[4];
314 double diag = std::sqrt(dx * dx + dy * dy + dz * dz);
315
316 double mag = amount * diag * 1.5;
317 actor->SetPosition(
318 m_explodeDir.x() * mag,
319 m_explodeDir.y() * mag,
320 m_explodeDir.z() * mag);
321}
